1//! # hisi-riscv-rt
2//!
3//! Runtime entry support for HiSilicon RISC-V firmware.
4//!
5//! The public crate interface is intentionally small: it re-exports the
6//! `riscv-rt` entry attributes, the selected chip PAC's interrupt enum, and the
7//! single-hart critical-section implementation configured through the `riscv`
8//! crate. Chip-specific reset code, linker fragments, and image headers live
9//! behind startup adapters. Chip interrupt symbols (`device.x`) are owned by the
10//! selected PAC's `rt` feature.
11//!
12//! Current adapters:
13//! - `chip-ws63`: WS63 startup, linker layout, `ws63-pac/rt` interrupt symbols,
14//!   and optional link-time boot header.
15//! - `chip-bs21` + `unstable`: BS2X compatibility path. This crate provides a
16//!   BS21/BS2X default `memory.x`, a BS2X layout adapter, legacy startup, and
17//!   `bs2x-pac/rt` provides `device.x`. BS20/custom boards should provide their
18//!   own `memory.x`.
19//!
20//! Downstream binaries should link with `-Thisi-riscv-link.x`.
21

39// ---- Default startup path: custom asm/ws63/startup.S ----
40#[cfg(all(feature = "chip-ws63", not(feature = "riscv-rt-start-experiment")))]
41core::arch::global_asm!(concat!(
42    ".set __hisi_chip_ws63, 1\n",
43    include_str!("../asm/ws63/startup.S")
44));
45
46// ---- Experimental path: riscv-rt _start + WS63 overrides ----
47// When riscv-rt-start-experiment is enabled, we defer to riscv-rt's _start
48// (which handles GPR zero, GP/SP, .data/.bss, FPU) and provide:
49//   - __pre_init: WS63 stack canary
50//   - _setup_interrupts: WS63 mtvec + MIE bits
51//   - trap_vector + all trap/IRQ handlers (WS63 direct-mode dispatch)
52// The ROM/TCM/SRAM multi-region relocation is still handled by runtime_init,
53// called from __pre_init via our Rust startup adapter.
54// When riscv-rt-start-experiment is enabled, all WS63 startup assembly
55// (__pre_init, _setup_interrupts, trap_vector, default handlers) is compiled
56// via build.rs+cc into a separate .o to avoid LTO symbol conflicts with
57// riscv-rt's weak defaults. See asm/ws63/startup_riscvrt.S.
58
59#[cfg(all(feature = "chip-bs21", not(feature = "riscv-rt-start-experiment")))]
60core::arch::global_asm!(concat!(
61    ".set __hisi_chip_bs2x, 1\n",
62    include_str!("../asm/ws63/startup.S")
63));
