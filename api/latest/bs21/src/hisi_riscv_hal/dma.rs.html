1//! DMA (Direct Memory Access) driver for WS63.
2//!
3//! The WS63 has two DMA controllers:
4//! - **DMA** (MDMA) at 0x4A00_0000 — 4 channels, used for general-purpose transfers
5//! - **SDMA** (Secure DMA) at 0x520A_0000 — 4 additional channels, same register layout
6//!
7//! Each channel supports:
8//! - Memory-to-memory, memory-to-peripheral, peripheral-to-memory transfers
9//! - Linked list (scatter-gather) mode
10//! - Configurable source/destination burst sizes and widths
11//! - Address increment control
12//! - Transfer complete and error interrupts
13//!
14//! # Channel addressing
15//!
16//! The WS63 numbers DMA channels in a single **logical** space:
17//! - **MDMA** (`Dma0`) owns logical channels **0-3**
18//! - **SDMA** (`Sdma0`) owns logical channels **8-11**
19//!
20//! Both are backed by physical channels 0-3 on their own register block, so
21//! `DmaDriver<Sdma0>` accepts channel numbers 8-11 and translates them to the
22//! controller-local 0-3 internally (matching the C SDK `hal_dma_ch_get` /
23//! `hal_dma_type_get` convention in fbb_ws63 `hal_dmac_v151.c`). Passing a
24//! channel outside a controller's logical range panics.
25

35/// DMA instance trait.
36pub trait DmaInstance: sealed::DmaInstanceSealed {
37    /// Returns the PAC pointer for this DMA controller.
38    fn ptr() -> *const crate::soc::pac::dma::RegisterBlock;
39
40    /// Logical channel number of this controller's first physical channel.
41    ///
42    /// MDMA exposes logical channels `CHANNEL_BASE..CHANNEL_BASE+4` (0-3); SDMA
43    /// exposes 8-11. The driver subtracts this base to index the controller's
44    /// physical channels 0-3 — see the module-level "Channel addressing" docs.
45    const CHANNEL_BASE: u8;
46
47    /// Bypass auto clock-gating for this controller, if needed.
48    ///
49    /// The WS63 M_DMA clock is auto-gated when the controller is idle; the
50    /// primary controller must set this bit so its clock keeps running across a
51    /// transfer, otherwise a started transfer never advances (its done bit never
52    /// sets). Mirrors the vendor `dma_port_register_irq()` "BYPASS dma自动门控"
53    /// (`DMA_CLK_AUTO_CTRL_REG |= DMA_CLK_ON_MASK`).
54    fn bypass_auto_clock_gate() {}
55}

85/// Translate a logical channel number to this controller's physical channel
86/// index (0-3), validating it falls in the controller's logical range.
87#[inline]
88fn physical_channel_index(base: u8, channel: u8) -> usize {
89    assert!(channel >= base && channel < base + 4, "DMA channel out of range for this controller");
90    (channel - base) as usize
91}

93/// DMA transfer length in source-width beats.
94///
95/// The DesignWare v151 single-block `trans_size` field is 12 bits, so a single
96/// descriptor can transfer at most 4095 beats. Values are constructed fallibly so
97/// callers cannot accidentally rely on the old silent low-12-bit truncation.
98#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
99#[instability::unstable]
100pub struct DmaTransferSize(u16);
101
102impl DmaTransferSize {
103    /// Maximum number of beats in one single-block DMA transfer.
104    pub const MAX_BEATS: usize = 0x0fff;
105
106    /// Build a transfer size from a beat count, rejecting values that do not fit
107    /// the hardware `trans_size` field.
108    pub const fn from_beats(beats: usize) -> Option<Self> {
109        if beats <= Self::MAX_BEATS { Some(Self(beats as u16)) } else { None }
110    }
111
112    #[inline]
113    const fn get(self) -> u16 {
114        self.0
115    }
116}

118/// DMA sync register mask for the four physical channels of one controller.
119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120#[instability::unstable]
121pub struct DmaSyncMask(u8);
122
123impl DmaSyncMask {
124    /// Build a sync mask. Only bits 0..=3 are valid because each controller has
125    /// four physical channels.
126    pub const fn from_bits(bits: u16) -> Option<Self> {
127        if bits & !0x000f == 0 { Some(Self(bits as u8)) } else { None }
128    }
129
130    /// No channel sync bypass bits set.
131    pub const NONE: Self = Self(0);
132
133    /// All channel sync bypass bits set.
134    pub const ALL: Self = Self(0x0f);
135
136    #[inline]
137    const fn bits(self) -> u16 {
138        self.0 as u16
139    }
140}

224impl DmaChannelConfig {
225    /// Memory-to-memory, 32-bit beat, incrementing on both sides.
226    pub const fn mem_to_mem() -> Self {
227        Self {
228            src_peripheral: 0,
229            dst_peripheral: 0,
230            flow_control: FlowControl::MemToMem,
231            src_width: TransferWidth::Width32,
232            dst_width: TransferWidth::Width32,
233            src_burst: BurstSize::Beats1,
234            dst_burst: BurstSize::Beats1,
235            src_inc: true,
236            dst_inc: true,
237            transfer_int: false,
238            error_int: false,
239            bus_lock: false,
240        }
241    }
242
243    #[inline]
244    fn control_word(self, transfer_size: DmaTransferSize) -> u32 {
245        let mut control: u32 = 0;
246        control |= (transfer_size.get() as u32) & 0x0fff; // trans_size [0:11]
247        control |= ((self.src_burst as u32) & 0x07) << 12; // s_bsize [12:14]
248        control |= ((self.dst_burst as u32) & 0x07) << 15; // d_bsize [15:17]
249        control |= ((self.src_width as u32) & 0x07) << 18; // s_width [18:20]
250        control |= ((self.dst_width as u32) & 0x07) << 21; // d_width [21:23]
251        if self.src_inc {
252            control |= 1 << 26;
253        }
254        if self.dst_inc {
255            control |= 1 << 27;
256        }
257        if self.transfer_int {
258            control |= 1 << 31;
259        }
260        control
261    }
262
263    #[inline]
264    fn channel_config_word(self) -> u32 {
265        let mut ch_cfg: u32 = 0;
266        ch_cfg |= 0x01; // chn_en
267        ch_cfg |= ((self.src_peripheral as u32) & 0x0f) << 1; // s_peripheral [1:4]
268        ch_cfg |= ((self.dst_peripheral as u32) & 0x0f) << 5; // d_peripheral [5:8]
269        ch_cfg |= ((self.flow_control as u32) & 0x07) << 9; // flow_ctl [9:11]
270        if self.error_int {
271            ch_cfg |= 1 << 12;
272        }
273        if self.transfer_int {
274            ch_cfg |= 1 << 13;
275        }
276        if self.bus_lock {
277            ch_cfg |= 1 << 14;
278        }
279        ch_cfg
280    }
281}

290impl<'d, T: DmaInstance> DmaDriver<'d, T> {
291    #[cfg(all(test, not(target_arch = "riscv32")))]
292    fn new_for_test() -> Self {
293        Self { _instance: PhantomData }
294    }
295
296    fn regs() -> &'static crate::soc::pac::dma::RegisterBlock {
297        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid
298        unsafe { &*T::ptr() }
299    }
300
301    /// Translate a logical channel number (0-3 for MDMA, 8-11 for SDMA) to this
302    /// controller's physical channel index (0-3). Panics if out of range.
303    #[inline]
304    fn physical_channel(channel: u8) -> usize {
305        physical_channel_index(T::CHANNEL_BASE, channel)
306    }
307
308    /// Enable the DMA controller.
309    ///
310    /// First bypasses the controller's auto clock-gate (if it has one — the WS63
311    /// M_DMA does), so the DMA clock stays running across a transfer. Without
312    /// this the idle-gated clock leaves a started transfer stuck (done bit never
313    /// sets); this was why `dma_mem_to_mem` failed on silicon while passing under
314    /// QEMU, which models no clock gating. Mirrors the vendor `dma_porting`.
315    pub fn enable_controller(&mut self) {
316        T::bypass_auto_clock_gate();
317        let r = Self::regs();
318        unsafe {
319            r.dmac_config().write(|w| w.bits(0x01));
320        }
321    }
322
323    /// Disable the DMA controller.
324    pub fn disable_controller(&mut self) {
325        unsafe {
326            Self::regs().dmac_config().write(|w| w.bits(0));
327        }
328    }
329
330    /// Configure a DMA channel.
331    ///
332    /// * `channel` — Logical channel number (0-3 for MDMA, 8-11 for SDMA).
333    /// * `src_addr` — Source address.
334    /// * `dst_addr` — Destination address.
335    /// * `transfer_size` — Number of source-width beats to transfer.
336    /// * `config` — Channel configuration.
337    pub(crate) fn configure_channel_raw(
338        &mut self,
339        channel: u8,
340        src_addr: u32,
341        dst_addr: u32,
342        transfer_size: DmaTransferSize,
343        config: &DmaChannelConfig,
344    ) {
345        let ch = Self::physical_channel(channel);
346        let r = Self::regs();
347
348        // Disable channel before configuration
349        unsafe {
350            r.dmac_chn_config_0(ch).write(|w| w.bits(0));
351        }
352
353        // Set source address
354        unsafe {
355            r.dmac_s_addr_0(ch).write(|w| w.bits(src_addr));
356        }
357
358        // Set destination address
359        unsafe {
360            r.dmac_d_addr_0(ch).write(|w| w.bits(dst_addr));
361        }
362
363        // Clear linked list pointer
364        unsafe {
365            r.dmac_lli_0(ch).write(|w| w.bits(0));
366        }
367
368        let control = config.control_word(transfer_size);
369        unsafe {
370            r.dmac_chn_control_0(ch).write(|w| w.bits(control));
371        }
372
373        let ch_cfg = config.channel_config_word();
374        unsafe {
375            r.dmac_chn_config_0(ch).write(|w| w.bits(ch_cfg));
376        }
377
378        // Actually START the transfer: set this channel's bit in the global
379        // channel-enable register (`dmac_en_chns`, the DesignWare ChEnReg). On
380        // real silicon the per-channel CFG `ch_enable` bit alone does NOT kick the
381        // engine — the vendor `hal_dma_v151_enable` writes `en_chns`, and the
382        // hardware auto-clears the bit when the (single-block) transfer completes.
383        // (QEMU runs its synchronous memcpy on the CFG write and may not model
384        // `en_chns`, so this is a harmless extra write there.) Completion is then
385        // observed via `channel_enabled()` going false — see the HIL test.
386        let en = r.dmac_en_chns().read().bits();
387        unsafe {
388            r.dmac_en_chns().write(|w| w.bits(en | (1 << ch)));
389        }
390    }
391
392    pub(crate) fn enable_channel_raw(&mut self, channel: u8) {
393        let ch = Self::physical_channel(channel);
394        let r = Self::regs();
395        let cfg = r.dmac_chn_config_0(ch).read().bits();
396        unsafe {
397            r.dmac_chn_config_0(ch).write(|w| w.bits(cfg | 0x01));
398        }
399    }
400
401    pub(crate) fn disable_channel_raw(&mut self, channel: u8) {
402        let ch = Self::physical_channel(channel);
403        let r = Self::regs();
404        let cfg = r.dmac_chn_config_0(ch).read().bits();
405        unsafe {
406            r.dmac_chn_config_0(ch).write(|w| w.bits(cfg & !0x01));
407        }
408    }
409
410    pub(crate) fn channel_enabled_raw(&self, channel: u8) -> bool {
411        let ch = Self::physical_channel(channel);
412        let mask = 1u32 << ch;
413        Self::regs().dmac_en_chns().read().bits() & mask != 0
414    }
415
416    pub(crate) fn channel_active_raw(&self, channel: u8) -> bool {
417        let ch = Self::physical_channel(channel);
418        Self::regs().dmac_chn_config_0(ch).read().bits() & (1 << 15) != 0
419    }
420
421    pub(crate) fn halt_channel_raw(&mut self, channel: u8) {
422        let ch = Self::physical_channel(channel);
423        let r = Self::regs();
424        let cfg = r.dmac_chn_config_0(ch).read().bits();
425        unsafe {
426            r.dmac_chn_config_0(ch).write(|w| w.bits(cfg | (1 << 16)));
427        }
428    }
429
430    pub(crate) fn resume_channel_raw(&mut self, channel: u8) {
431        let ch = Self::physical_channel(channel);
432        let r = Self::regs();
433        let cfg = r.dmac_chn_config_0(ch).read().bits();
434        unsafe {
435            r.dmac_chn_config_0(ch).write(|w| w.bits(cfg & !(1 << 16)));
436        }
437    }
438
439    pub(crate) fn burst_request_raw(&mut self, channel: u8) {
440        let ch = Self::physical_channel(channel);
441        unsafe {
442            Self::regs().dmac_burst_req().write(|w| w.bits(1 << ch));
443        }
444    }
445
446    pub(crate) fn single_request_raw(&mut self, channel: u8) {
447        let ch = Self::physical_channel(channel);
448        unsafe {
449            Self::regs().dmac_single_req().write(|w| w.bits(1 << ch));
450        }
451    }
452
453    /// Get the raw interrupt status.
454    ///
455    /// Returns `(transfer_done_mask, error_mask)`. Bit `n` of each mask is the
456    /// **physical** channel `n` of this controller — i.e. logical channel
457    /// `CHANNEL_BASE + n`. For SDMA, logical channel 8 is bit 0, channel 9 is
458    /// bit 1, etc. (the controller's per-channel status registers are local).
459    pub fn raw_interrupt_status(&self) -> (u8, u8) {
460        let sts = Self::regs().dmac_ori_int_st().read().bits();
461        ((sts & 0xFF) as u8, ((sts >> 8) & 0xFF) as u8)
462    }
463
464    /// Get the masked interrupt status.
465    ///
466    /// Returns `(transfer_done_mask, error_mask)`, with the same **physical**
467    /// channel bit indexing as [`raw_interrupt_status`](Self::raw_interrupt_status)
468    /// (bit `n` = physical channel `n` = logical `CHANNEL_BASE + n`).
469    pub fn interrupt_status(&self) -> (u8, u8) {
470        let sts = Self::regs().dmac_int_st().read().bits();
471        ((sts & 0xFF) as u8, ((sts >> 16) & 0xFF) as u8)
472    }
473
474    pub(crate) fn clear_transfer_interrupt_raw(&mut self, channel: u8) {
475        let ch = Self::physical_channel(channel);
476        unsafe {
477            Self::regs().dmac_int_clr().write(|w| w.bits(1 << ch));
478        }
479    }
480
481    pub(crate) fn clear_error_interrupt_raw(&mut self, channel: u8) {
482        let ch = Self::physical_channel(channel);
483        unsafe {
484            Self::regs().dmac_int_clr().write(|w| w.bits(1 << (ch + 8)));
485        }
486    }
487
488    /// Set DMA sync configuration.
489    ///
490    /// Each bit controls sync for the corresponding channel
491    /// (0 = enable sync logic, 1 = disable sync logic).
492    #[instability::unstable]
493    pub fn set_sync(&mut self, sync_mask: DmaSyncMask) {
494        unsafe {
495            Self::regs().dmac_sync().write(|w| w.bits(sync_mask.bits() as u32));
496        }
497    }
498}

511/// DMA peripheral hardware-handshaking request ID.
512///
513/// Values are the `HAL_DMA_HANDSHAKING_*` indices from fbb_ws63
514/// `drivers/chips/ws63/porting/dma/dma_porting.h` — the hardware request line a
515/// channel uses for peripheral-paced flow control. They go into the channel
516/// config's `src_peripheral` / `dst_peripheral` field (a 4-bit field; all the
517/// IDs below fit). UART bus mapping per `platform_core.h`: UART0 = UART_L,
518/// UART1 = UART_H0, UART2 = UART_H1.
519///
520/// (These superseded the earlier fabricated sequential 0..11 values; only the
521/// main-DMA (MDMA) sources hisi-riscv-hal models are listed — the SDMA-group I2C IDs
522/// (≥29) don't fit the 4-bit field and aren't modelled here.)
523// Peripheral-paced DMA request IDs are WS63-specific (the fbb_ws63 vs fbb_bs2x
524// dma_porting.h handshake enums are entirely different). chip-ws63 only; BS2X DMA
525// is enabled for memory-to-memory transfers, whose flow control needs no request
526// ID. A BS2X request-ID table can be added later from fbb_bs2x dma_porting.h.
527#[cfg(feature = "chip-ws63")]
528#[derive(Debug, Clone, Copy, PartialEq, Eq)]
529#[repr(u8)]
530pub enum DmaPeripheral {
531    /// No handshaking (tie-off) — used for memory-to-memory transfers.
532    Tie0 = 0,
533    /// UART0 (UART_L) transmit.
534    Uart0Tx = 1,
535    /// UART0 (UART_L) receive.
536    Uart0Rx = 2,
537    /// UART1 (UART_H0) transmit.
538    Uart1Tx = 3,
539    /// UART1 (UART_H0) receive.
540    Uart1Rx = 4,
541    /// UART2 (UART_H1) transmit.
542    Uart2Tx = 5,
543    /// UART2 (UART_H1) receive.
544    Uart2Rx = 6,
545    /// SPI0 (SPI_MS0) transmit.
546    Spi0Tx = 7,
547    /// SPI0 (SPI_MS0) receive.
548    Spi0Rx = 8,
549    /// I2S transmit.
550    I2sTx = 11,
551    /// I2S receive.
552    I2sRx = 12,
553    /// SPI1 (SPI_MS1) transmit.
554    ///
555    /// NOTE: the PAC's `Spi1` instance (0x4402_1000) may actually be QSPI0_2CS
556    /// (vendor `HAL_DMA_HANDSHAKING_QSPI0_2CS_TX` = 9) per `spi_porting.h:38` —
557    /// silicon-unverified. The correct `Qspi02csTx = 9` variant is deferred to P2
558    /// (when the SPI1 DMA wiring is silicon-verified) so an unverified breaking
559    /// enum addition isn't shipped in P0.
560    Spi1Tx = 13,
561    /// SPI1 (SPI_MS1) receive. See [`Spi1Tx`](Self::Spi1Tx) — QSPI0_2CS (10) unverified.
562    Spi1Rx = 14,
563}

627use core::mem::ManuallyDrop;
628use embedded_dma::{ReadBuffer, WriteBuffer};

630/// The DMA cache-maintenance granularity (the 32-byte D-cache line). Source and
631/// destination buffers should be [`DMA_ALIGN`]-aligned so a partial-line invalidate
632/// on `wait()` cannot clobber a neighbouring allocation's dirty data (see
633/// [`crate::cache::invalidate_range`]).
634pub const DMA_ALIGN: usize = 32;
635
636/// Bounded completion poll for [`Transfer::wait`] — a wedged channel returns control
637/// (the transfer is then treated as done) instead of spinning the CPU forever.
638const DMA_WAIT_LOOPS: u32 = 5_000_000;
639
640impl<'d> DmaDriver<'d, Dma0> {
641    /// Launch a **memory-to-memory** transfer on `channel` that OWNS both buffers
642    /// for its whole lifetime, returning a [`Transfer`] guard you [`wait`] to reclaim
643    /// the driver + buffers.
644    ///
645    /// Because the guard owns the buffers (and aborts the channel on `Drop`), a
646    /// use-after-free of an in-flight DMA region is **unrepresentable in safe code**:
647    /// the buffers cannot be touched, moved, or freed while the DMA reads/writes them.
648    /// The source cache lines are cleaned before launch and the destination
649    /// invalidated on `wait()` (the WS63 core is non-coherent), so the cache
650    /// maintenance lives inside the type rather than being a caller obligation.
651    ///
652    /// The DMA beat is a 32-bit word (`Word = u32`); buffers should be
653    /// [`DMA_ALIGN`]-aligned. The transfer length is `min(src.len(), dst.len())` words.
654    ///
655    /// [`wait`]: Transfer::wait
656    #[instability::unstable]
657    pub fn start_mem_to_mem<SRC, DST>(
658        mut self,
659        channel: DmaChannel,
660        src: SRC,
661        mut dst: DST,
662    ) -> Result<Transfer<'d, SRC, DST>, DmaStartError<'d, SRC, DST>>
663    where
664        SRC: ReadBuffer<Word = u32>,
665        DST: WriteBuffer<Word = u32>,
666    {
667        // SAFETY: `src`/`dst` are moved into the returned Transfer and held until
668        // wait() reclaims them, so these pointers stay valid for the whole transfer.
669        let (src_ptr, src_len) = unsafe { src.read_buffer() };
670        let (dst_ptr, dst_len) = unsafe { dst.write_buffer() };
671        let words = src_len.min(dst_len);
672        let transfer_size = match DmaTransferSize::from_beats(words) {
673            Some(size) => size,
674            None => return Err(DmaStartError { error: DmaError::TransferTooLarge, driver: self, channel, src, dst }),
675        };
676        let bytes = words * core::mem::size_of::<u32>();
677
678        self.enable_controller();
679
680        // Clean the source so the DMA master reads CPU-written data (non-coherent core).
681        #[cfg(feature = "chip-ws63")]
682        unsafe {
683            crate::cache::clean_range(src_ptr as usize, bytes);
684        }
685
686        // mem-to-mem, 32-bit, both addresses incrementing (the Default config).
687        let cfg = DmaChannelConfig::default();
688        let ch = channel.logical;
689        self.configure_channel_raw(ch, src_ptr as u32, dst_ptr as u32, transfer_size, &cfg);
690
691        Ok(Transfer { driver: self, channel, src, dst, dst_addr: dst_ptr as usize, bytes })
692    }
693}

695/// An in-flight DMA transfer that **owns the driver and both buffers**. Reclaim them
696/// with [`wait`](Self::wait); dropping the guard early aborts the channel (so the DMA
697/// stops touching the buffers before they are freed). Owning the buffers is what makes
698/// a use-after-free of an active DMA region unrepresentable in safe code.
699#[instability::unstable]
700pub struct Transfer<'d, SRC, DST> {
701    driver: DmaDriver<'d, Dma0>,
702    channel: DmaChannel,
703    src: SRC,
704    dst: DST,
705    // Only read by the chip-ws63 cache maintenance in `wait()`; BS2X DMA is coherent
706    // here, so these are unused there.
707    #[cfg_attr(not(feature = "chip-ws63"), allow(dead_code))]
708    dst_addr: usize,
709    #[cfg_attr(not(feature = "chip-ws63"), allow(dead_code))]
710    bytes: usize,
711}
712
713/// Driver, channel token, and buffers recovered from a completed DMA transfer.
714#[instability::unstable]
715pub type DmaTransferParts<'d, SRC, DST> = (DmaDriver<'d, Dma0>, DmaChannel, SRC, DST);
716
717/// Result returned by [`Transfer::wait`].
718#[instability::unstable]
719pub type DmaWaitResult<'d, SRC, DST> = Result<DmaTransferParts<'d, SRC, DST>, DmaWaitError<'d, SRC, DST>>;
720
721impl<'d, SRC, DST> Transfer<'d, SRC, DST> {
722    /// True once the channel has auto-cleared its enable bit (single-block done).
723    pub fn is_done(&self) -> bool {
724        !self.driver.channel_enabled_raw(self.channel.logical)
725    }
726
727    /// Block (bounded) until the transfer completes, invalidate the destination cache
728    /// lines, and return the driver + both buffers. Consumes the guard, so no abort
729    /// `Drop` runs.
730    ///
731    /// On `DMA_WAIT_LOOPS` exhaustion (a wedged channel — rare for mem-to-mem, which
732    /// always self-completes) the channel is **quiesced** (halt → drain `active` →
733    /// disable) *before* the buffers are handed back, so the DMA engine is provably
734    /// stopped before the buffers can be freed. A wedged transfer is reported as
735    /// [`DmaError::Timeout`] with the driver, channel token, and buffers returned.
736    pub fn wait(self) -> DmaWaitResult<'d, SRC, DST> {
737        // Skip the abort-on-drop: the transfer is completing normally.
738        let mut this = ManuallyDrop::new(self);
739        let mut n = DMA_WAIT_LOOPS;
740        while this.driver.channel_enabled_raw(this.channel.logical) {
741            n -= 1;
742            if n == 0 {
743                break;
744            }
745            core::hint::spin_loop();
746        }
747        // If the channel never auto-cleared, quiesce it before the buffers escape —
748        // otherwise a still-live engine could write a buffer the caller then frees.
749        // (Copy `channel` out so the `&mut this.driver` calls don't borrow `this` whole.)
750        let ch = this.channel.logical;
751        let timed_out = this.driver.channel_enabled_raw(ch);
752        if timed_out {
753            quiesce_channel(&mut this.driver, ch);
754        }
755        // Drop the stale destination cache lines so the CPU re-reads what DMA wrote.
756        #[cfg(feature = "chip-ws63")]
757        unsafe {
758            crate::cache::invalidate_range(this.dst_addr, this.bytes);
759        }
760        // SAFETY: `this` is ManuallyDrop (its Drop never runs) and each field is read
761        // exactly once, so there is no double-read or double-drop.
762        unsafe {
763            let driver = core::ptr::read(&this.driver);
764            let channel = core::ptr::read(&this.channel);
765            let src = core::ptr::read(&this.src);
766            let dst = core::ptr::read(&this.dst);
767            if timed_out {
768                Err(DmaWaitError { error: DmaError::Timeout, driver, channel, src, dst })
769            } else {
770                Ok((driver, channel, src, dst))
771            }
772        }
773    }
774}

833impl<SRC, DST> Drop for Transfer<'_, SRC, DST> {

836    ///
837    /// Uses `quiesce_channel`: halt, then drain the `active` (FIFO/bus) bit, then

842    }

845// ── Channel teardown helpers (cancel-then-quiesce) ───────────────────────────
846
847/// Halt a DMA channel, drain its in-flight beats (the `active`/FIFO-busy bit), then
848/// clear `ch_enable`. This is the DesignWare quiesce sequence — clearing `ch_enable`
849/// mid-burst (the original `Drop`) could let an outstanding bus write land after the
850/// owned buffer is freed, a use-after-free window on the non-coherent core. The poll
851/// is bounded by `DMA_WAIT_LOOPS`.
852///
853/// Factored as a free function over `&mut DmaDriver` so both the mem-to-mem
854/// [`Transfer`] and the peripheral [`PeripheralTransfer`] share one teardown path.
855fn quiesce_channel<T: DmaInstance>(driver: &mut DmaDriver<'_, T>, channel: u8) {
856    driver.halt_channel_raw(channel);
857    let mut n = DMA_WAIT_LOOPS;
858    while driver.channel_active_raw(channel) {
859        if n == 0 {
860            break;
861        }
862        n -= 1;
863        core::hint::spin_loop();
864    }
865    driver.disable_channel_raw(channel);
866}
867
868/// Cancel-then-quiesce for a peripheral-paced transfer: clear the peripheral's
869/// DMA-enable **first** (so it stops asserting DMA requests), then
870/// `quiesce_channel` (halt → drain `active` → disable). esp-hal's
871/// cancel-then-quiesce (`spi/master/dma.rs:1608`) disables the peripheral DMA
872/// first; doing it the other way leaves a spurious request latched.
873#[cfg(feature = "chip-ws63")]
874fn cancel_then_quiesce(peri_dis: &mut PeriDmaCtl, driver: &mut DmaDriver<'_, Dma0>, channel: u8) {
875    peri_dis.disable();
876    quiesce_channel(driver, channel);
877}

934/// A POD handle describing how to clear a peripheral's DMA-enable on teardown, so the
935/// [`PeripheralTransfer`] guard can stop the peripheral asserting DMA requests
936/// without being generic over the driver. Built by the driver's `with_dma`.
937#[cfg(feature = "chip-ws63")]

939#[derive(Debug, Clone, Copy)]
940#[cfg_attr(not(target_arch = "riscv32"), allow(dead_code))]
941pub struct PeriDmaCtl {
942    /// The peripheral's register-block base (e.g. `0x4402_0060`-style — the exact
943    /// data-register address is passed to `start_*` separately; this is the base the
944    /// DMA-enable register lives at).
945    base: usize,
946    kind: PeriKind,
947    dir: DmaDirection,
948}

950#[cfg(feature = "chip-ws63")]
951impl PeriDmaCtl {
952    /// Build a teardown handle for a peripheral at `base` with the given kind/dir.
953    #[instability::unstable]
954    pub const fn new(base: usize, kind: PeriKind, dir: DmaDirection) -> Self {
955        Self { base, kind, dir }
956    }

958    /// Clear the peripheral's DMA-enable so it stops asserting DMA requests. Called
959    /// by `cancel_then_quiesce` **before** halting the channel.
960    ///
961    /// On the host (non-riscv) build this is a no-op so unit tests don't touch MMIO;
962    /// the teardown *order* is exercised by the `cancel_then_quiesce` host test.
963    #[cfg(target_arch = "riscv32")]
964    fn disable(&mut self) {
965        match self.kind {
966            PeriKind::Spi => {

969                        DmaDirection::Tx => r.spi_dcr().modify(|_, w| w.tdmae().clear_bit()),

971                    };
972                }
973            }
974            // UART DMA is paced by FIFO_CTL trigger levels; UART_PARAMETER.dma_mode is
975            // read-only in the PAC, so there is no peripheral DMA-enable bit to clear.
976            PeriKind::Uart => {}
977        }
978    }
979
980    #[cfg(not(target_arch = "riscv32"))]
981    fn disable(&mut self) {}
982}

1112// ── Peripheral-paced transfer guard ──────────────────────────────────────────
1113
1114#[cfg(feature = "chip-ws63")]
1115impl<'d> DmaDriver<'d, Dma0> {
1116    /// Launch a **memory → peripheral** (TX) transfer on `channel` that OWNS `src`
1117    /// for its whole lifetime, returning a [`PeripheralTransfer`] guard you
1118    /// [`wait`](PeripheralTransfer::wait) to reclaim the driver + channel + buffer.
1119    ///
1120    /// Cleans the source cache before launch (non-coherent core) and pins
1121    /// `peri_data_addr` (a peripheral data register) as the fixed destination. The
1122    /// beat width is `frame` (not the mem-to-mem `Word = u32`). The peripheral-side
1123    /// DMA-enable is the caller's responsibility until P1's `with_dma` wires it
1124    /// (vendor order: configure channel → write watermark → clean → start → set
1125    /// peripheral DMA-enable); this guard tears it down on `Drop` via `peri_dis`.
1126    ///
1127    /// Returns [`DmaError::TransferTooLarge`] if `src` exceeds 4095 beats (a
1128    /// single-block DMA `trans_size` cap — chunk the buffer). On `Err` the driver,
1129    /// channel, and buffer are dropped (the channel claim is released).
1130    #[instability::unstable]
1131    pub fn start_mem_to_peripheral<SRC>(
1132        mut self,
1133        channel: DmaChannel,
1134        src: SRC,
1135        peri_data_addr: usize,
1136        peri: DmaPeripheral,
1137        frame: DmaFrame,
1138        peri_dis: PeriDmaCtl,
1139    ) -> Result<PeripheralTransfer<'d, SRC>, DmaError>
1140    where
1141        SRC: ReadBuffer<Word = u8>,
1142    {
1143        let ch = channel.logical;
1144        // SAFETY: `src` is moved into the returned guard and held until wait(), so the
1145        // pointer stays valid for the whole transfer.
1146        let (src_ptr, beats) = unsafe { src.read_buffer() };
1147        if beats > 0xFFF {
1148            // Drop the consumed driver (ZST), channel (releases claim), and buffer.
1149            drop(channel);
1150            return Err(DmaError::TransferTooLarge);
1151        }
1152        let bytes = beats;
1153
1154        // Clean the source so the DMA master reads CPU-written data (non-coherent core).
1155        // The peripheral DATA register is uncached MMIO — never cache-maintain it.
1156        unsafe {
1157            crate::cache::clean_range(src_ptr as usize, bytes);
1158        }
1159
1160        let size = DmaTransferSize::from_beats(beats).expect("beats checked above");
1161        let cfg = DmaChannelConfig::default().mem_to_peripheral(peri).with_width(frame).with_transfer_int(true);
1162        self.configure_channel_raw(ch, src_ptr as u32, peri_data_addr as u32, size, &cfg);
1163
1164        Ok(PeripheralTransfer {
1165            driver: self,
1166            channel,
1167            buf: src,
1168            dir: DmaDirection::Tx,
1169            mem_addr: src_ptr as usize,
1170            bytes,
1171            peri_dis,
1172        })

1175    /// Launch a **peripheral → memory** (RX) transfer on `channel` that OWNS `dst`,
1176    /// returning a [`PeripheralTransfer`] guard. The destination cache is invalidated
1177    /// on [`wait`](PeripheralTransfer::wait) AFTER completion. See
1178    /// [`start_mem_to_peripheral`] for the lifetime / cap / teardown contract.
1179    #[instability::unstable]
1180    pub fn start_peripheral_to_mem<DST>(
1181        mut self,
1182        channel: DmaChannel,
1183        mut dst: DST,
1184        peri_data_addr: usize,
1185        peri: DmaPeripheral,
1186        frame: DmaFrame,
1187        peri_dis: PeriDmaCtl,
1188    ) -> Result<PeripheralTransfer<'d, DST>, DmaError>
1189    where
1190        DST: WriteBuffer<Word = u8>,
1191    {
1192        let ch = channel.logical;
1193        // SAFETY: `dst` is moved into the returned guard and held until wait().
1194        let (dst_ptr, beats) = unsafe { dst.write_buffer() };
1195        if beats > 0xFFF {
1196            drop(channel);
1197            return Err(DmaError::TransferTooLarge);
1198        }
1199        let bytes = beats;
1200
1201        // No clean before launch — the CPU hasn't written the RX destination; the DMA
1202        // will. invalidate_range runs on wait() AFTER completion. The peripheral DATA
1203        // register is uncached MMIO.
1204        let size = DmaTransferSize::from_beats(beats).expect("beats checked above");
1205        let cfg = DmaChannelConfig::default().peripheral_to_mem(peri).with_width(frame).with_transfer_int(true);
1206        self.configure_channel_raw(ch, peri_data_addr as u32, dst_ptr as u32, size, &cfg);
1207
1208        Ok(PeripheralTransfer {
1209            driver: self,
1210            channel,
1211            buf: dst,
1212            dir: DmaDirection::Rx,
1213            mem_addr: dst_ptr as usize,
1214            bytes,
1215            peri_dis,
1216        })
1217    }

1220/// An in-flight peripheral-paced DMA transfer that **owns the driver, the channel
1221/// token, and the single memory buffer**. Reclaim them with
1222/// [`wait`](Self::wait); dropping the guard early runs cancel-then-quiesce: clear
1223/// the peripheral DMA-enable, halt, drain `active`, disable — so neither the channel
1224/// nor the peripheral keeps touching the buffer before it is freed.
1225///
1226/// Owning the buffer (by value, via `embedded_dma`) is what makes a use-after-free of
1227/// an active DMA region unrepresentable in safe code: the buffer cannot be touched,
1228/// moved, or freed while the DMA reads/writes it.
1229#[cfg(feature = "chip-ws63")]
1230#[instability::unstable]
1231pub struct PeripheralTransfer<'d, BUF> {
1232    driver: DmaDriver<'d, Dma0>,
1233    channel: DmaChannel,
1234    buf: BUF,
1235    dir: DmaDirection,
1236    mem_addr: usize,
1237    bytes: usize,
1238    peri_dis: PeriDmaCtl,
1239}
1240
1241#[cfg(feature = "chip-ws63")]
1242impl<'d, BUF> PeripheralTransfer<'d, BUF> {
1243    /// True once the channel has auto-cleared its enable bit (single-block done).
1244    #[instability::unstable]
1245    pub fn is_done(&self) -> bool {
1246        !self.driver.channel_enabled_raw(self.channel.logical)
1247    }
1248
1249    /// Block (bounded) until the transfer completes, do the direction-correct cache
1250    /// maintenance, and return the driver + channel + buffer. Consumes the guard, so
1251    /// no teardown `Drop` runs on the success path.
1252    ///
1253    /// On `DMA_WAIT_LOOPS` exhaustion the channel is cancel-then-quiesced first, then
1254    /// [`Err(DmaError::Timeout)`](DmaError::Timeout) is returned (the driver, channel,
1255    /// and buffer are dropped — the channel claim is released, the buffer is safe
1256    /// because the engine was stopped). A wedged transfer is therefore *observable*,
1257    /// not silently "done" — the UAF-on-timeout hole the mem-to-mem guard had.
1258    #[instability::unstable]
1259    pub fn wait(self) -> Result<(DmaDriver<'d, Dma0>, DmaChannel, BUF), DmaError> {
1260        // Skip the abort-on-drop: the transfer is completing normally.
1261        let mut this = ManuallyDrop::new(self);
1262        let ch = this.channel.logical;
1263        let mut n = DMA_WAIT_LOOPS;
1264        while this.driver.channel_enabled_raw(ch) {
1265            n -= 1;
1266            if n == 0 {
1267                break;
1268            }
1269            core::hint::spin_loop();
1270        }
1271
1272        if this.driver.channel_enabled_raw(ch) {
1273            // Timed out: cancel-then-quiesce so the engine is stopped before any
1274            // buffer/cache teardown, then invalidate the RX destination (the DMA may
1275            // have written partial data) before the buffer drops. The two `&mut`
1276            // field borrows are sequential (not simultaneous) so they reborrow
1277            // `&mut *this` one at a time.
1278            this.peri_dis.disable();
1279            quiesce_channel(&mut this.driver, ch);
1280            if this.dir == DmaDirection::Rx {
1281                unsafe {
1282                    crate::cache::invalidate_range(this.mem_addr, this.bytes);
1283                }
1284            }
1285            // Drop driver + channel (its Drop releases the claim bitmask) + buf.
1286            // `into_inner` unwraps the ManuallyDrop so the inner values actually drop
1287            // (otherwise the channel claim would leak on the timeout path).
1288            drop(ManuallyDrop::into_inner(this));
1289            return Err(DmaError::Timeout);
1290        }
1291
1292        // Completed: invalidate the RX destination so the CPU re-reads DMA-written data.
1293        // (TX needs no invalidate — the CPU didn't write the peripheral, and the source
1294        // was cleaned before launch.)
1295        if this.dir == DmaDirection::Rx {
1296            unsafe {
1297                crate::cache::invalidate_range(this.mem_addr, this.bytes);
1298            }
1299        }
1300
1301        // SAFETY: `this` is ManuallyDrop (its Drop never runs) and each field is read
1302        // exactly once, so there is no double-read or double-drop.
1303        unsafe {
1304            let driver = core::ptr::read(&this.driver);
1305            let channel = core::ptr::read(&this.channel);
1306            let buf = core::ptr::read(&this.buf);
1307            Ok((driver, channel, buf))
1308        }
1309    }
1310}

1313impl<BUF> Drop for PeripheralTransfer<'_, BUF> {
1314    /// Cancel-then-quiesce: clear the peripheral DMA-enable, halt, drain `active`,
1315    /// disable — so neither the channel nor the peripheral keeps asserting requests
1316    /// against a buffer that's about to be freed.
1317    fn drop(&mut self) {
1318        cancel_then_quiesce(&mut self.peri_dis, &mut self.driver, self.channel.logical);
1319    }
1320}

1779// ── Async DMA completion (bespoke; DMA_INT = IRQ 59) ────────────────────────
1780#[cfg(all(feature = "chip-ws63", feature = "async", feature = "unstable"))]
1781mod asynch_impl {
1782    use super::{Dma0, DmaDriver, DmaInstance};
1783    use crate::asynch::IrqSignal;
1784    use crate::interrupt::{self, Interrupt};
1785    use core::future::Future;
1786    use core::pin::Pin;
1787    use core::task::{Context, Poll};
1788
1789    // Per-physical-channel signal (Dma0 has 4 physical channels 0..=3). Replaces the
1790    // earlier single global `DMA_SIGNAL` so two concurrent transfers (e.g. SPI
1791    // full-duplex TX + RX on ch0 + ch1) don't false-wake each other.
1792    static DMA_SIGNAL: [IrqSignal; 4] = [IrqSignal::new(), IrqSignal::new(), IrqSignal::new(), IrqSignal::new()];
1793
1794    /// DMA trap hook (IRQ 59): **demux per channel** — for each completed channel,
1795    /// signal its waker AND clear its `dmac_int_clr` done bit **in-ISR** (a
1796    /// level-triggered done line would re-fire forever otherwise), then clear the
1797    /// ECLIC pending bit. Reads the raw transfer-done mask (`dmac_ori_int_st[7:0]`).
1798    pub fn on_interrupt() {
1799        // SAFETY: Dma0::ptr() is a static physical MMIO address, always valid.
1800        let r = unsafe { &*Dma0::ptr() };
1801        let done = (r.dmac_ori_int_st().read().bits() & 0xFF) as u8;
1802        let mut clr = 0u32;
1803        for ch in 0..4u8 {
1804            if done & (1 << ch) != 0 {
1805                DMA_SIGNAL[ch as usize].signal();
1806                clr |= 1 << ch;
1807            }
1808        }
1809        if clr != 0 {
1810            // SAFETY: writing the per-channel done-clear bits (bit n = channel n).
1811            unsafe { r.dmac_int_clr().write(|w| w.bits(clr)) };
1812        }
1813        interrupt::clear_pending(Interrupt::DMA_INT);
1814    }
1815
1816    /// Named device.x handler (DMA_INT = IRQ 59): the rt routes the DMA IRQ here by
1817    /// number, so an async DMA app needs no `mcause` trap.
1818    #[unsafe(no_mangle)]
1819    extern "C" fn DMA_INT() {
1820        on_interrupt();
1821    }
1822
1823    /// A future that resolves once `DMA_SIGNAL[ch]` fires (the ISR signals it when
1824    /// channel `ch`'s transfer-done bit asserts).
1825    struct DmaDoneFuture {
1826        ch: u8,
1827    }
1828    impl Future for DmaDoneFuture {
1829        type Output = ();
1830        fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
1831            let sig = &DMA_SIGNAL[(self.ch as usize).min(3)];
1832            if sig.take_fired() {
1833                Poll::Ready(())
1834            } else {
1835                sig.register(cx.waker());
1836                Poll::Pending
1837            }
1838        }
1839    }
1840
1841    impl DmaDriver<'_, Dma0> {
1842        /// Await transfer-complete for `channel` (after configuring + enabling it),
1843        /// then clear its interrupt. Parks on the DMA IRQ (IRQ 59) on hardware; the
1844        /// per-channel demux means a concurrent transfer on another channel won't
1845        /// false-wake this one. Silicon-verified: IRQ 59 fires for both mem-to-mem
1846        /// and peripheral-paced single-block completion (spi_dma_irq59 test).
1847        pub async fn wait_transfer_done(&mut self, channel: &super::DmaChannel) {
1848            let channel = channel.logical;
1849            let bit = 1u8 << channel; // Dma0: physical channel == logical channel
1850            if self.raw_interrupt_status().0 & bit == 0 {
1851                DMA_SIGNAL[(channel as usize).min(3)].reset();
1852                // SAFETY: enabling a known, fixed WS63 IRQ line; enable() also raises
1853                // its LOCIPRI above the reset-0 threshold so it is deliverable.
1854                unsafe { interrupt::enable(Interrupt::DMA_INT) };
1855                if self.raw_interrupt_status().0 & bit == 0 {
1856                    DmaDoneFuture { ch: channel }.await;
1857                }
1858            }
1859            self.clear_transfer_interrupt_raw(channel);
1860        }
1861    }
1862}
