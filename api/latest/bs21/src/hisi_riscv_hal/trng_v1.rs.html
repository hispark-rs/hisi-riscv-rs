1//! BS2X TRNG — true random number generator (IP v1).
2//!
3//! BS2X-only (`chip-bs21`) implementation of the `trng` module (WS63 has a
4//! different TRNG, `trng.rs`). Register map from fbb_bs2x `hal_trng_v1` (sole
5//! ground truth): enable the ring oscillators (`trng_ring_en`), poll the FIFO
6//! ready+done bits, read a 32-bit random word from `trng_fifo_data` (the FIFO
7//! auto-advances per read). bs2x-pac `Trng` @ 0x5200_9000.
8
9use crate::peripherals::Trng as TrngPeriph;
10use core::marker::PhantomData;
11
12const POLL_LIMIT: u32 = 1_000_000;
13
14#[derive(Debug, PartialEq, Eq)]
15#[cfg_attr(feature = "defmt", derive(defmt::Format))]
16#[non_exhaustive]
17/// Errors returned by the BS2X TRNG driver.
18pub enum TrngError {
19    /// The FIFO never signalled ready+done within the bounded wait.
20    Timeout,
21}
22
23/// BS2X true random number generator (IP v1) driver.
24pub struct Trng<'d> {
25    _trng: PhantomData<TrngPeriph<'d>>,
26}
27
28impl<'d> Trng<'d> {
29    fn regs(&self) -> &'static crate::soc::pac::trng::RegisterBlock {
30        // SAFETY: static physical MMIO base (0x5200_9000) from bs2x-pac.
31        unsafe { &*TrngPeriph::ptr() }
32    }
33
34    /// Enable the entropy ring oscillators (`hal_trng_v1_start`).
35    ///
36    /// NB: the vendor also enables a TRNG clock/ring/reset in the M_CTL fabric
37    /// (0x5200_0xxx) before this; that is the clock-tree's job. TODO(bs2x): wire
38    /// the M_CTL clock-enable when bringing TRNG up on silicon (the QEMU model
39    /// reports ready regardless).
40    pub fn new(_trng: TrngPeriph<'d>) -> Self {
41        let this = Self { _trng: PhantomData };
42        this.regs().trng_ring_en().write(|w| {
43            w.ro_en().set_bit();
44            w.tero_en().set_bit();
45            w.fro_en().set_bit()
46        });
47        this
48    }
49
50    /// Read one 32-bit random word (`hal_trng_v1_get`): wait for the FIFO to be
51    /// ready AND done, then pop `trng_fifo_data`.
52    pub fn next_u32(&self) -> Result<u32, TrngError> {
53        let r = self.regs();
54        for _ in 0..POLL_LIMIT {
55            let rdy = r.trng_fifo_ready().read();
56            if rdy.trng_ready().bit_is_set() && rdy.trng_done().bit_is_set() {
57                return Ok(r.trng_fifo_data().read().trng_data().bits());
58            }
59            core::hint::spin_loop();
60        }
61        Err(TrngError::Timeout)
62    }
63}
