1//! BS2X RTC — real-time-clock / timer (IP v150).
2//!
3//! BS2X-only (`chip-bs21`) implementation of the `rtc` module (WS63 has the older
4//! v100 RTC, `rtc.rs`). The v150 RTC is a 64-bit down-counter with a coherent-read
5//! handshake. Register map from fbb_bs2x `hal_rtc_v150` (sole ground truth); this
6//! drives the RTC0 instance (bs2x-pac `Rtc` @ 0x5702_4100).
7//!
8//! Reading the 64-bit counter requires the cnt_req → cnt_lock latch handshake
9//! (so the two 32-bit halves are coherent); the wait for `cnt_lock` is bounded.
10//!
11//! # Preconditions (board / analog)
12//!
13//! The RTC counts on the 32.768 kHz clock domain, sourced from an **external
14//! crystal that a board may not populate**. With the crystal absent the counter
15//! never advances and register access can stall the bus — a board provisioning
16//! fact with no software guard. Treat a populated 32.768 kHz crystal as a hard
17//! precondition before constructing the driver.
18
19use crate::peripherals::Rtc as RtcPeriph;
20use core::marker::PhantomData;
21
22/// Counter run mode (`control.mode`).
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24#[repr(u8)]
25pub enum Mode {
26    /// Count down once from `load` to zero, then stop.
27    OneShot = 0,
28    /// Count down from `load`, reload and repeat (periodic interrupt).
29    Periodic = 1,
30    /// Free-running counter (wraps past zero; bit pattern is 3, not 2).
31    FreeRun = 3,
32}
33
34const POLL_LIMIT: u32 = 0xFFFF;
35
36/// Driver for the BS2X 64-bit v150 RTC (RTC0 instance).
37pub struct Rtc<'d> {
38    _rtc: PhantomData<RtcPeriph<'d>>,
39}
40
41impl<'d> Rtc<'d> {
42    fn regs(&self) -> &'static crate::soc::pac::rtc::RegisterBlock {
43        // SAFETY: static physical MMIO base (0x5702_4100, RTC0) from bs2x-pac.
44        unsafe { &*RtcPeriph::ptr() }
45    }
46
47    /// Start the RTC counting from `load` in `mode` (`hal_rtc_v150_config_load` +
48    /// `_start`). For a free-running counter use `Mode::FreeRun`.
49    ///
50    /// typed-config exemption: `load` is written verbatim into the 32-bit
51    /// `load_count0` register (the low half of the 64-bit counter), so every `u32`
52    /// is a valid, runnable value — nothing to truncate or clamp. (The board-crystal
53    /// precondition is documented at the module level.)
54    pub fn new(_rtc: RtcPeriph<'d>, load: u32, mode: Mode) -> Self {
55        let this = Self { _rtc: PhantomData };
56        let r = this.regs();
57        unsafe {
58            r.control().modify(|_, w| w.enable().clear_bit()); // stop before load
59            r.control().modify(|_, w| w.mode().bits(mode as u8));
60            r.load_count0().write(|w| w.load_count0().bits(load));
61            r.control().modify(|_, w| w.enable().set_bit());
62        }
63        this
64    }
65
66    /// Read the live 64-bit counter using the cnt_req → cnt_lock coherent-read
67    /// handshake (`hal_rtc_v150_get_current_value`).
68    pub fn read_count(&self) -> u64 {
69        let r = self.regs();
70        if r.control().read().enable().bit_is_clear() {
71            return 0;
72        }
73        // Request a latched snapshot, then wait for the lock to be granted.
74        r.control().modify(|_, w| w.cnt_req().set_bit());
75        for _ in 0..POLL_LIMIT {
76            if r.control().read().cnt_lock().bit_is_set() {
77                break;
78            }
79            core::hint::spin_loop();
80        }
81        let lo = r.current_value0().read().current_value0().bits();
82        let hi = r.current_value1().read().current_value1().bits();
83        ((hi as u64) << 32) | lo as u64
84    }
85
86    /// True if the timer interrupt is pending (`raw_intr` bit0).
87    pub fn is_pending(&self) -> bool {
88        self.regs().raw_intr().read().int_status().bit_is_set()
89    }
90
91    /// Clear the timer interrupt by writing the per-instance EOI bit.
92    pub fn clear_interrupt(&self) {
93        self.regs().eoi_ren().write(|w| w.eoi().set_bit());
94    }
95}

99#[cfg(all(test, not(target_arch = "riscv32")))]
100mod tests {
101    use super::*;
102
103    /// Reassemble the 64-bit counter exactly as `read_count` does from its two
104    /// latched 32-bit halves (`current_value1` = high word, `current_value0` = low).
105    fn combine(hi: u32, lo: u32) -> u64 {
106        ((hi as u64) << 32) | lo as u64
107    }
108
109    #[test]
110    fn mode_bits_match_hardware() {
111        // The control.mode field is programmed with `mode as u8`; the v150 RTC
112        // uses 0/1/3 (FreeRun is 3, NOT 2 — the bit pattern is intentionally sparse).
113        assert_eq!(Mode::OneShot as u8, 0);
114        assert_eq!(Mode::Periodic as u8, 1);
115        assert_eq!(Mode::FreeRun as u8, 3);
116    }
117
118    #[test]
119    fn poll_limit_value() {
120        // The coherent-read handshake spins at most POLL_LIMIT times; pin the
121        // documented 16-bit ceiling so a future edit can't silently shrink it.
122        assert_eq!(POLL_LIMIT, 0xFFFF);
123    }
124
125    #[test]
126    fn combine_low_word_only() {
127        // hi == 0 → the value is exactly the low 32-bit half (no high bits set).
128        assert_eq!(combine(0, 0x1234_5678), 0x0000_0000_1234_5678);
129    }
130
131    #[test]
132    fn combine_high_word_only() {
133        // lo == 0 → the high half lands in bits 63..32, low half stays zero.
134        assert_eq!(combine(0x1234_5678, 0), 0x1234_5678_0000_0000);
135    }
136
137    #[test]
138    fn combine_no_bit_overlap() {
139        // The two halves occupy disjoint bit ranges: combining all-ones halves
140        // yields the full u64::MAX with no truncation or carry between them.
141        assert_eq!(combine(u32::MAX, u32::MAX), u64::MAX);
142    }
143
144    #[test]
145    fn combine_round_trips() {
146        // Splitting any u64 into its two 32-bit halves and recombining is the
147        // identity — confirms the shift/mask ordering matches a clean split.
148        let v: u64 = 0xDEAD_BEEF_CAFE_F00D;
149        let hi = (v >> 32) as u32;
150        let lo = v as u32;
151        assert_eq!(combine(hi, lo), v);
152    }
153}

157#[cfg(all(test, not(target_arch = "riscv32")))]
158mod proptests {
159    use proptest::prelude::*;
160
161    fn combine(hi: u32, lo: u32) -> u64 {
162        ((hi as u64) << 32) | lo as u64
163    }
164
165    proptest! {
166        /// Fuzz: combine then split is the identity for any pair of halves
167        /// (the high half is recovered by >>32, the low half by truncation).
168        #[test]
169        fn combine_split_round_trips(hi in any::<u32>(), lo in any::<u32>()) {
170            let v = combine(hi, lo);
171            prop_assert_eq!((v >> 32) as u32, hi);
172            prop_assert_eq!(v as u32, lo);
173        }
174
175        /// Fuzz: any u64 split into halves and recombined returns the original.
176        #[test]
177        fn split_combine_round_trips(v in any::<u64>()) {
178            prop_assert_eq!(combine((v >> 32) as u32, v as u32), v);
179        }
180    }
