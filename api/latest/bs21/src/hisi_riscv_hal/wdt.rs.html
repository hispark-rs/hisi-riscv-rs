1//! Watchdog Timer (WDT) driver for WS63.
2//!
3//! The WS63 WDT is a 24-bit down-counter that can generate an interrupt
4//! or system reset when it reaches zero. The watchdog must be periodically
5//! "fed" (restarted) to prevent a timeout.
6//!
7//! # Lock mechanism
8//!
9//! The WDT registers are protected by a lock. Write `0x5A5A5A5A` to
10//! `WDT_LOCK` to unlock, any other value to lock. The driver handles
11//! this automatically.
12//!
13//! # Operating modes
14//!
15//! - **Single-interrupt mode** (`wdt_mode = 0`): One interrupt before reset
16//! - **Double-interrupt mode** (`wdt_mode = 1`): Two interrupts before reset
17//!
18//! # Clock source
19//!
20//! The WDT counts at the **TCXO crystal clock** ([`WDT_CLOCK_HZ`] = 24 MHz on
21//! 24 MHz-crystal boards) — NOT a 32.768 kHz RTC clock — programmed by the vendor
22//! `watchdog_port_set_clock(REQ_24M)` in `clock_init`. The 24-bit `WDT_LOAD` field
23//! lives in bits [31:8], so the effective counter has 256-cycle resolution
24//! (`WDT_LOAD_RESEV = 8`) and the maximum timeout is `(0xFFFFFF << 8) / 24 MHz ≈ 178 s`.
25
26use crate::peripherals::Wdt;

28/// WDT reset pulse length in clock cycles.
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub enum ResetPulseLength {
31    /// 2 clock cycles (~83 ns at 24 MHz)
32    Cycles2 = 0,
33    /// 4 clock cycles
34    Cycles4 = 1,
35    /// 8 clock cycles
36    Cycles8 = 2,
37    /// 16 clock cycles
38    Cycles16 = 3,
39    /// 32 clock cycles
40    Cycles32 = 4,
41    /// 64 clock cycles
42    Cycles64 = 5,
43    /// 128 clock cycles
44    Cycles128 = 6,
45    /// 256 clock cycles (~10.7 µs at 24 MHz)
46    Cycles256 = 7,
47}
48
49/// WDT operating mode.
50#[derive(Debug, Clone, Copy, PartialEq, Eq)]
51pub enum WdtMode {
52    /// One interrupt, then reset on next timeout.
53    SingleInterrupt,
54    /// Two interrupts, then reset on third timeout.
55    DoubleInterrupt,
56}

58/// Watchdog configuration / operation error.
59#[derive(Debug, Clone, Copy, PartialEq, Eq)]
60#[cfg_attr(feature = "defmt", derive(defmt::Format))]
61#[non_exhaustive]
62pub enum WdtError {
63    /// The counter-value latch did not become valid within the bounded poll
64    /// (the WDT block is unclocked or wedged).
65    Busy,
66}

68/// A validated watchdog timeout in milliseconds.
69///
70/// The constructor [`WdtTimeout::from_ms`] returns `None` for any value that would
71/// fall outside the representable range — instead of the old silent saturation
72/// (`timeout > ~178 s → WDT_MAX_LOAD`). A `WdtTimeout` in hand always programs a
73/// `WDT_LOAD` field in `[1, WDT_MAX_LOAD]`, so [`Watchdog::configure`] can never
74/// clamp or truncate it.
75#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
76#[cfg_attr(feature = "defmt", derive(defmt::Format))]
77pub struct WdtTimeout(u32);
78
79impl WdtTimeout {
80    /// The largest timeout (ms) whose 24-bit `WDT_LOAD` field does not overflow:
81    /// `(WDT_MAX_LOAD << RESEV) / WDT_CLOCK_HZ · 1000`, ≈ 178 000 ms at 24 MHz.
82    pub const MAX_MS: u32 = {
83        let max_cycles = (WDT_MAX_LOAD as u64) << WDT_LOAD_RESEV;
84        (max_cycles * 1000 / (WDT_CLOCK_HZ as u64)) as u32
85    };
86
87    /// Construct from milliseconds. `None` if `ms == 0` (no useful watchdog) or if
88    /// the resulting `WDT_LOAD` field would exceed [`WDT_MAX_LOAD`] (i.e.
89    /// `ms > MAX_MS`, the old silent-saturation boundary).
90    pub const fn from_ms(ms: u32) -> Option<Self> {
91        if ms == 0 {
92            return None;
93        }
94        let cycles = (ms as u64) * (WDT_CLOCK_HZ as u64) / 1000;
95        if (cycles >> WDT_LOAD_RESEV) > WDT_MAX_LOAD as u64 {
96            return None;
97        }
98        Some(WdtTimeout(ms))
99    }
100
101    /// The timeout in milliseconds.
102    pub const fn as_ms(self) -> u32 {
103        self.0
104    }
105
106    /// The 24-bit `WDT_LOAD` field value (`cycles >> RESEV`), guaranteed
107    /// `<= WDT_MAX_LOAD` by construction.
108    const fn load_field(self) -> u32 {
109        let cycles = (self.0 as u64) * (WDT_CLOCK_HZ as u64) / 1000;
110        (cycles >> WDT_LOAD_RESEV) as u32
111    }
112}
113
114/// Bounded spin limit for the `WDT_CCVR` counter-value latch (`request_counter`).
115/// The latch normally validates in a handful of cycles; this is a generous cap so a
116/// wedged/unclocked block returns [`WdtError::Busy`] instead of hanging forever.
117const WDT_CCVR_POLL_LIMIT: u32 = 100_000;
118
119/// Watchdog Timer driver.
120pub struct Watchdog<'d> {
121    _wdt: Wdt<'d>,
122}
123
124/// WDT counting clock = the TCXO crystal ([`crate::soc::chip::TCXO_HZ`], 24 MHz).
125pub const WDT_CLOCK_HZ: u32 = crate::soc::chip::TCXO_HZ;
126
127/// Maximum `WDT_LOAD` field value (24-bit; the field occupies `WDT_LOAD` bits [31:8]).
128pub const WDT_MAX_LOAD: u32 = 0x00FF_FFFF;
129
130/// Reserved low bits of `WDT_LOAD`: the 24-bit load field sits in bits [31:8], so
131/// the total timeout cycle count is shifted right by this to form the field value
132/// (matches the vendor `LOAD_RESEV = 8` in `hal_watchdog_v151`).
133const WDT_LOAD_RESEV: u32 = 8;
134
135impl<'d> Watchdog<'d> {
136    /// Create a new watchdog driver from the WDT peripheral.
137    pub fn new(wdt: Wdt<'d>) -> Self {
138        let wd = Self { _wdt: wdt };
139        wd.unlock();
140        wd
141    }
142
143    fn regs(&self) -> &'static crate::soc::pac::wdt::RegisterBlock {
144        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid
145        unsafe { &*Wdt::ptr() }
146    }
147
148    /// Unlock the WDT registers for configuration.
149    fn unlock(&self) {
150        unsafe {
151            self.regs().wdt_lock().write(|w| w.bits(0x5A5A5A5A));
152        }
153    }
154
155    /// Lock the WDT registers to prevent accidental modification.
156    fn lock(&self) {
157        unsafe {
158            self.regs().wdt_lock().write(|w| w.bits(0x0000_0000));
159        }
160    }
161
162    /// Configure and enable the watchdog.
163    ///
164    /// The timeout is a validated [`WdtTimeout`] (built via [`WdtTimeout::from_ms`]),
165    /// so an out-of-range period is rejected at construction rather than silently
166    /// saturated here.
167    ///
168    /// # Arguments
169    ///
170    /// * `timeout` - Validated timeout period (see [`WdtTimeout`]).
171    /// * `mode` - Interrupt mode (single or double interrupt before reset).
172    /// * `reset_enable` - Whether to enable system reset on timeout.
173    /// * `reset_pulse` - Reset pulse length if reset is enabled.
174    ///
175    /// # Errors
176    ///
177    /// [`WdtError::Busy`] if the counter-value latch does not validate within the
178    /// bounded poll (an unclocked/wedged WDT block).
179    pub fn configure(
180        &mut self,
181        timeout: WdtTimeout,
182        mode: WdtMode,
183        reset_enable: bool,
184        reset_pulse: ResetPulseLength,
185    ) -> Result<(), WdtError> {
186        // The 24-bit WDT_LOAD field, guaranteed <= WDT_MAX_LOAD by `WdtTimeout`.
187        let load = timeout.load_field();
188
189        self.unlock();
190
191        // Write the 24-bit load field into WDT_LOAD bits [31:8].
192        unsafe {
193            self.regs().wdt_load().write(|w| w.bits(load << WDT_LOAD_RESEV));
194        }
195
196        // Configure control register
197        let mut cr: u32 = 0;
198        cr |= 0x01; // wdt_en = 1
199        if reset_enable {
200            cr |= 1 << 2; // rst_en
201        }
202        cr |= (reset_pulse as u32) << 3; // rst_pl
203        cr |= 1 << 6; // wdt_imsk = 1 (mask interrupt initially)
204        if matches!(mode, WdtMode::DoubleInterrupt) {
205            cr |= 1 << 7; // wdt_mode = 1 (double interrupt)
206        }
207
208        unsafe {
209            self.regs().wdt_cr().write(|w| w.bits(cr));
210        }
211
212        // Request counter value update (bounded; propagates Busy on a wedged block).
213        let res = self.request_counter();
214
215        self.lock();
216        res
217    }
218
219    /// Enable the watchdog.
220    pub fn enable(&mut self) {
221        self.unlock();
222        let cr = self.regs().wdt_cr().read().bits();
223        unsafe {
224            self.regs().wdt_cr().write(|w| w.bits(cr | 0x01));
225        }
226        self.lock();
227    }
228
229    /// Disable the watchdog.
230    ///
231    /// # Safety
232    ///
233    /// Disabling the watchdog may leave the system vulnerable to lock-ups.
234    /// Only use this when you have an alternative safety mechanism.
235    pub fn disable(&mut self) {
236        self.unlock();
237        let cr = self.regs().wdt_cr().read().bits();
238        unsafe {
239            self.regs().wdt_cr().write(|w| w.bits(cr & !0x01));
240        }
241        self.lock();
242    }
243
244    /// Feed (restart) the watchdog to prevent timeout.
245    ///
246    /// Writes any value other than `0x5A5A5A5A` to the restart register.
247    pub fn feed(&mut self) {
248        self.unlock();
249        unsafe {
250            self.regs().wdt_restart().write(|w| w.bits(0x0000_0001));
251        }
252        self.lock();
253    }
254
255    /// Read the current counter value.
256    ///
257    /// # Errors
258    ///
259    /// [`WdtError::Busy`] if the counter-value latch does not validate within the
260    /// bounded poll.
261    pub fn counter_value(&self) -> Result<u32, WdtError> {
262        self.request_counter()?;
263        Ok(self.regs().wdt_cnt().read().bits())
264    }
265
266    /// Request a counter-value latch and wait (bounded) for it to validate.
267    ///
268    /// Returns [`WdtError::Busy`] after [`WDT_CCVR_POLL_LIMIT`] spins instead of
269    /// hanging forever on an unclocked/wedged block.
270    fn request_counter(&self) -> Result<(), WdtError> {
271        unsafe {
272            self.regs().wdt_ccvr_en().write(|w| w.bits(0x01));
273        }
274        for _ in 0..WDT_CCVR_POLL_LIMIT {
275            if self.regs().wdt_ccvr_en().read().bits() & 0x02 != 0 {
276                return Ok(());
277            }
278            core::hint::spin_loop();
279        }
280        Err(WdtError::Busy)
281    }
282
283    /// Check if a watchdog interrupt is pending (raw, unmasked).
284    #[instability::unstable]
285    pub fn interrupt_pending(&self) -> bool {
286        self.regs().wdt_raw_intr().read().bits() & 0x01 != 0
287    }
288
289    /// Check if a watchdog interrupt is pending (after mask).
290    #[instability::unstable]
291    pub fn interrupt_masked(&self) -> bool {
292        self.regs().wdt_intr().read().bits() & 0x01 != 0
293    }
294
295    /// Clear the watchdog interrupt (read from `WDT_EOI`).
296    #[instability::unstable]
297    pub fn clear_interrupt(&self) {
298        let _ = self.regs().wdt_eoi().read().bits();
299    }
300
301    /// Enable the watchdog interrupt (unmask).
302    #[instability::unstable]
303    pub fn enable_interrupt(&mut self) {
304        self.unlock();
305        let cr = self.regs().wdt_cr().read().bits();
306        unsafe {
307            self.regs().wdt_cr().write(|w| w.bits(cr & !(1 << 6)));
308        }
309        self.lock();
310    }
311
312    /// Disable the watchdog interrupt (mask).
313    #[instability::unstable]
314    pub fn disable_interrupt(&mut self) {
315        self.unlock();
316        let cr = self.regs().wdt_cr().read().bits();
317        unsafe {
318            self.regs().wdt_cr().write(|w| w.bits(cr | (1 << 6)));
319        }
320        self.lock();
321    }
322
323    /// Check if the watchdog is currently busy.
324    pub fn is_busy(&self) -> bool {
325        self.regs().wdt_status().read().bits() & 0x01 == 0
326    }
327
328    /// Consume the watchdog, leaving it **armed** past this scope — the escape hatch
329    /// from the disabling [`Drop`](Watchdog#impl-Drop-for-Watchdog). Use this once
330    /// the watchdog must keep guarding the system after the configuring code
331    /// returns (the normal production case). Returns a [`WatchdogArmed`] marker so
332    /// the intent is explicit.
333    #[must_use = "the watchdog is now armed forever; bind the marker to make that explicit"]
334    pub fn into_armed(self) -> WatchdogArmed {
335        core::mem::forget(self); // skip the disabling Drop — keep guarding the system
336        WatchdogArmed(())
337    }
338
339    /// Alias for [`into_armed`](Self::into_armed): consume the handle and leave the
340    /// watchdog running.
341    #[must_use = "the watchdog is now armed forever; bind the marker to make that explicit"]
342    pub fn leak(self) -> WatchdogArmed {
343        self.into_armed()
344    }
345}
346
347/// Proof token from [`Watchdog::into_armed`] / [`Watchdog::leak`]: the watchdog was
348/// intentionally left armed past the driver's scope (no disabling `Drop` ran).
349#[derive(Debug)]
350#[must_use]
351pub struct WatchdogArmed(());
352
353impl Drop for Watchdog<'_> {
354    /// Scoped safety: a dropped (un-armed) watchdog is **stopped** so it cannot
355    /// reset the system after its configuring scope ends. Clears only `WDT_CR.wdt_en`
356    /// under the lock handshake — never a shared clock gate. Call
357    /// [`Watchdog::into_armed`] to keep it guarding past the handle's scope.
358    fn drop(&mut self) {
359        self.disable();
360    }
361}

365#[cfg(all(test, not(target_arch = "riscv32")))]
366mod tests {
367    use super::{ResetPulseLength, WDT_CLOCK_HZ, WDT_LOAD_RESEV, WDT_MAX_LOAD, WatchdogArmed, WdtMode, WdtTimeout};
368
369    /// The `into_armed`/`leak` escape-hatch marker is zero-sized (a pure type-level
370    /// proof token). The disabling-Drop register effect itself is HIL-validated on
371    /// silicon (`wdt_drop_disables_unless_armed`) — the host has no MMIO.
372    #[test]
373    fn armed_marker_is_zero_sized() {
374        assert_eq!(core::mem::size_of::<WatchdogArmed>(), 0);
375    }
376
377    /// The `WDT_LOAD` field a *valid* `WdtTimeout` programs — `from_ms(ms)` then the
378    /// private `load_field()`. Panics on an out-of-range `ms` (the test's contract:
379    /// only valid timeouts reach the field math now; over-range is rejected up front).
380    fn load_field(timeout_ms: u32) -> u32 {
381        WdtTimeout::from_ms(timeout_ms).expect("ms in range").load_field()
382    }
383
384    /// Re-derivation of the control-register encoding from `configure()`.
385    fn control_bits(mode: WdtMode, reset_enable: bool, reset_pulse: ResetPulseLength) -> u32 {
386        let mut cr: u32 = 0;
387        cr |= 0x01; // wdt_en
388        if reset_enable {
389            cr |= 1 << 2; // rst_en
390        }
391        cr |= (reset_pulse as u32) << 3; // rst_pl
392        cr |= 1 << 6; // wdt_imsk
393        if matches!(mode, WdtMode::DoubleInterrupt) {
394            cr |= 1 << 7; // wdt_mode
395        }
396        cr
397    }
398
399    #[test]
400    fn reset_pulse_discriminants_are_the_3bit_field() {
401        // The eight pulse lengths map to the contiguous 3-bit rst_pl field 0..=7.
402        assert_eq!(ResetPulseLength::Cycles2 as u32, 0);
403        assert_eq!(ResetPulseLength::Cycles4 as u32, 1);
404        assert_eq!(ResetPulseLength::Cycles8 as u32, 2);
405        assert_eq!(ResetPulseLength::Cycles16 as u32, 3);
406        assert_eq!(ResetPulseLength::Cycles32 as u32, 4);
407        assert_eq!(ResetPulseLength::Cycles64 as u32, 5);
408        assert_eq!(ResetPulseLength::Cycles128 as u32, 6);
409        assert_eq!(ResetPulseLength::Cycles256 as u32, 7);
410    }
411
412    #[test]
413    fn zero_timeout_rejected() {
414        // A 0 ms timeout is not a useful watchdog → rejected at construction.
415        assert!(WdtTimeout::from_ms(0).is_none());
416    }
417
418    #[test]
419    fn known_timeout_matches_hand_computed_field() {
420        // 1000 ms = WDT_CLOCK_HZ cycles; the field is that shifted right by RESEV.
421        let cycles = WDT_CLOCK_HZ as u64; // 1000 ms * clock / 1000
422        let expected = (cycles >> WDT_LOAD_RESEV) as u32;
423        assert_eq!(load_field(1000), expected);
424        assert!(expected <= WDT_MAX_LOAD);
425    }
426
427    #[test]
428    fn over_range_timeout_rejected() {
429        // u32::MAX ms vastly exceeds the 24-bit field → rejected (was silently
430        // saturated to WDT_MAX_LOAD before the tightening).
431        assert!(WdtTimeout::from_ms(u32::MAX).is_none());
432        // The boundary is exact: MAX_MS constructs, MAX_MS+1 does not.
433        assert!(WdtTimeout::from_ms(WdtTimeout::MAX_MS).is_some());
434        assert!(WdtTimeout::from_ms(WdtTimeout::MAX_MS + 1).is_none());
435        // The largest accepted timeout still fits the 24-bit field.
436        assert!(WdtTimeout::from_ms(WdtTimeout::MAX_MS).unwrap().load_field() <= WDT_MAX_LOAD);
437    }
438
439    #[test]
440    fn load_field_is_monotonic_in_range() {
441        // More milliseconds never produce a smaller field (within the valid range).
442        let a = load_field(1000);
443        let b = load_field(2000);
444        assert!(b >= a);
445        assert!(b <= WDT_MAX_LOAD);
446    }
447
448    #[test]
449    fn first_timeout_past_boundary_is_rejected() {
450        // A timeout past the (WDT_MAX_LOAD+1)<<RESEV cycle boundary no longer
451        // clamps — it is rejected. Compute that boundary in ms with CEILING
452        // division (+1 ms margin) so the input definitely clears the threshold.
453        let threshold_cycles = (WDT_MAX_LOAD as u64 + 1) << WDT_LOAD_RESEV;
454        let big_ms = (threshold_cycles * 1000).div_ceil(WDT_CLOCK_HZ as u64) + 1;
455        let ms = big_ms.min(u32::MAX as u64) as u32;
456        assert!(WdtTimeout::from_ms(ms).is_none());
457    }
458
459    #[test]
460    fn control_bits_minimal_config() {
461        // Single-interrupt, no reset, pulse 0: only wdt_en (bit0) + wdt_imsk (bit6).
462        let cr = control_bits(WdtMode::SingleInterrupt, false, ResetPulseLength::Cycles2);
463        assert_eq!(cr, 0x01 | (1 << 6));
464    }
465
466    #[test]
467    fn control_bits_full_config() {
468        // Double-interrupt + reset + pulse 256 (field 7): all encoded bits set.
469        let cr = control_bits(WdtMode::DoubleInterrupt, true, ResetPulseLength::Cycles256);
470        let expected = 0x01            // wdt_en
471            | (1 << 2)                 // rst_en
472            | (7 << 3)                 // rst_pl = Cycles256
473            | (1 << 6)                 // wdt_imsk
474            | (1 << 7); // wdt_mode (double)
475        assert_eq!(cr, expected);
476    }
477
478    #[test]
479    fn control_bits_reset_pulse_occupies_field() {
480        // rst_pl lives in bits [5:3]; verify each variant lands there and nowhere else.
481        for (variant, val) in
482            [(ResetPulseLength::Cycles2, 0u32), (ResetPulseLength::Cycles16, 3), (ResetPulseLength::Cycles256, 7)]
483        {
484            let cr = control_bits(WdtMode::SingleInterrupt, false, variant);
485            assert_eq!((cr >> 3) & 0x7, val);
486        }
487    }
488
489    #[test]
490    fn mode_only_affects_bit7() {
491        // The only difference between single and double interrupt is wdt_mode (bit7).
492        let single = control_bits(WdtMode::SingleInterrupt, true, ResetPulseLength::Cycles32);
493        let double = control_bits(WdtMode::DoubleInterrupt, true, ResetPulseLength::Cycles32);
494        assert_eq!(single ^ double, 1 << 7);
495    }
