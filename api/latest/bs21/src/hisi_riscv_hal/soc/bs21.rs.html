1//! BS21 / BS2X chip-specific description (SparkLink/NearLink, no Wi-Fi).
2//!
3//! The active SoC description under `--features chip-bs21,unstable`. Provides the same
4//! contract `soc/ws63.rs` does (the `Interrupt` enum + clock constants + instance
5//! counts) so the chip-neutral drivers read BS21 numbers through `soc::chip`.
6//!
7//! Facts from the fbb_bs2x SDK (see `docs/bs21-recon.md`). Some clock values are
8//! placeholders pending the BS21 `clocks.c` audit; they do not affect milestone-1
9//! (blinky busy-loops on the CPU clock, and the QEMU UART model ignores baud).
10#![allow(dead_code)]
11
12pub use bs2x_pac::interrupt::ExternalInterrupt as Interrupt;
13
14/// Interrupt fired by the TIMER channel the embassy-time driver uses for alarms.
15///
16/// BS2X routes TIMER channel 0 to `TIMER_0` (IRQ 53, a HiSilicon custom LOCI
17/// local interrupt — `chip_core_irq.h`: `LOCAL_INTERRUPT0 + 27`). The `embassy`
18/// time-driver and the app's trap handler reach this through
19/// [`crate::soc::chip::ALARM_INTERRUPT`] so the alarm wiring is chip-neutral
20/// (WS63 uses IRQ 26 — see `soc/ws63.rs`). The LOCI controller delivers it with
21/// `mcause = 53`, identical decode to the WS63 path.
22pub const ALARM_INTERRUPT: Interrupt = Interrupt::TIMER_0;
23
24/// CPU clock (64 MHz — BS21/BS21E/BS22 all run the app core at 64 MHz).
25pub const SYSTEM_CLOCK_HZ: u32 = 64_000_000;

27/// TCXO / crystal reference — the Timer/WDT counting clock and TCXO time base.
