1//! BS2X USB 2.0 OTG — Synopsys DesignWare DWC2 OTG controller, device mode.
2//!
3//! BS2X-only (`chip-bs21`); no WS63 analogue. This brings the DWC2 OTG **device
4//! controller** up to the enumerated state: core soft-reset → force device mode →
5//! global config → device config → soft-connect → wait for the host USB-reset and
6//! enumeration-done, then read the negotiated speed. Higher layers (descriptors,
7//! endpoint transfers, the USB class stack) sit on top and are out of scope here.
8//!
9//! Sequence + bits from the fbb_bs2x DWC OTG PCD (`dwc_otgreg.h` / `dwc_otg_pcd.h`,
10//! the canonical Synopsys DWC2 device init). All registers are in bs2x-pac's `usb`
11//! block; `Usb` @ 0x5800_0000. The PHY/PMU power-on (a CRG/PHY concern) is assumed
12//! done by the boot/clock layer.
13
14use crate::peripherals::Usb as UsbPeriph;
15use core::marker::PhantomData;
16
17/// The Synopsys signature in the top half of GSNPSID ("OT").
18pub const SNPS_SIGNATURE: u16 = 0x4F54;
19
20const POLL_LIMIT: u32 = 1_000_000;
21
22/// True if a raw `gsnpsid` core-ID value carries the Synopsys DWC OTG signature
23/// in its top 16 bits.
24const fn id_has_signature(core_id: u32) -> bool {
25    (core_id >> 16) as u16 == SNPS_SIGNATURE
26}
27
28/// Decode the 2-bit `DSTS.ENUMSPD` field into the negotiated [`Speed`].
29const fn speed_from_enumspd(enumspd: u8) -> Speed {
30    match enumspd {
31        0 => Speed::High,
32        1 => Speed::FullFs,
33        2 => Speed::Low,
34        _ => Speed::Full,
35    }
36}
37
38/// Errors from bringing up the DWC2 OTG device controller.
39#[derive(Debug, PartialEq, Eq)]
40#[cfg_attr(feature = "defmt", derive(defmt::Format))]
41#[non_exhaustive]
42pub enum UsbError {
43    /// A status bit (reset-done / USB reset / enumeration done) never asserted.
44    Timeout,
45    /// The DWC OTG core-ID signature was wrong (controller not present).
46    NotPresent,
47}
48
49/// The USB speed the host enumerated the device at (`DSTS.ENUMSPD`).
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub enum Speed {
52    /// High speed (`ENUMSPD` = 0).
53    High,
54    /// Full speed via the high-speed PHY (`ENUMSPD` = 3).
55    Full,
56    /// Low speed (`ENUMSPD` = 2).
57    Low,
58    /// Full speed via a full-speed PHY (`ENUMSPD` = 1).
59    FullFs,
60}
61
62/// Driver for the BS2X DWC2 OTG controller in USB 2.0 device mode.
63pub struct Usb<'d> {
64    _u: PhantomData<UsbPeriph<'d>>,
65}
66
67impl<'d> Usb<'d> {
68    fn regs(&self) -> &'static crate::soc::pac::usb::RegisterBlock {
69        // SAFETY: static physical MMIO base (0x5800_0000) from bs2x-pac.
70        unsafe { &*UsbPeriph::ptr() }
71    }
72
73    /// Create the driver from the `Usb` peripheral token.
74    pub fn new(_u: UsbPeriph<'d>) -> Self {
75        Self { _u: PhantomData }
76    }
77
78    /// Read the DWC OTG core-ID register (`gsnpsid`); the top 16 bits are the
79    /// Synopsys "OT" signature.
80    pub fn core_id(&self) -> u32 {
81        self.regs().gsnpsid().read().bits()
82    }
83
84    /// True if the core-ID carries the Synopsys DWC OTG signature.
85    pub fn is_present(&self) -> bool {
86        id_has_signature(self.core_id())
87    }
88
89    /// Bring the DWC2 OTG device controller up and wait until the host has reset
90    /// and enumerated it; returns the negotiated [`Speed`].
91    ///
92    /// Steps (DWC2 device init): core soft reset → force device mode → GAHBCFG +
93    /// interrupt unmask → DCFG → enable global interrupts → soft-connect (clear
94    /// `DCTL.SFTDISCON`) → wait `GINTSTS.USBRST` → wait `GINTSTS.ENUMDONE` → read
95    /// `DSTS.ENUMSPD`.
96    pub fn device_enumerate(&mut self) -> Result<Speed, UsbError> {
97        let r = self.regs();
98        if !self.is_present() {
99            return Err(UsbError::NotPresent);
100        }
101
102        // A. Core soft reset: wait AHB idle, assert CSFTRST, wait it self-clear.
103        self.poll(|| r.grstctl().read().ahbidle().bit_is_set())?;
104        r.grstctl().write(|w| w.csftrst().set_bit());
105        self.poll(|| r.grstctl().read().csftrst().bit_is_clear())?;
106        self.poll(|| r.grstctl().read().ahbidle().bit_is_set())?;
107
108        // B. Global config: force device mode, clear pending, unmask reset + enum.
109        r.gusbcfg().modify(|_, w| w.forcedevmode().set_bit());
110        unsafe {
111            r.gintsts().write(|w| w.bits(0xFFFF_FFFF)); // clear all (W1C)
112        }
113        r.gintmsk().modify(|_, w| {
114            w.usbrstmsk().set_bit();
115            w.enumdonemsk().set_bit()
116        });
117
118        // C. Device config: high speed, address 0.
119        unsafe {
120            r.dcfg().modify(|_, w| {
121                w.devspd().bits(0); // 0 = high speed
122                w.devaddr().bits(0)
123            });
124        }
125
126        // D. Enable global interrupts.
127        r.gahbcfg().modify(|_, w| w.glblintrmsk().set_bit());
128
129        // E. Soft-connect (attach D+ pull-up): clear SFTDISCON.
130        r.dctl().modify(|_, w| w.sftdiscon().clear_bit());
131
132        // F. Wait for the host reset + enumeration, clearing each (W1C).
133        self.poll(|| r.gintsts().read().usbrst().bit_is_set())?;
134        r.gintsts().write(|w| w.usbrst().set_bit());
135        self.poll(|| r.gintsts().read().enumdone().bit_is_set())?;
136        r.gintsts().write(|w| w.enumdone().set_bit());
137
138        // G. Read the negotiated speed.
139        Ok(speed_from_enumspd(r.dsts().read().enumspd().bits()))
140    }
141
142    fn poll(&self, mut ready: impl FnMut() -> bool) -> Result<(), UsbError> {
143        for _ in 0..POLL_LIMIT {
144            if ready() {
145                return Ok(());
146            }
147            core::hint::spin_loop();
148        }
149        Err(UsbError::Timeout)
150    }
151}

155#[cfg(all(test, not(target_arch = "riscv32")))]
156mod tests {
157    use super::*;
158
159    #[test]
160    fn signature_is_ascii_ot() {
161        // The GSNPSID top-half signature is the ASCII bytes "OT" (0x4F, 0x54).
162        assert_eq!(SNPS_SIGNATURE, u16::from_be_bytes([b'O', b'T']));
163        assert_eq!(SNPS_SIGNATURE, 0x4F54);
164    }
165
166    #[test]
167    fn signature_check_uses_high_half_only() {
168        // A core-ID whose top 16 bits are the signature is recognized regardless
169        // of the (revision) low half; the low half must not affect the decision.
170        assert!(id_has_signature(0x4F54_0000));
171        assert!(id_has_signature(0x4F54_FFFF));
172        assert!(id_has_signature((SNPS_SIGNATURE as u32) << 16 | 0x1234));
173    }
174
175    #[test]
176    fn signature_check_rejects_wrong_high_half() {
177        // The signature living in the low half (or anywhere but the top) is not
178        // a match — the controller is considered absent.
179        assert!(!id_has_signature(0x0000_4F54));
180        assert!(!id_has_signature(0));
181        assert!(!id_has_signature(0xFFFF_FFFF));
182        // Off-by-one in the high half is rejected.
183        assert!(!id_has_signature(0x4F55_0000));
184        assert!(!id_has_signature(0x4F53_0000));
185    }
186
187    #[test]
188    fn enumspd_known_values() {
189        // DSTS.ENUMSPD encoding from the DWC2 device init (dwc_otgreg.h):
190        // 0=High, 1=Full(FS PHY), 2=Low, 3=Full.
191        assert_eq!(speed_from_enumspd(0), Speed::High);
192        assert_eq!(speed_from_enumspd(1), Speed::FullFs);
193        assert_eq!(speed_from_enumspd(2), Speed::Low);
194        assert_eq!(speed_from_enumspd(3), Speed::Full);
195    }
196
197    #[test]
198    fn enumspd_out_of_range_is_full() {
199        // The field is 2 bits in hardware (masked & 3), but the decode catches
200        // every value ≥ 3 via the wildcard arm → Full (never panics).
201        for v in 3u8..=u8::MAX {
202            assert_eq!(speed_from_enumspd(v), Speed::Full);
203        }
204    }
205
206    #[test]
207    fn speed_variants_are_distinct() {
208        // The four documented field values map to four distinct speeds, so the
209        // round-trip through the decoder loses no information for 0..=3.
210        let speeds = [speed_from_enumspd(0), speed_from_enumspd(1), speed_from_enumspd(2), speed_from_enumspd(3)];
211        for i in 0..speeds.len() {
212            for j in (i + 1)..speeds.len() {
213                assert_ne!(speeds[i], speeds[j]);
214            }
215        }
216    }
217}

219// ── Property-based fuzz tests ──────────────────────────────────

221#[cfg(all(test, not(target_arch = "riscv32")))]
222mod proptests {
223    use super::*;
224    use proptest::prelude::*;
225
226    proptest! {
227        /// Fuzz: signature decode never panics and matches the explicit
228        /// high-half comparison for any 32-bit core-ID.
229        #[test]
230        fn signature_matches_high_half(id in any::<u32>()) {
231            prop_assert_eq!(id_has_signature(id), (id >> 16) as u16 == SNPS_SIGNATURE);
232        }
233
234        /// Fuzz: only IDs with exactly the signature in the top half are present;
235        /// changing any high-half bit away from the signature breaks the match.
236        #[test]
237        fn signature_iff_high_half_equals(low in any::<u16>(), high in any::<u16>()) {
238            let id = ((high as u32) << 16) | low as u32;
239            prop_assert_eq!(id_has_signature(id), high == SNPS_SIGNATURE);
240        }
241
242        /// Fuzz: speed decode is total over all u8 inputs and only 0/1/2 differ
243        /// from Full; everything else collapses to Full.
244        #[test]
245        fn speed_decode_total(v in any::<u8>()) {
246            let s = speed_from_enumspd(v);
247            match v {
248                0 => prop_assert_eq!(s, Speed::High),
249                1 => prop_assert_eq!(s, Speed::FullFs),
250                2 => prop_assert_eq!(s, Speed::Low),
251                _ => prop_assert_eq!(s, Speed::Full),
252            }
253        }
254    }
