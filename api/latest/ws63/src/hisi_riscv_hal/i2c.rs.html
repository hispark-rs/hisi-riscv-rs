1//! I2C master driver for WS63 (I2C0/1, FIFO-capable).
2//!
3//! SCL = I2C_CLK / (2 * (scl_h + scl_l) * 2) where each scl value*2 = actual period.
4//! The I2C clock is the **24 MHz TCXO crystal** ([`crate::soc::chip::I2C_CLOCK_HZ`]),
5//! NOT the 240 MHz CPU clock: unlike UART/SPI, the vendor `clock_init` leaves I2C on
6//! the crystal (`i2c_port_set_clock_value(REQ_24M)`).
7
8use crate::peripherals::{I2c0, I2c1};
9use core::marker::PhantomData;
10
11/// I2C bus speed. A finite set of standard-defined modes — so a frequency the SCL
12/// divider can't actually realise is unrepresentable (the old `freq: u32` could
13/// divide to a 0 / overflowed SCL count). Matches the BS2X `i2c_v151::Speed`
14/// surface, so `hal::i2c::Speed` reads the same on both chips.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16#[cfg_attr(feature = "defmt", derive(defmt::Format))]
17#[non_exhaustive]
18pub enum Speed {
19    /// Standard mode (100 kHz).
20    Standard,
21    /// Fast mode (400 kHz).
22    Fast,
23}
24
25impl Speed {
26    /// The SCL frequency in Hz.
27    pub const fn hz(self) -> u32 {
28        match self {
29            Speed::Standard => 100_000,
30            Speed::Fast => 400_000,
31        }
32    }
33}

35/// I2C master driver bound to instance `T` (`I2c0`/`I2c1`).
36pub struct I2c<'d, T> {
37    idx: u8,
38    _peripheral: PhantomData<&'d T>,
39}
40
41fn i2c_regs(idx: u8) -> &'static crate::soc::pac::i2c0::RegisterBlock {
42    unsafe {
43        match idx {
44            0 => &*I2c0::ptr(),
45            1 => &*I2c1::ptr(),
46            _ => unreachable!(),
47        }
48    }
49}

54        configure_i2c(0, speed.hz());
55        Self { idx: 0, _peripheral: PhantomData }
56    }

68fn configure_i2c(idx: u8, freq: u32) {
69    let r = i2c_regs(idx);
70    let pclk = crate::soc::chip::I2C_CLOCK_HZ;
71    let freq = if freq == 0 { 1 } else { freq };
72    let period = pclk / (2 * freq);
73    let half = period / 2;
74    r.i2c_scl_h().write(|w| unsafe { w.bits(half) });
75    r.i2c_scl_l().write(|w| unsafe { w.bits(half) });
76    // Enable I2C in FIFO mode with all interrupts unmasked
77    r.i2c_ctrl().write(|w| unsafe {
78        w.bits(0);
79        w.i2c_en().set_bit();
80        w.mode_ctrl().set_bit();
81        // Unmask ACK error interrupt so we can detect NACK
82        w.int_ack_err_mask().set_bit()
83    });
84}

86/// Bounded busy-wait. Returns [`I2cError::Timeout`] instead of hanging the CPU
87/// forever when the bus or a peer never drives the expected status bit (mirrors
88/// the bounded waits in `spi.rs`). Previously these were unbounded `while !..{}`
89/// loops that would deadlock the core on a stuck/absent slave.
90const I2C_WAIT_LOOPS: u32 = 1_000_000;
91
92#[inline]
93fn wait_until(mut ready: impl FnMut() -> bool) -> Result<(), I2cError> {
94    let mut n = I2C_WAIT_LOOPS;
95    while !ready() {
96        n -= 1;
97        if n == 0 {
98            return Err(I2cError::Timeout);
99        }
100    }
101    Ok(())
102}

104#[inline]
105fn validate_7bit_addr(addr: u8) -> Result<u32, I2cError> {
106    if addr <= 0x7f { Ok(addr as u32) } else { Err(I2cError::InvalidAddress) }
107}

109impl<T> I2c<'_, T> {
110    #[allow(dead_code)]
111    fn wait_not_busy(&self) -> Result<(), I2cError> {
112        let r = i2c_regs(self.idx);
113        wait_until(|| !r.i2c_sr().read().bus_busy().bit_is_set())
114    }
115
116    fn clear_interrupts(&self) {
117        let r = i2c_regs(self.idx);
118        // Write 1 to each bit to clear: done, arb_loss, ack_err, rx, tx, stop, start, rxtide, txtide
119        unsafe { r.i2c_icr().write(|w| w.bits(0x7FF)) };
120    }
121
122    /// Check for ACK error (NACK from slave). Reads I2C_SR bit[2] per fbb_ws63.
123    fn check_ack(&self) -> Result<(), I2cError> {
124        let r = i2c_regs(self.idx);
125        if r.i2c_sr().read().int_ack_err().bit_is_set() {
126            return Err(I2cError::Ack);
127        }
128        Ok(())
129    }
130
131    /// Wait for TX ready and check for ACK error.
132    fn wait_tx_ack(&self) -> Result<(), I2cError> {
133        let r = i2c_regs(self.idx);
134        wait_until(|| r.i2c_sr().read().int_tx().bit_is_set())?;
135        // Check ACK after each TX (address and data bytes)
136        self.check_ack()
137    }
138
139    /// Send START + address + R/W bit via I2C_COM register.
140    /// Uses direct write (not RMW) to avoid restoring auto-cleared bits.
141    fn send_start(&self, addr_byte: u32, is_read: bool) -> Result<(), I2cError> {
142        let r = i2c_regs(self.idx);
143        self.clear_interrupts();
144
145        // Load address into TXR
146        r.i2c_txr().write(|w| unsafe { w.bits(addr_byte) });
147
148        // Direct write to COM register (bits[3:0] auto-clear after operation)
149        unsafe {
150            r.i2c_com().write(|w| w.bits(0));
151        }
152        let mut com: u32 = 0;
153        com |= 1 << 3; // op_start
154        com |= 1 << 1; // op_we (write address byte)
155        if is_read {
156            // For read, after START+WE sends address+R, we need START+RD
157            // The direction bit is encoded in addr_byte, so for address+R/W=1
158            // we just send START+WE; the read command comes separately
159        }
160        unsafe {
161            r.i2c_com().write(|w| w.bits(com));
162        }
163
164        self.wait_tx_ack()
165    }
166
167    /// Write `data` to the 7-bit `addr` (START, address+W, bytes, STOP).
168    pub fn write(&mut self, addr: u8, data: &[u8]) -> Result<(), I2cError> {
169        let addr = validate_7bit_addr(addr)?;
170        let r = i2c_regs(self.idx);
171
172        // Start + address (R/W=0)
173        self.send_start(addr << 1, false)?;
174
175        // Write data bytes
176        for &byte in data {
177            r.i2c_txr().write(|w| unsafe { w.bits(byte as u32) });
178            // Direct write to COM: op_we
179            unsafe { r.i2c_com().write(|w| w.bits(1 << 1)) };
180            self.wait_tx_ack()?;
181            self.clear_interrupts();
182        }
183
184        // Stop
185        r.i2c_com().write(|w| w.op_stop().set_bit());
186        wait_until(|| r.i2c_sr().read().int_stop().bit_is_set())?;
187        self.clear_interrupts();
188
189        Ok(())
190    }
191
192    /// Read `buf.len()` bytes from the 7-bit `addr` (START, address+R, bytes with
193    /// NACK on the last, STOP).
194    pub fn read(&mut self, addr: u8, buf: &mut [u8]) -> Result<(), I2cError> {
195        let addr = validate_7bit_addr(addr)?;
196        let r = i2c_regs(self.idx);
197
198        // Start + address (R/W=1)
199        self.send_start((addr << 1) | 1, true)?;
200
201        // Read bytes
202        let buf_len = buf.len();
203        for (i, byte) in buf.iter_mut().enumerate() {
204            let is_last = i == buf_len - 1;
205            // Direct write to COM: op_rd, optionally op_ack (NACK on last byte)
206            let mut com: u32 = 1 << 2; // op_rd
207            if is_last {
208                com |= 1 << 4; // op_ack (host sends NACK on last byte)
209            }
210            unsafe { r.i2c_com().write(|w| w.bits(com)) };
211
212            wait_until(|| r.i2c_sr().read().int_rx().bit_is_set())?;
213            *byte = r.i2c_rxr().read().bits() as u8;
214            self.clear_interrupts();
215        }
216
217        // Stop
218        r.i2c_com().write(|w| w.op_stop().set_bit());
219        wait_until(|| r.i2c_sr().read().int_stop().bit_is_set())?;
220        self.clear_interrupts();
221
222        Ok(())
223    }
224
225    /// Combined write-then-read with repeated START (Sr) between operations,
226    /// matching the I2C specification for register-based device access.
227    pub fn write_read(&mut self, addr: u8, wr_buf: &[u8], rd_buf: &mut [u8]) -> Result<(), I2cError> {
228        let addr = validate_7bit_addr(addr)?;
229        let r = i2c_regs(self.idx);
230
231        if !wr_buf.is_empty() {
232            // Start + address (R/W=0)
233            self.send_start(addr << 1, false)?;
234
235            // Write register address / data bytes
236            for &byte in wr_buf {
237                r.i2c_txr().write(|w| unsafe { w.bits(byte as u32) });
238                unsafe { r.i2c_com().write(|w| w.bits(1 << 1)) }; // op_we
239                self.wait_tx_ack()?;
240                self.clear_interrupts();
241            }
242            // NO STOP here — go directly to repeated START
243        }
244
245        if !rd_buf.is_empty() {
246            // Repeated START + address (R/W=1)
247            self.send_start((addr << 1) | 1, true)?;
248
249            let buf_len = rd_buf.len();
250            for (i, byte) in rd_buf.iter_mut().enumerate() {
251                let is_last = i == buf_len - 1;
252                let mut com: u32 = 1 << 2; // op_rd
253                if is_last {
254                    com |= 1 << 4; // op_ack (NACK on last)
255                }
256                unsafe { r.i2c_com().write(|w| w.bits(com)) };
257                wait_until(|| r.i2c_sr().read().int_rx().bit_is_set())?;
258                *byte = r.i2c_rxr().read().bits() as u8;
259                self.clear_interrupts();
260            }
261        }
262
263        // Stop (at end of combined transaction)
264        r.i2c_com().write(|w| w.op_stop().set_bit());
265        wait_until(|| r.i2c_sr().read().int_stop().bit_is_set())?;
266        self.clear_interrupts();
267
268        Ok(())
269    }
270
271    /// Core transaction implementation with repeated START support.
272    ///
273    /// Uses repeated START (Sr) between operations as required by the
274    /// embedded-hal I2c trait contract. Only emits STOP after the last operation.
275    fn transaction_impl(
276        &mut self,
277        address: u8,
278        operations: &mut [embedded_hal::i2c::Operation<'_>],
279    ) -> Result<(), I2cError> {
280        let r = i2c_regs(self.idx);
281        let address = validate_7bit_addr(address)?;
282        let addr_w = address << 1; // R/W=0 for write
283        let addr_r = (address << 1) | 1; // R/W=1 for read
284
285        for op in operations.iter_mut() {
286            match op {
287                embedded_hal::i2c::Operation::Write(data) => {
288                    self.send_start(addr_w, false)?;
289                    self.clear_interrupts();
290
291                    for &byte in data.iter() {
292                        r.i2c_txr().write(|w| unsafe { w.bits(byte as u32) });
293                        unsafe { r.i2c_com().write(|w| w.bits(1 << 1)) }; // op_we
294                        self.wait_tx_ack()?;
295                        self.clear_interrupts();
296                    }
297                    // NO STOP between operations — next START will be repeated START
298                }
299                embedded_hal::i2c::Operation::Read(buf) => {
300                    self.send_start(addr_r, true)?;
301                    self.clear_interrupts();
302
303                    let buf_len = buf.len();
304                    for (i, byte) in buf.iter_mut().enumerate() {
305                        let is_last = i == buf_len - 1;
306                        let mut com: u32 = 1 << 2; // op_rd
307                        if is_last {
308                            com |= 1 << 4; // op_ack (host sends NACK on last byte)
309                        }
310                        unsafe { r.i2c_com().write(|w| w.bits(com)) };
311                        wait_until(|| r.i2c_sr().read().int_rx().bit_is_set())?;
312                        *byte = r.i2c_rxr().read().bits() as u8;
313                        self.clear_interrupts();
314                    }
315                    // NO STOP between operations — next START will be repeated START
316                }
317            }
318        }
319
320        // STOP at end of all operations
321        r.i2c_com().write(|w| w.op_stop().set_bit());
322        wait_until(|| r.i2c_sr().read().int_stop().bit_is_set())?;
323        self.clear_interrupts();
324
325        Ok(())
326    }
327}

329/// WS63 I2C master transaction error.
330#[derive(Debug)]
331#[cfg_attr(feature = "defmt", derive(defmt::Format))]
332#[non_exhaustive]
333pub enum I2cError {
334    /// The addressed device did not acknowledge (NACK on address or data).
335    Ack,
336    /// Bus-level fault (e.g. arbitration loss / SCL stuck).
337    BusError,
338    /// A status bit never asserted within the bounded wait (stuck/absent slave).
339    Timeout,
340    /// The supplied 7-bit address was outside `0x00..=0x7f`.
341    InvalidAddress,
342}
343
344impl embedded_hal::i2c::Error for I2cError {
345    fn kind(&self) -> embedded_hal::i2c::ErrorKind {
346        match self {
347            I2cError::Ack => {
348                embedded_hal::i2c::ErrorKind::NoAcknowledge(embedded_hal::i2c::NoAcknowledgeSource::Unknown)
349            }
350            I2cError::BusError => embedded_hal::i2c::ErrorKind::Bus,
351            I2cError::Timeout | I2cError::InvalidAddress => embedded_hal::i2c::ErrorKind::Other,
352        }
353    }
354}

428// ── Property-based fuzz tests ──────────────────────────────────
429
430#[cfg(all(test, not(target_arch = "riscv32")))]
431mod proptests {
432    use crate::soc::chip::I2C_CLOCK_HZ;
433    use proptest::prelude::*;
434
435    // Re-derivation of the SCL-divider math in `configure_i2c` (same u32 types,
436    // same rounding, same zero-guard) WITHOUT touching any MMIO register:
437    //   freq   = if freq == 0 { 1 } else { freq };
438    //   period = pclk / (2 * freq);
439    //   half   = period / 2;
440    // The `2 * freq` term is a u32 multiply, so the driver's math is only
441    // well-defined for freq <= u32::MAX/2 (a value the divider would never be
442    // configured with — kHz..MHz bus speeds). We fuzz that defined regime here
443    // and exercise the zero-guard / overflow boundary in dedicated tests.
444    fn scl_half(freq: u32) -> u32 {
445        let pclk = I2C_CLOCK_HZ;
446        let freq = if freq == 0 { 1 } else { freq };
447        let period = pclk / (2 * freq);
448        period / 2
449    }
450
451    proptest! {
452        /// Fuzz: the divider never panics (no div-by-zero) across the full u32
453        /// frequency range — the `freq == 0` guard forces the divisor to >= 2.
454        #[test]
455        fn divider_never_div_by_zero(freq in any::<u32>()) {
456            // Mirror the guard, then the divisor `2 * freq` is computed in u64 to
457            // avoid the test itself overflowing; the property under test is that a
458            // non-zero divisor exists for every input (no div-by-zero panic).
459            let f = if freq == 0 { 1u64 } else { freq as u64 };
460            let divisor = 2 * f;
461            prop_assert!(divisor >= 2);
462            let _ = I2C_CLOCK_HZ as u64 / divisor;
463        }
464
465        /// Fuzz: across realistic bus frequencies (1 Hz ..= I2C_CLOCK_HZ) the
466        /// half-period field never exceeds the source clock — it is a fraction of
467        /// `pclk`, so it always fits whatever sane SCL_H/SCL_L width the HW has.
468        #[test]
469        fn half_period_within_clock(freq in 1u32..=I2C_CLOCK_HZ) {
470            prop_assert!(scl_half(freq) <= I2C_CLOCK_HZ);
471        }
472
473        /// Fuzz: the divider is monotonic — a higher requested SCL frequency never
474        /// yields a larger half-period (faster bus ⇒ shorter clock half-period).
475        #[test]
476        fn divider_is_monotonic(a in 1u32..=I2C_CLOCK_HZ, b in 1u32..=I2C_CLOCK_HZ) {
477            let (lo, hi) = if a <= b { (a, b) } else { (b, a) };
478            // lo <= hi requested freq ⇒ half(lo) >= half(hi)
479            prop_assert!(scl_half(lo) >= scl_half(hi));
480        }
481
482        /// Fuzz: the `freq == 0` guard means a 0 request is treated identically to
483        /// a 1 Hz request (no panic, deterministic value), never a div-by-zero.
484        #[test]
485        fn zero_freq_guard_matches_one(_seed in 0u32..4) {
486            prop_assert_eq!(scl_half(0), scl_half(1));
487        }
488
489        /// Fuzz: the 7-bit write-address encoding (`addr << 1`, R/W = 0) keeps the
490        /// LSB clear and round-trips back to the original address.
491        #[test]
492        fn write_addr_encoding_roundtrips(addr in 0u8..=0x7F) {
493            let addr_w = (addr as u32) << 1;
494            prop_assert_eq!(addr_w & 0x01, 0, "write R/W bit must be clear");
495            prop_assert_eq!(addr_w >> 1, addr as u32, "address must round-trip");
496            prop_assert!(addr_w <= 0xFE, "7-bit write byte stays in a u8");
497        }
498
499        /// Fuzz: the read-address encoding (`(addr << 1) | 1`, R/W = 1) sets the
500        /// LSB, round-trips, and differs from the write byte in exactly bit 0.
501        #[test]
502        fn read_addr_encoding_roundtrips(addr in 0u8..=0x7F) {
503            let addr_w = (addr as u32) << 1;
504            let addr_r = ((addr as u32) << 1) | 1;
505            prop_assert_eq!(addr_r & 0x01, 1, "read R/W bit must be set");
506            prop_assert_eq!(addr_r >> 1, addr as u32, "address must round-trip");
507            prop_assert_eq!(addr_w ^ addr_r, 1, "write/read bytes differ only in bit 0");
508        }
509    }
