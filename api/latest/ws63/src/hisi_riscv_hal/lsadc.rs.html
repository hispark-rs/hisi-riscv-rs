1//! Low-Speed ADC (LSADC) driver for WS63 — v154 controller.
2//!
3//! Register map and bit fields are cross-checked against the WS63 C SDK
4//! (`hal_adc_v154_regs_def.h`, `hal_adc_v154_regs_op.{c,h}`, `hal_adc_v154.c`).
5//! The control bank is the contiguous `adc_regs_t` struct at base `0x4400_C000`:
6//!
7//! | Register      | Offset | Purpose                                            |
8//! |---------------|--------|----------------------------------------------------|
9//! | `LSADC_CTRL_0`  | 0x00 | scan config: per-channel enable + sample timing    |
10//! | `LSADC_CTRL_1`  | 0x04 | FIFO status (`rne`/`rff`/`bsy`) + waterline        |
11//! | `LSADC_CTRL_2`  | 0x08 | interrupt mask/status                              |
12//! | `LSADC_CTRL_8`  | 0x1C | scan start/stop                                    |
13//! | `LSADC_CTRL_9`  | 0x20 | FIFO read data (`data[13:0]`, `channel[16:14]`)    |
14//! | `LSADC_CTRL_11` | 0x24 | analog enable (`da_lsadc_en`) + reset (`rstn`@16)  |
15//! | `CFG_DATA_SEL`  | 0xDC | data output select                                 |
16//! | `CFG_OFFSET`    | 0xE0 | offset correction                                  |
17//! | `CFG_GAIN`      | 0xE4 | gain correction                                    |
18//! | `CFG_CIC_FILTER_EN` | 0xE8 | CIC filter enable                              |
19//! | `CFG_CIC_OSR`   | 0xEC | CIC oversampling ratio                             |
20//!
21//! # Status
22//!
23//! Not validated on silicon. The full analog power-up sequence (the SDK's
24//! `hal_adc_simulation_cfg` magic writes to `da_lsadc_en` and the `da_lsadc_rwreg`
25//! registers, plus offset/cap/gain calibration) is **not** implemented here;
26//! [`LsAdc::set_analog_enable`] exposes `da_lsadc_en` for callers that port it.
27
28use crate::peripherals::Lsadc;

30/// LSADC channel (0-5).
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub enum AdcChannel {
33    /// Channel 0 (`LSADC_CTRL_0.channel` bit 0).
34    Channel0 = 0,
35    /// Channel 1 (`LSADC_CTRL_0.channel` bit 1).
36    Channel1 = 1,
37    /// Channel 2 (`LSADC_CTRL_0.channel` bit 2).
38    Channel2 = 2,
39    /// Channel 3 (`LSADC_CTRL_0.channel` bit 3).
40    Channel3 = 3,
41    /// Channel 4 (`LSADC_CTRL_0.channel` bit 4).
42    Channel4 = 4,
43    /// Channel 5 (`LSADC_CTRL_0.channel` bit 5).
44    Channel5 = 5,
45}
46
47impl AdcChannel {
48    /// Convert a channel index (0-5) to an `AdcChannel`.
49    pub fn from_index(idx: u8) -> Option<Self> {
50        match idx {
51            0 => Some(Self::Channel0),
52            1 => Some(Self::Channel1),
53            2 => Some(Self::Channel2),
54            3 => Some(Self::Channel3),
55            4 => Some(Self::Channel4),
56            5 => Some(Self::Channel5),
57            _ => None,
58        }
59    }
60}

62/// Averaging mode (`equ_model_sel`, `LSADC_CTRL_0[7:6]`).
63#[derive(Debug, Clone, Copy, PartialEq, Eq)]
64pub enum Averaging {
65    /// 1 sample averaged.
66    One = 0,
67    /// 2 samples averaged.
68    Two = 1,
69    /// 4 samples averaged.
70    Four = 2,
71    /// 8 samples averaged.
72    Eight = 3,
73}

75/// A validated `sample_cnt` (`LSADC_CTRL_0`, 5-bit). [`SampleCount::new`] returns
76/// `None` for `> 31`, so [`AdcConfig`] can never silently `& 0x1F`-truncate it.
77#[derive(Debug, Clone, Copy, PartialEq, Eq)]
78#[cfg_attr(feature = "defmt", derive(defmt::Format))]
79pub struct SampleCount(u8);
80
81impl SampleCount {
82    /// Construct from a raw count. `None` if `> 31` (the 5-bit field maximum).
83    pub const fn new(n: u8) -> Option<Self> {
84        if n <= 0x1F { Some(Self(n)) } else { None }
85    }
86    /// The raw 5-bit count.
87    pub const fn bits(self) -> u8 {
88        self.0
89    }
90}
91
92/// A validated `cast_cnt` (`LSADC_CTRL_0`, 7-bit). [`CastCount::new`] returns `None`
93/// for `> 127`, so [`AdcConfig`] can never silently `& 0x7F`-truncate it.
94#[derive(Debug, Clone, Copy, PartialEq, Eq)]
95#[cfg_attr(feature = "defmt", derive(defmt::Format))]
96pub struct CastCount(u8);
97
98impl CastCount {
99    /// Construct from a raw count. `None` if `> 127` (the 7-bit field maximum).
100    pub const fn new(n: u8) -> Option<Self> {
101        if n <= 0x7F { Some(Self(n)) } else { None }
102    }
103    /// The raw 7-bit count.
104    pub const fn bits(self) -> u8 {
105        self.0
106    }
107}

109/// A validated RX-FIFO interrupt waterline (`LSADC_CTRL_1.rxintsize`, 3-bit).
110/// [`FifoWaterline::new`] returns `None` for `> 7`, so the currently unstable
111/// `LsAdc::set_fifo_waterline` can never silently `& 0x07`-truncate it.
112#[derive(Debug, Clone, Copy, PartialEq, Eq)]
113#[cfg_attr(feature = "defmt", derive(defmt::Format))]
114pub struct FifoWaterline(u8);
115
116impl FifoWaterline {
117    /// Construct from a raw level. `None` if `> 7` (the 3-bit field maximum).
118    pub const fn new(level: u8) -> Option<Self> {
119        if level <= 0x07 { Some(Self(level)) } else { None }
120    }
121    /// The raw 3-bit level.
122    pub const fn bits(self) -> u8 {
123        self.0
124    }
125}

127/// Scan-mode sample timing (`LSADC_CTRL_0`). Defaults match the SDK's
128/// `hal_adc_auto_scan_mode_set` (8× averaging, `sample_cnt=8`, `start_cnt=0x18`).
129///
130/// The count fields are validated newtypes ([`SampleCount`]/[`CastCount`]) so an
131/// out-of-range value is rejected at construction instead of silently masked here.
132#[derive(Debug, Clone, Copy)]
133pub struct AdcConfig {
134    /// Averaging mode (`equ_model_sel`, 2-bit).
135    pub averaging: Averaging,
136    /// Sample count (`sample_cnt`, validated 5-bit).
137    pub sample_count: SampleCount,
138    /// Start count (`start_cnt` / SDK `satrt_cnt`, 8-bit — the full `u8` field).
139    pub start_count: u8,
140    /// Cast count (`cast_cnt`, validated 7-bit).
141    pub cast_count: CastCount,
142}
143
144impl Default for AdcConfig {
145    fn default() -> Self {
146        Self {
147            averaging: Averaging::Eight,
148            // 0x8 and 0x0 are in range, so these unwraps never fire.
149            sample_count: SampleCount::new(0x8).unwrap(),
150            start_count: 0x18,
151            cast_count: CastCount::new(0x0).unwrap(),
152        }
153    }
154}

156/// ADC conversion result.
157#[derive(Debug, Clone, Copy, PartialEq, Eq)]
158pub struct AdcSample {
159    /// 14-bit conversion code.
160    pub data: u16,
161    /// Channel that produced this sample (0-5).
162    pub channel: u8,
163}
164
165/// Parse a raw `LSADC_CTRL_9` word into a sample: `data` = bits `[13:0]`,
166/// `channel` = bits `[16:14]` (matches `adc_fifo_data_str` in the SDK).
167#[inline]
168const fn parse_sample(raw: u32) -> AdcSample {
169    AdcSample { data: (raw & 0x3FFF) as u16, channel: ((raw >> 14) & 0x07) as u8 }
170}

172/// LSADC driver.
173pub struct LsAdc<'d> {
174    _lsadc: Lsadc<'d>,
175}
176
177impl<'d> LsAdc<'d> {
178    /// Create a new LSADC driver from the LSADC peripheral.
179    pub fn new(lsadc: Lsadc<'d>) -> Self {
180        Self { _lsadc: lsadc }
181    }
182
183    fn regs(&self) -> &'static crate::soc::pac::lsadc::RegisterBlock {
184        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid.
185        unsafe { &*Lsadc::ptr() }
186    }
187
188    /// Release the analog reset (`LSADC_CTRL_11.da_lsadc_rstn = 1`, active-low).
189    #[instability::unstable]
190    pub fn enable(&mut self) {
191        self.regs().lsadc_ctrl_11().modify(|_, w| w.da_lsadc_rstn().set_bit());
192    }
193
194    /// Assert the analog reset (`LSADC_CTRL_11.da_lsadc_rstn = 0`).
195    #[instability::unstable]
196    pub fn disable(&mut self) {
197        self.regs().lsadc_ctrl_11().modify(|_, w| w.da_lsadc_rstn().clear_bit());
198    }
199
200    /// Write the 16-bit analog-enable field (`LSADC_CTRL_11.da_lsadc_en`).
201    ///
202    /// The SDK power-up sequence ORs in `0x7000`, `0xE7F`, `0x100`, `0x80`
203    /// across several steps; this exposes the raw field for porting it.
204    #[instability::unstable]
205    pub fn set_analog_enable(&mut self, bits: u16) {
206        self.regs().lsadc_ctrl_11().modify(|_, w| unsafe { w.da_lsadc_en().bits(bits) });
207    }
208
209    /// Enable a channel and program the scan timing (`LSADC_CTRL_0`).
210    ///
211    /// Sets the per-channel enable bit (preserving any already-enabled channels)
212    /// and the averaging/sample/start/cast counts, matching
213    /// `hal_adc_auto_scan_mode_set`.
214    pub fn configure_scan(&mut self, channel: AdcChannel, config: &AdcConfig) {
215        self.regs().lsadc_ctrl_0().modify(|r, w| {
216            let ch = r.channel().bits() | (1 << (channel as u8));
217            unsafe {
218                w.channel().bits(ch);
219                w.equ_model_sel().bits(config.averaging as u8);
220                // Pre-validated by the newtypes — no silent mask needed.
221                w.sample_cnt().bits(config.sample_count.bits());
222                w.start_cnt().bits(config.start_count);
223                w.cast_cnt().bits(config.cast_count.bits())
224            }
225        });
226    }
227
228    /// Start an ADC scan (`LSADC_CTRL_8.lsadc_start = 1`).
229    #[instability::unstable]
230    pub fn start_scan(&mut self) {
231        self.regs().lsadc_ctrl_8().write(|w| w.lsadc_start().set_bit());
232    }
233
234    /// Stop an ADC scan (`LSADC_CTRL_8.lsadc_stop = 1`).
235    #[instability::unstable]
236    pub fn stop_scan(&mut self) {
237        self.regs().lsadc_ctrl_8().write(|w| w.lsadc_stop().set_bit());
238    }
239
240    /// Set the RX-FIFO interrupt waterline (`LSADC_CTRL_1.rxintsize`, 3-bit). The
241    /// level is a validated [`FifoWaterline`] so an out-of-range value is rejected
242    /// at construction rather than silently masked here.
243    #[instability::unstable]
244    pub fn set_fifo_waterline(&mut self, level: FifoWaterline) {
245        self.regs().lsadc_ctrl_1().modify(|_, w| unsafe { w.rxintsize().bits(level.bits()) });
246    }
247
248    /// True if the RX FIFO holds at least one sample (`LSADC_CTRL_1.rne`).
249    ///
250    /// This is the reliable empty check — read it before [`Self::read_sample`].
251    #[instability::unstable]
252    pub fn data_ready(&self) -> bool {
253        self.regs().lsadc_ctrl_1().read().rne().bit_is_set()
254    }
255
256    /// Read one sample from the FIFO (`LSADC_CTRL_9`).
257    ///
258    /// Returns `None` when the FIFO is empty (checked via `rne`), so a genuine
259    /// 0-code reading is **not** mistaken for "no data".
260    #[instability::unstable]
261    pub fn read_sample(&self) -> Option<AdcSample> {
262        if !self.data_ready() {
263            return None;
264        }
265        Some(parse_sample(self.regs().lsadc_ctrl_9().read().bits()))
266    }
267
268    /// Enable the CIC filter with the given oversampling ratio.
269    #[instability::unstable]
270    pub fn enable_cic_filter(&mut self, oversampling_ratio: u8) {
271        let r = self.regs();
272        unsafe {
273            r.cfg_cic_osr().write(|w| w.cic_osr().bits(oversampling_ratio));
274        }
275        r.cfg_cic_filter_en().write(|w| w.cic_filter_en().set_bit());
276    }
277
278    /// Disable the CIC filter.
279    #[instability::unstable]
280    pub fn disable_cic_filter(&mut self) {
281        self.regs().cfg_cic_filter_en().write(|w| w.cic_filter_en().clear_bit());
282    }
283
284    /// Set the ADC offset correction value (`CFG_OFFSET`).
285    #[instability::unstable]
286    pub fn set_offset(&mut self, offset: u16) {
287        self.regs().cfg_offset().write(|w| unsafe { w.offset().bits(offset) });
288    }
289
290    /// Set the ADC gain correction value (`CFG_GAIN`).
291    #[instability::unstable]
292    pub fn set_gain(&mut self, gain: u16) {
293        self.regs().cfg_gain().write(|w| unsafe { w.gain().bits(gain) });
294    }
295
296    /// Select the data source: `true` = post-processed, `false` = raw ADC code.
297    #[instability::unstable]
298    pub fn set_data_select(&mut self, processed: bool) {
299        self.regs().cfg_data_sel().write(|w| w.data_sel().bit(processed));
300    }
301}

305#[cfg(all(test, not(target_arch = "riscv32")))]
306mod tests {
307    use super::*;
308
309    #[test]
310    fn test_parse_sample_data_and_channel() {
311        // channel 3, code 0x0AAA  ->  raw = (3 << 14) | 0x0AAA
312        let raw: u32 = (3 << 14) | 0x0AAA;
313        let s = parse_sample(raw);
314        assert_eq!(s.data, 0x0AAA);
315        assert_eq!(s.channel, 3);
316    }
317
318    #[test]
319    fn test_parse_sample_max_code() {
320        let s = parse_sample(0x3FFF); // all data bits, channel 0
321        assert_eq!(s.data, 0x3FFF);
322        assert_eq!(s.channel, 0);
323    }
324
325    #[test]
326    fn test_parse_sample_channel_field_is_3_bits() {
327        // bits above [16:14] must not leak into channel
328        let raw: u32 = 0xFFFF_FFFF;
329        let s = parse_sample(raw);
330        assert_eq!(s.channel, 0x07);
331        assert_eq!(s.data, 0x3FFF);
332    }
333
334    #[test]
335    fn test_channel_from_index() {
336        assert_eq!(AdcChannel::from_index(0), Some(AdcChannel::Channel0));
337        assert_eq!(AdcChannel::from_index(5), Some(AdcChannel::Channel5));
338        assert_eq!(AdcChannel::from_index(6), None);
339        assert_eq!(AdcChannel::from_index(255), None);
340    }
341
342    #[test]
343    fn test_default_config_matches_sdk() {
344        let c = AdcConfig::default();
345        assert_eq!(c.averaging as u8, 3); // AVERAGE_OF_EIGHT_SAMPLES
346        assert_eq!(c.sample_count.bits(), 0x8);
347        assert_eq!(c.start_count, 0x18);
348        assert_eq!(c.cast_count.bits(), 0x0);
349    }
350
351    #[test]
352    fn count_newtypes_reject_out_of_range() {
353        // 5-bit sample_cnt, 7-bit cast_cnt, 3-bit waterline — boundaries enforced.
354        assert!(SampleCount::new(0x1F).is_some());
355        assert!(SampleCount::new(0x20).is_none());
356        assert!(CastCount::new(0x7F).is_some());
357        assert!(CastCount::new(0x80).is_none());
358        assert!(FifoWaterline::new(7).is_some());
359        assert!(FifoWaterline::new(8).is_none());
360    }
361}

399// ── Async LSADC (bespoke; LSADC_INTR = IRQ 72) ──────────────────────────────
400#[cfg(all(feature = "chip-ws63", feature = "async", feature = "unstable"))]
401mod asynch_impl {
402    use super::{AdcSample, LsAdc};
403    use crate::asynch::IrqSignal;
404    use crate::interrupt::{self, Interrupt};
405    use core::future::Future;
406    use core::pin::Pin;
407    use core::task::{Context, Poll};
408
409    static LSADC_SIGNAL: IrqSignal = IrqSignal::new();
410
411    /// LSADC trap hook (IRQ 72): wake the awaiting conversion. The sample is
412    /// read-cleared by `read_sample`, so the ISR just signals + clears pending.
413    pub fn on_interrupt() {
414        LSADC_SIGNAL.signal();
415        interrupt::clear_pending(Interrupt::LSADC_INTR);
416    }
417
418    /// Named device.x handler (LSADC_INTR = IRQ 72): the rt routes the LSADC IRQ
419    /// here by number, so an async LSADC app needs no `mcause` trap.
420    #[unsafe(no_mangle)]
421    extern "C" fn LSADC_INTR() {
422        on_interrupt();
423    }
424
425    struct ConvFuture;
426    impl Future for ConvFuture {
427        type Output = ();
428        fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
429            if LSADC_SIGNAL.take_fired() {
430                Poll::Ready(())
431            } else {
432                LSADC_SIGNAL.register(cx.waker());
433                Poll::Pending
434            }
435        }
436    }
437
438    impl LsAdc<'_> {
439        /// Start a scan and await conversion-complete (IRQ 72), returning a sample.
440        /// On hardware this parks until the IRQ; the WS63 model fills the FIFO
441        /// synchronously, so the fast path returns without parking.
442        pub async fn read_async(&mut self) -> Option<AdcSample> {
443            LSADC_SIGNAL.reset();
444            // SAFETY: enabling a known, fixed WS63 IRQ line.
445            unsafe { interrupt::enable(Interrupt::LSADC_INTR) };
446            self.start_scan();
447            if !self.data_ready() {
448                ConvFuture.await;
449            }
450            self.read_sample()
451        }
452    }
453}
