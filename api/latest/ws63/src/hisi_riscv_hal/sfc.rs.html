1//! SFC (SPI Flash Controller) driver for WS63.
2//!
3//! The WS63 SFC provides a high-speed interface to external SPI NOR Flash
4//! memory. It supports standard, dual, and quad SPI modes, configurable
5//! timing, command-based operations, and bus DMA for efficient data transfer.
6//!
7//! # Features
8//!
9//! - Standard/Dual/Quad SPI modes
10//! - 3-byte and 4-byte addressing
11//! - Configurable SPI mode (Mode0/Mode3)
12//! - Command-based flash operations (read, write, erase)
13//! - 16-word (64-byte) data buffer for indirect operations
14//! - Bus DMA for direct memory-mapped flash access
15//! - Hardware write protect
16//! - AES low-power mode for XIP encryption
17

20/// SPI interface type for flash operations.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum SpiIfType {
23    /// Standard SPI (1 data line).
24    Standard = 0,
25    /// Dual I/O (2 data lines).
26    DualIO = 1,
27    /// Dual I/O continuous.
28    DualIOCont = 2,
29    /// Quad I/O (4 data lines).
30    QuadIO = 3,
31    /// Quad I/O continuous.
32    QuadIOCont = 4,
33}

120    /// Configure the global SFC settings.
121    pub fn configure_global(
122        &mut self,
123        spi_mode: FlashSpiMode,
124        addr_mode: AddressMode,
125        read_delay: ReadDelay,
126        write_protect: bool,
127    ) {
128        let mut val: u32 = 0;
129        val |= spi_mode as u32; // mode [0]
130        if write_protect {
131            val |= 1 << 1; // wp_en [1]
132        }
133        if matches!(addr_mode, AddressMode::FourByte) {
134            val |= 1 << 2; // flash_addr_mode [2]
135        }
136        val |= (read_delay as u32) << 3; // rd_delay [3:5]
137
138        unsafe {
139            self.regs().global_config().write(|w| w.bits(val));
140        }
141    }

143    /// Configure the SFC timing parameters.
144    ///
145    /// * `tshsl` — Inter-operation delay: (tshsl + 2) clock cycles
146    /// * `tcss` — CS setup time: (tcss + 1) clock cycles
147    /// * `tcsh` — CS hold time: (tcsh + 1) clock cycles
148    pub fn configure_timing(&mut self, tshsl: u8, tcss: u8, tcsh: u8) {
149        // Mask to the 4-bit field FIRST, then floor to the minimum. Doing it the
150        // other way (max() before the mask) lets a value whose low nibble is
151        // below the floor — e.g. 16/32/48, low nibble 0 — survive the max() only
152        // to be masked back down to 0, defeating the MIN_TSHSL guarantee. Mask
153        // then floor keeps the field in [MIN_TSHSL-2, 0xF] for every input.
154        let tshsl = ((tshsl as u32) & 0x0F).max(MIN_TSHSL - 2);
155        let tcss = (tcss as u32) & 0x07;
156        let tcsh = (tcsh as u32) & 0x07;
157
158        let val = tshsl | (tcss << 8) | (tcsh << 12);
159
160        unsafe {
161            self.regs().timing().write(|w| w.bits(val));
162        }
163    }

167    /// Configure the bus read/write parameters for memory-mapped access.
168    pub fn configure_bus(&mut self, config: &BusConfig) {
169        let r = self.regs();
170
171        // BUS_CONFIG1
172        let mut cfg1: u32 = 0;
173        cfg1 |= (config.read_if_type as u32) & 0x07; // rd_mem_if_type [0:2]
174        cfg1 |= ((config.read_dummy_bytes as u32) & 0x07) << 3; // rd_dummy_bytes [3:5]
175        cfg1 |= ((config.read_prefetch_cnt as u32) & 0x03) << 6; // rd_prefetch_cnt [6:7]
176        cfg1 |= ((config.read_instruction as u32) & 0xFF) << 8; // rd_ins [8:15]
177        cfg1 |= ((config.write_if_type as u32) & 0x07) << 16; // wr_mem_if_type [16:18]
178        cfg1 |= ((config.write_dummy_bytes as u32) & 0x07) << 19; // wr_dummy_bytes [19:21]
179        cfg1 |= ((config.write_instruction as u32) & 0xFF) << 22; // wr_ins [22:29]
180
181        unsafe {
182            r.bus_config1().write(|w| w.bits(cfg1));
183        }
184    }

282    /// Execute a flash command (no data phase).
283    ///
284    /// * `instruction` — Flash operation code.
285    /// * `address` — Operation address (for commands with address phase).
286    /// * `address_enable` — Whether the command includes an address phase.
287    pub fn send_command(&mut self, instruction: u8, address: u32, address_enable: bool) {
288        let r = self.regs();
289
290        // Write instruction
291        unsafe {
292            r.cmd_ins().write(|w| w.bits(instruction as u32));
293        }
294
295        // Write address
296        if address_enable {
297            unsafe {
298                r.cmd_addr().write(|w| w.bits(address));
299            }
300        }
301
302        // Build command config
303        let mut cmd_cfg: u32 = 0;
304        cmd_cfg |= 0x01; // start
305        if address_enable {
306            cmd_cfg |= 1 << 2; // addr_en
307        }
308        cmd_cfg |= 0 << 17; // mem_if_type = Standard
309
310        unsafe {
311            r.cmd_config().write(|w| w.bits(cmd_cfg));
312        }
313
314        // Wait for command completion
315        while !self.command_done() {}
316        self.clear_interrupts();
317    }

319    /// Execute a flash command with data phase.
320    ///
321    /// * `instruction` — Flash operation code.
322    /// * `address` — Operation address.
323    /// * `write_data` — Data to write (for write commands).
324    /// * `read` — `true` for read commands, `false` for write commands.
325    ///
326    /// Returns the read data (up to 64 bytes) for read commands.
327    pub fn command_with_data(
328        &mut self,
329        instruction: u8,
330        address: u32,
331        write_data: &[u8],
332        read: bool,
333    ) -> Result<[u8; 64], SfcError> {
334        let r = self.regs();
335        // The command data buffer is 64 bytes; reject an over-long write rather than
336        // silently truncating it (the old `.min(64)` dropped the tail).
337        if write_data.len() > 64 {
338            return Err(SfcError::BufferTooLong);
339        }
340        let data_len = write_data.len();
341
342        if !read && !write_data.is_empty() {
343            // Load write data into data buffer
344            for (i, chunk) in write_data[..data_len].chunks(4).enumerate() {
345                let mut word: u32 = 0;
346                for (j, &b) in chunk.iter().enumerate() {
347                    word |= (b as u32) << (j * 8);
348                }
349                Self::write_databuf(r, i, word);
350            }
351        }
352
353        // Write instruction and address
354        unsafe {
355            r.cmd_ins().write(|w| w.bits(instruction as u32));
356            r.cmd_addr().write(|w| w.bits(address));
357        }
358
359        // Build command config
360        let mut cmd_cfg: u32 = 0;
361        cmd_cfg |= 0x01; // start
362        cmd_cfg |= 1 << 2; // addr_en
363        cmd_cfg |= 1 << 7; // data_en
364        if read {
365            cmd_cfg |= 1 << 8; // rw = read
366        }
367        cmd_cfg |= (((data_len.saturating_sub(1)) as u32) & 0x3F) << 9; // data_cnt
368        cmd_cfg |= 0 << 17; // mem_if_type = Standard
369
370        unsafe {
371            r.cmd_config().write(|w| w.bits(cmd_cfg));
372        }
373
374        // Wait for command completion
375        while !self.command_done() {}
376        self.clear_interrupts();
377
378        // Read back data for read commands
379        let mut result = [0u8; 64];
380        if read {
381            let mut idx = 0;
382            for i in 0..data_len.div_ceil(4) {
383                let word = Self::read_databuf(r, i);
384                let bytes = word.to_le_bytes();
385                for &b in &bytes {
386                    if idx < data_len {
387                        result[idx] = b;
388                        idx += 1;
389                    }
390                }
391            }
392        }
393
394        Ok(result)
395    }

399    /// Start a bus DMA transfer between flash and memory.
400    ///
401    /// * `mem_addr` — Memory address (must be in valid range).
402    /// * `flash_addr` — Flash address.
403    /// * `length` — Number of bytes to transfer.
404    /// * `read` — `true` for flash-to-memory read, `false` for memory-to-flash write.
405    pub fn bus_dma_start(&mut self, mem_addr: u32, flash_addr: u32, length: u32, read: bool) {
406        let r = self.regs();
407
408        unsafe {
409            r.bus_dma_mem_saddr().write(|w| w.bits(mem_addr));
410            r.bus_dma_flash_saddr().write(|w| w.bits(flash_addr));
411            r.bus_dma_len().write(|w| w.bits(length & 0x3FFF_FFFF));
412        }
413
414        let mut ctrl: u32 = 0;
415        ctrl |= 0x01; // dma_start
416        if read {
417            ctrl |= 1 << 1; // dma_rw = read
418        }
419
420        unsafe {
421            r.bus_dma_ctrl().write(|w| w.bits(ctrl));
422        }
423    }

425    /// Wait for bus DMA to complete.
426    pub fn bus_dma_wait(&self) {
427        while !self.dma_done() {}
428        self.clear_interrupts();
429    }

448    /// Enable specific SFC interrupts.
449    ///
450    /// * `cmd_done` — Command operation complete interrupt.
451    /// * `dma_done` — DMA transfer complete interrupt.
452    pub fn enable_interrupts(&mut self, cmd_done: bool, dma_done: bool) {
453        let mut mask: u32 = 0;
454        if cmd_done {
455            mask |= 0x01;
456        }
457        if dma_done {
458            mask |= 0x02;
459        }
460        unsafe {
461            self.regs().int_mask().write(|w| w.bits(mask));
462        }
463    }

527    /// Re-derivation of the GLOBAL_CONFIG word built by `configure_global`.
528    fn global_config_bits(
529        spi_mode: FlashSpiMode,
530        addr_mode: AddressMode,
531        read_delay: ReadDelay,
532        write_protect: bool,
533    ) -> u32 {
534        let mut val: u32 = 0;
535        val |= spi_mode as u32; // mode [0]
536        if write_protect {
537            val |= 1 << 1; // wp_en [1]
538        }
539        if matches!(addr_mode, AddressMode::FourByte) {
540            val |= 1 << 2; // flash_addr_mode [2]
541        }
542        val |= (read_delay as u32) << 3; // rd_delay [3:5]
543        val
544    }

546    /// Re-derivation of the TIMING word built by `configure_timing`.
547    fn timing_bits(tshsl: u8, tcss: u8, tcsh: u8) -> u32 {
548        let tshsl = ((tshsl as u32) & 0x0F).max(MIN_TSHSL - 2);
549        let tcss = (tcss as u32) & 0x07;
550        let tcsh = (tcsh as u32) & 0x07;
551        tshsl | (tcss << 8) | (tcsh << 12)
552    }

554    /// Re-derivation of the BUS_CONFIG1 word built by `configure_bus`.
555    fn bus_config1_bits(config: &BusConfig) -> u32 {
556        let mut cfg1: u32 = 0;
557        cfg1 |= (config.read_if_type as u32) & 0x07;
558        cfg1 |= ((config.read_dummy_bytes as u32) & 0x07) << 3;
559        cfg1 |= ((config.read_prefetch_cnt as u32) & 0x03) << 6;

561        cfg1 |= ((config.write_if_type as u32) & 0x07) << 16;
562        cfg1 |= ((config.write_dummy_bytes as u32) & 0x07) << 19;
563        cfg1 |= ((config.write_instruction as u32) & 0xFF) << 22;
564        cfg1

759#[cfg(all(test, not(target_arch = "riscv32")))]
760mod proptests {
761    use super::MIN_TSHSL;
762    use proptest::prelude::*;
763
764    fn timing_bits(tshsl: u8, tcss: u8, tcsh: u8) -> u32 {
765        let tshsl = ((tshsl as u32) & 0x0F).max(MIN_TSHSL - 2);
766        let tcss = (tcss as u32) & 0x07;
767        let tcsh = (tcsh as u32) & 0x07;
768        tshsl | (tcss << 8) | (tcsh << 12)
769    }
770
771    proptest! {
772        /// Fuzz: the timing word only ever sets bits inside its three fields.
773        #[test]
774        fn timing_sets_no_stray_bits(tshsl in any::<u8>(), tcss in any::<u8>(), tcsh in any::<u8>()) {
775            let v = timing_bits(tshsl, tcss, tcsh);
776            // Allowed bits: [0:3] | [8:10] | [12:14].
777            let allowed: u32 = 0x0F | (0x07 << 8) | (0x07 << 12);
778            prop_assert_eq!(v & !allowed, 0);
779        }
780
781        /// Fuzz: tshsl field is always at least the clamped floor (MIN_TSHSL-2).
782        #[test]
783        fn timing_tshsl_never_below_floor(tshsl in any::<u8>()) {
784            let floor = (MIN_TSHSL as u8 - 2) as u32;
785            let field = timing_bits(tshsl, 0, 0) & 0x0F;
786            prop_assert!(field >= floor);
787        }
788
789        /// Fuzz: LE word packing of a 4-byte chunk always round-trips.
790        #[test]
791        fn databuf_pack_round_trips(b in any::<[u8; 4]>()) {
792            let mut word: u32 = 0;
793            for (j, &x) in b.iter().enumerate() {
794                word |= (x as u32) << (j * 8);
795            }
796            prop_assert_eq!(word.to_le_bytes(), b);
797        }
798
799        /// Fuzz: the data-count field never exceeds its 6-bit width.
800        #[test]
801        fn data_count_fits_6_bits(len in 0usize..=64) {
802            let field = ((len.saturating_sub(1)) as u32) & 0x3F;
803            prop_assert!(field <= 0x3F);
804        }
805    }
