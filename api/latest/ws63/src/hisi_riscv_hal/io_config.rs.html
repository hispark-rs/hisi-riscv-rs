1//! IO Configuration (Pin Mux and Pad Control) driver for WS63.
2//!
3//! The WS63 IO_CONFIG peripheral controls pin function selection (muxing)
4//! and pad characteristics (drive strength, pull resistors, Schmitt trigger,
5//! input enable) for all GPIO, UART, and SFC pads.
6//!
7//! # Pin mux
8//!
9//! Each GPIO pin can be assigned one of several functions via a 3-bit
10//! mux select register. The exact function mapping depends on the chip
11//! configuration.
12//!
13//! # Pad control
14//!
15//! Each pad has independent control over:
16//! - Drive strength (3-bit, 000 = strongest, 111 = weakest)
17//! - Pull resistor (none, pull-up, pull-down)
18//! - Schmitt trigger (enable/disable)
19//! - Input enable (enable/disable)
20

466/// Build a pad control register value.
467fn build_pad_ctrl(drive: DriveStrength, pull: PullResistor, schmitt_trigger: bool, input_enable: bool) -> u32 {
468    let mut val: u32 = 0;
469
470    // Drive strength: ds2=bit6, ds1=bit5, ds0=bit4
471    let ds = drive as u32;
472    val |= ((ds & 0x04) << 4) | ((ds & 0x02) << 4) | ((ds & 0x01) << 4); // ds2:6, ds1:5, ds0:4
473
474    // Pull resistor: PE=bit9, PS=bit10
475    match pull {
476        PullResistor::None => {} // PE=0, PS=0
477        PullResistor::Up => {
478            val |= (1 << 9) | (1 << 10); // PE=1, PS=1
479        }
480        PullResistor::Down => {
481            val |= 1 << 9; // PE=1, PS=0
482        }
483    }
484
485    // Schmitt trigger: ST=bit3
486    if schmitt_trigger {
487        val |= 1 << 3;
488    }
489
490    // Input enable: IE=bit11
491    if input_enable {
492        val |= 1 << 11;
493    }
494
495    val
496}

500#[cfg(all(test, not(target_arch = "riscv32")))]
501mod tests {

504    // Bit positions in the pad-control register (mirrors build_pad_ctrl).
505    const ST_BIT: u32 = 1 << 3; // Schmitt trigger
506    const DS_MASK: u32 = 0x7 << 4; // drive strength ds[2:0] -> bits [6:4]
507    const PE_BIT: u32 = 1 << 9; // pull enable
508    const PS_BIT: u32 = 1 << 10; // pull select (1 = up, 0 = down)
509    const IE_BIT: u32 = 1 << 11; // input enable
510
511    #[test]
512    fn drive_strength_discriminants_are_dense() {
513        // The enum encodes DS[2:0] directly; values must be 0..=7 contiguous.
514        assert_eq!(DriveStrength::Strongest as u32, 0);
515        assert_eq!(DriveStrength::Strong as u32, 1);
516        assert_eq!(DriveStrength::Medium as u32, 2);
517        assert_eq!(DriveStrength::Weak as u32, 3);
518        assert_eq!(DriveStrength::Weaker as u32, 4);
519        assert_eq!(DriveStrength::VeryWeak as u32, 5);
520        assert_eq!(DriveStrength::Weakest as u32, 6);
521        assert_eq!(DriveStrength::Minimum as u32, 7);
522    }
523
524    #[test]
525    fn drive_strength_lands_in_bits_4_to_6() {
526        // ds[2:0] must be shifted intact into register bits [6:4] (val == ds << 4).
527        for (ds, expect) in [
528            (DriveStrength::Strongest, 0u32),
529            (DriveStrength::Strong, 1 << 4),
530            (DriveStrength::Medium, 2 << 4),
531            (DriveStrength::Minimum, 7 << 4),
532        ] {
533            let val = build_pad_ctrl(ds, PullResistor::None, false, false);
534            assert_eq!(val & DS_MASK, expect, "ds {ds:?} mis-placed");
535            // Drive strength must not bleed into any other field.
536            assert_eq!(val & !DS_MASK, 0);
537        }
538    }
539
540    #[test]
541    fn drive_strength_covers_full_mask() {
542        // The strongest..minimum span must exactly fill the 3-bit DS field.
543        let strongest = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, false, false);
544        let minimum = build_pad_ctrl(DriveStrength::Minimum, PullResistor::None, false, false);
545        assert_eq!(strongest & DS_MASK, 0);
546        assert_eq!(minimum & DS_MASK, DS_MASK);
547    }

549    #[test]
550    fn pull_none_sets_neither_pe_nor_ps() {
551        // No pull => both PE and PS clear.
552        let val = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, false, false);
553        assert_eq!(val & (PE_BIT | PS_BIT), 0);
554    }

556    #[test]

568        assert_eq!(val & PE_BIT, PE_BIT);
569        assert_eq!(val & PS_BIT, 0);

572    #[test]
573    fn schmitt_and_input_enable_are_independent_bits() {
574        // ST (bit3) and IE (bit11) toggle only their own bits.
575        let none = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, false, false);
576        assert_eq!(none & (ST_BIT | IE_BIT), 0);
577
578        let st = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, true, false);
579        assert_eq!(st, ST_BIT);

581        let ie = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, false, true);
582        assert_eq!(ie, IE_BIT);
583
584        let both = build_pad_ctrl(DriveStrength::Strongest, PullResistor::None, true, true);
585        assert_eq!(both, ST_BIT | IE_BIT);

598        // Every legal combination must stay within the defined field mask;
599        // nothing should ever set a reserved bit.
600        let defined = DS_MASK | PE_BIT | PS_BIT | ST_BIT | IE_BIT;
601        for &drive in &[DriveStrength::Strongest, DriveStrength::Medium, DriveStrength::Minimum] {
602            for &pull in &[PullResistor::None, PullResistor::Up, PullResistor::Down] {
603                for st in [false, true] {

606                        assert_eq!(val & !defined, 0, "reserved bit set for {drive:?}/{pull:?}/{st}/{ie}");
607                    }

614// ── Property-based fuzz tests ──────────────────────────────────
615
616#[cfg(all(test, not(target_arch = "riscv32")))]
617mod proptests {
618    use super::*;
619    use proptest::prelude::*;
620
621    const ST_BIT: u32 = 1 << 3;
622    const DS_MASK: u32 = 0x7 << 4;
623    const PE_BIT: u32 = 1 << 9;
624    const PS_BIT: u32 = 1 << 10;
625    const IE_BIT: u32 = 1 << 11;
626    const DEFINED: u32 = DS_MASK | PE_BIT | PS_BIT | ST_BIT | IE_BIT;
627
628    fn drive_from(i: u8) -> DriveStrength {
629        match i % 8 {
630            0 => DriveStrength::Strongest,
631            1 => DriveStrength::Strong,
632            2 => DriveStrength::Medium,
633            3 => DriveStrength::Weak,
634            4 => DriveStrength::Weaker,
635            5 => DriveStrength::VeryWeak,
636            6 => DriveStrength::Weakest,
637            _ => DriveStrength::Minimum,
638        }
639    }
640
641    fn pull_from(i: u8) -> PullResistor {
642        match i % 3 {
643            0 => PullResistor::None,
644            1 => PullResistor::Up,
645            _ => PullResistor::Down,
646        }
647    }
648
649    proptest! {
650        /// Fuzz: any field combination only ever sets defined bits, never reserved ones.
651        #[test]
652        fn never_sets_reserved_bits(d in any::<u8>(), p in any::<u8>(), st: bool, ie: bool) {
653            let val = build_pad_ctrl(drive_from(d), pull_from(p), st, ie);
654            prop_assert_eq!(val & !DEFINED, 0);
655        }
656
657        /// Fuzz: the DS field always equals the drive enum value shifted left by 4.
658        #[test]
659        fn ds_field_equals_discriminant_shifted(d in any::<u8>(), p in any::<u8>(), st: bool, ie: bool) {
660            let drive = drive_from(d);
661            let val = build_pad_ctrl(drive, pull_from(p), st, ie);
662            prop_assert_eq!(val & DS_MASK, (drive as u32) << 4);
663        }
664
665        /// Fuzz: ST and IE bits track their boolean inputs exactly, regardless of other fields.
666        #[test]
667        fn st_ie_track_inputs(d in any::<u8>(), p in any::<u8>(), st: bool, ie: bool) {
668            let val = build_pad_ctrl(drive_from(d), pull_from(p), st, ie);
669            prop_assert_eq!(val & ST_BIT != 0, st);
670            prop_assert_eq!(val & IE_BIT != 0, ie);
671        }
672
673        /// Fuzz: PE is set iff a pull is requested; PS is set iff that pull is up.
674        #[test]
675        fn pull_encoding_is_consistent(d in any::<u8>(), p in any::<u8>(), st: bool, ie: bool) {
676            let pull = pull_from(p);
677            let val = build_pad_ctrl(drive_from(d), pull, st, ie);
678            let pe = val & PE_BIT != 0;
679            let ps = val & PS_BIT != 0;
680            match pull {
681                PullResistor::None => { prop_assert!(!pe); prop_assert!(!ps); }
682                PullResistor::Up => { prop_assert!(pe); prop_assert!(ps); }
683                PullResistor::Down => { prop_assert!(pe); prop_assert!(!ps); }
684            }
685        }
686    }
