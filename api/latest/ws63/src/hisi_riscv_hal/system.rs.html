5/// System control handle.
6///
7/// Holds the SYS_CTL0, GLB_CTL_M, and CLDO_CRG peripherals for clock and
8/// reset configuration.
9pub struct System<'d> {
10    /// SYS_CTL0 peripheral, controlling core system clock and reset settings.
11    pub sys_ctl0: SysCtl0<'d>,
12    /// GLB_CTL_M peripheral, controlling global clock/reset and chip reset.
13    pub glb_ctl_m: GlbCtlM<'d>,
14    /// CLDO_CRG peripheral (clock and reset generator) for peripheral gates.
15    pub cldo_crg: CldoCrg<'d>,
16}

18impl<'d> System<'d> {
19    /// Create a `System` handle from the SYS_CTL0, GLB_CTL_M, and CLDO_CRG peripherals.

21        Self { sys_ctl0, glb_ctl_m, cldo_crg }
22    }

25/// Reason for the last reset.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum ResetReason {
28    /// Power-on reset (cold boot).
29    PowerOn,
30    /// External reset pin.
31    ExternalPin,
32    /// Watchdog timer reset.
33    Watchdog,
34    /// Software reset.
35    Software,
36    /// Brown-out reset.
37    BrownOut,
38    /// Unknown reset reason.
39    Unknown,
40}
41
42// Reset registers, from fbb_ws63 drivers/chips/ws63/porting/reboot/reboot_porting.c:
43//   chip-reset trigger:  GLB_CTL_M (0x4000_2000) + 0x110, set bit 2 (HAL_CHIP_RESET_REG)
44//   reset-reason record: GLB_CTL   (0x4000_0000) + 0xA0    (SYS_RST_RECORD_0)
45//   reason-clear:        GLB_CTL   (0x4000_0000) + 0xA4    (SYS_DIAG_CLR_1)
46// History bits in SYS_RST_RECORD_0.
47const SYS_WDT_RST_HIS: u32 = 0x1;
48const SYS_SOFT_RST_HIS: u32 = 0x2;
49const POR_RST_FILTER_HIS: u32 = 0x8;
50
51impl System<'_> {
52    /// Read (and clear) the last reset reason from `SYS_RST_RECORD_0`.
53    ///
54    /// Decodes the WS63 reset-history record (`reboot_port_get_rst_reason`):
55    /// watchdog takes precedence over software over power-on. The matched bit is
56    /// cleared via `SYS_DIAG_CLR_1` so the next boot reports its own cause.
57    /// Reasons this SoC's record does not distinguish (`ExternalPin`, `BrownOut`)
58    /// are never returned here. An empty record reads back as [`ResetReason::Unknown`].
59    pub fn reset_reason(&self) -> ResetReason {
60        let r = unsafe { &*SysCtl0::ptr() };
61        let record = r.sys_rst_record_0().read();
62        let val = record.bits();
63        let (reason, clr) = if val & SYS_WDT_RST_HIS != 0 {
64            (ResetReason::Watchdog, SYS_WDT_RST_HIS)
65        } else if val & SYS_SOFT_RST_HIS != 0 {
66            (ResetReason::Software, SYS_SOFT_RST_HIS)
67        } else if val & POR_RST_FILTER_HIS != 0 {
68            (ResetReason::PowerOn, POR_RST_FILTER_HIS)
69        } else {
70            (ResetReason::Unknown, 0)
71        };
72        if clr != 0 {
73            r.sys_diag_clr_1().write(|w| unsafe { w.sys_diag_clr().bits(clr) });
74        }
75        reason
76    }
77
78    /// Trigger a full software reset of the chip and never return.
79    ///
80    /// Sets the chip-reset enable bit (bit 2) of `GLB_CTL_M + 0x110`, the same
81    /// register `reboot_port_reboot_chip` uses. The CPU is reset before the
82    /// following spin loop completes.
83    #[instability::unstable]
84    pub fn software_reset(&self) -> ! {
85        let r = unsafe { &*GlbCtlM::ptr() };
86        r.chip_reset().modify(|_, w| w.chip_reset_en().set_bit());
87        loop {
88            core::hint::spin_loop();
89        }
90    }
91
92    /// Trigger a software reset and never return.
93    ///
94    /// WS63's porting layer exposes only a whole-chip reset, so this is an alias
95    /// of [`software_reset`](Self::software_reset).
96    #[instability::unstable]
97    pub fn software_reset_cpu(&self) -> ! {
98        self.software_reset()
99    }
100}
