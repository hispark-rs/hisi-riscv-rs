1//! WS63 link-time boot header (`boot-header` feature).
2//!
3//! flashboot loads the WS63 application image from flash `0x230000` and jumps
4//! **unconditionally** to `0x230300` (= `app_partition + 0x300`). A bootable
5//! image is therefore a fixed `0x300`-byte HiSilicon header at `0x230000`
6//! immediately followed by the app code at `0x230300`.
7//!
8//! Normally that header is added *post-build* by `hisi-fwpkg image`. With the
9//! `boot-header` feature enabled, this module bakes a structurally-correct
10//! header **into the ELF at link time** (section `.boot_header`, placed by the
11//! linker at flash `0x230000`), so — after the post-link `hisi-fwpkg patch-hash`
12//! fills the body SHA-256 (see "Route 2" below) — `probe-rs download <elf>` /
13//! `probe-rs run <elf>` flash a directly bootable image with no separate
14//! `hisi-fwpkg image` packaging step. (`cargo flash` is NOT usable here: there
15//! is no runner slot to insert the mandatory post-link `patch-hash`.)
16//!
17//! ## "Route 2": link-time length, post-link hash
18//!
19//! On-silicon measurement shows flashboot checks the **body hash** even with
20//! secure-verify disabled (zeroing only `code_area_hash` of a known-good image
21//! stops it booting). So `code_area_hash` MUST be the real SHA-256 of the body
22//! and `code_area_len` the real body length (the hash is computed over the
23//! first `code_area_len` body bytes). A linker cannot compute SHA-256, so the
24//! hash is a **post-link** patch (`hisi-fwpkg patch-hash`). The length, however,
25//! *can* be a linker symbol — so this module bakes everything except the two
26//! body-dependent fields at link time:
27//!
28//! * `code_area_len` (code-info `+0x24` = `0x124`) and `code_uncompress_len`
29//!   (code-info `+0x80` = `0x180`) are emitted by the **linker** as
30//!   `__hisi_app_body_len` (`__hisi_body_end - 0x230300`), via `LONG(...)`
31//!   directives in `boot-header.x` — a Rust `static` initializer cannot
32//!   const-read a relocation. To make room for those linker words, the header
33//!   body is split here into three byte-array statics (`part0`/`part1`/`part2`)
34//!   placed in sections `.boot_header.part0/1/2`; `boot-header.x` stitches them
35//!   back together with the two `LONG` words interleaved at the right offsets.
36//! * `code_area_hash` (code-info `+0x28` = `0x128`, 32 bytes): left **zero** in
37//!   `part1` — patched post-link by `hisi-fwpkg patch-hash`.
38//! * `text_segment_size` (code-info `+0x84`): left **zero** (informational
39//!   only; the vendor default `0x10000` is not needed to boot).
40//!
41//! The three parts and the two interleaved `LONG` words reassemble to the exact
42//! same `0x300`-byte layout produced field-for-field by `hisi-fwpkg`'s
43//! `build_image_header` (`hisi-fwpkg/crates/hisi-fwpkg/src/image.rs`).
44
45/// `APPBOOT_KEY_AREA_IMAGE_ID` — magic of the key area (offset 0x000).
46const APP_KEY_AREA_IMAGE_ID: u32 = 0x4B0F_2D1E;
47/// `APPBOOT_CODE_INFO_IMAGE_ID` — magic of the code-info area (offset 0x100).
48const APP_CODE_INFO_IMAGE_ID: u32 = 0x4B0F_2D2D;
49
50/// Length of the key area, `KEY_AREA_STRUCTURE_LENGTH` (ECC/SM2 build).
51const KEY_AREA_LEN: usize = 0x100;
52/// Length of the code-info area, `CODE_INFO_STRUCTURE_LENGTH` (ECC/SM2 build).
53const CODE_INFO_LEN: usize = 0x200;
54/// Total fixed image header length, `APP_IMAGE_HEADER_LEN`.
55const IMAGE_HEADER_LEN: usize = KEY_AREA_LEN + CODE_INFO_LEN; // 0x300
56
57/// `structure_version`, vendor uses `0x00010000`.
58const STRUCTURE_VERSION: u32 = 0x0001_0000;
59/// ECC-bp256 signature length in bytes (`BOOT_SIG_LEN`).
60const SIG_LEN: u32 = 0x40;
61/// `KeyAlg` for ECC256 / brainpoolP256r1 (`0x2A13C812`).
62const KEY_ALG_ECC256: u32 = 0x2A13_C812;
63/// `ecc_curve_type` for brainpoolP256r1 (`0x2A13C812`).
64const ECC_CURVE_BP256R1: u32 = 0x2A13_C812;
65/// ECC-bp256 public key length in bytes (`BOOT_PUBLIC_KEY_LEN`).
66const PUB_KEY_LEN: u32 = 0x40;
67/// `FLASH_NO_ENCRY_FLAG` — `code_enc_flag` value meaning "**not** encrypted".
68///
69/// Counterintuitively non-zero: flashboot's `ws63_flash_encrypt_config()` does
70/// `if (code_enc_flag == FLASH_NO_ENCRY_FLAG) return;`, so a zero value would
71/// make it try to configure on-the-fly flash decryption and fail to boot a
72/// plaintext image.
73const FLASH_NO_ENCRY_FLAG: u32 = 0x3C78_96E1;
74
75// ---- ImageOptions defaults (matching hisi-fwpkg's ImageOptions::default) ----
76const KEY_OWNER_ID: u32 = 1;
77const KEY_ID: u32 = 1;
78const KEY_VERSION_EXT: u32 = 0;
79const KEY_VERSION_MASK: u32 = 0;
80const VERSION_EXT: u32 = 0;
81const VERSION_MASK: u32 = 0;
82const MSID: u32 = 0;
83const MSID_MASK: u32 = 0;
84/// `text_segment_size` — informational only; not needed to boot.
85const TEXT_SEGMENT_SIZE: u32 = 0;
86
87/// Write a little-endian `u32` into `buf` at byte offset `off` (const-fn).
88const fn put_u32(mut buf: [u8; IMAGE_HEADER_LEN], off: usize, v: u32) -> [u8; IMAGE_HEADER_LEN] {
89    let b = v.to_le_bytes();
90    buf[off] = b[0];
91    buf[off + 1] = b[1];
92    buf[off + 2] = b[2];
93    buf[off + 3] = b[3];
94    buf
95}

97/// Build the full `0x300`-byte WS63 app image header, with the body-dependent
98/// length fields (`code_area_len` @0x124, `code_uncompress_len` @0x180) and the
99/// `code_area_hash` (@0x128) left **zero**. The two lengths are overwritten by
100/// the linker (see `boot-header.x`); the hash is patched post-link. Mirrors
101/// `hisi-fwpkg::image::build_image_header` field-for-field. Used only to slice
102/// out the three constant parts below.
103// `CI + 0x00` is the first entry of a code-info offset table; the explicit `+ 0x00`
104// keeps every field's offset visually aligned with its siblings (`+ 0x04`, `+ 0x08`,
105// …), so allow the identity_op rather than break the table's readability.
106#[allow(clippy::identity_op)]
107const fn build_header() -> [u8; IMAGE_HEADER_LEN] {
108    let mut h = [0u8; IMAGE_HEADER_LEN];
109
110    // ---- Key area (image_key_area_t), offset 0x000, length 0x100 ----
111    h = put_u32(h, 0x00, APP_KEY_AREA_IMAGE_ID); // image_id
112    h = put_u32(h, 0x04, STRUCTURE_VERSION); // structure_version
113    h = put_u32(h, 0x08, KEY_AREA_LEN as u32); // structure_length (0x100)
114    h = put_u32(h, 0x0C, SIG_LEN); // signature_length (0x40)
115    h = put_u32(h, 0x10, KEY_OWNER_ID); // key_owner_id
116    h = put_u32(h, 0x14, KEY_ID); // key_id
117    h = put_u32(h, 0x18, KEY_ALG_ECC256); // key_alg
118    h = put_u32(h, 0x1C, ECC_CURVE_BP256R1); // ecc_curve_type
119    h = put_u32(h, 0x20, PUB_KEY_LEN); // key_length (0x40)
120    h = put_u32(h, 0x24, KEY_VERSION_EXT); // key_version_ext
121    h = put_u32(h, 0x28, KEY_VERSION_MASK); // mask_key_version_ext
122    h = put_u32(h, 0x2C, MSID); // msid_ext
123    h = put_u32(h, 0x30, MSID_MASK); // mask_msid_ext
124    h = put_u32(h, 0x34, 0); // maintenance_mode (disabled)
125    h = put_u32(h, 0x48, 0); // code_info_addr (0 = immediately follows)
126    // die_id[16] @ 0x38, ext_public_key_area + sig_key_area — dummy zero
127
128    // ---- Code-info area (image_code_info_t), offset 0x100, length 0x200 ----
129    const CI: usize = KEY_AREA_LEN;
130    h = put_u32(h, CI + 0x00, APP_CODE_INFO_IMAGE_ID); // image_id
131    h = put_u32(h, CI + 0x04, STRUCTURE_VERSION); // structure_version
132    h = put_u32(h, CI + 0x08, CODE_INFO_LEN as u32); // structure_length (0x200)
133    h = put_u32(h, CI + 0x0C, SIG_LEN); // signature_length (0x40)
134    h = put_u32(h, CI + 0x10, VERSION_EXT); // version_ext
135    h = put_u32(h, CI + 0x14, VERSION_MASK); // mask_version_ext
136    h = put_u32(h, CI + 0x18, MSID); // msid_ext
137    h = put_u32(h, CI + 0x1C, MSID_MASK); // mask_msid_ext
138    h = put_u32(h, CI + 0x20, 0); // code_area_addr (0 = immediately follows)
139    // code_area_len @ CI+0x24 (=0x124) — left zero, emitted by the linker
140    // code_area_hash[32] @ CI+0x28 (=0x128) — left zero, patched post-link
141    h = put_u32(h, CI + 0x48, FLASH_NO_ENCRY_FLAG); // code_enc_flag
142    // protection_key_l1/l2 + iv @ CI+0x4C.. — zero (encryption disabled)
143    h = put_u32(h, CI + 0x7C, 0); // code_compress_flag (0 = not compressed)
144    // code_uncompress_len @ CI+0x80 (=0x180) — left zero, emitted by the linker
145    h = put_u32(h, CI + 0x84, TEXT_SEGMENT_SIZE); // text_segment_size
146    // sig_code_info + ext — dummy zero
147
148    h
149}

151/// The full header laid out once at compile time; the three `BOOT_HEADER_PARTN`
152/// statics below are constant slices of it. The two length words at `0x124` and
153/// `0x180` are *holes* (zero here) filled by the linker; `0x128..0x148` (hash)
154/// is also zero, filled post-link.
155const HEADER: [u8; IMAGE_HEADER_LEN] = build_header();
156
157/// Linker-word offsets that split the header. `code_area_len` lives at
158/// `0x124`, `code_uncompress_len` at `0x180`; each is a 4-byte hole the linker
159/// fills, so the constant parts are the spans *around* them.
160const LEN_OFF: usize = 0x124; // code_area_len
161const UNCOMPLEN_OFF: usize = 0x180; // code_uncompress_len
162
163/// `[0x000 .. 0x124]` — everything up to (not including) `code_area_len`.
164const PART0_LEN: usize = LEN_OFF; // 0x124
165/// `[0x128 .. 0x180]` — between `code_area_len` and `code_uncompress_len`
166/// (includes the zero `code_area_hash` @0x128).
167const PART1_LEN: usize = UNCOMPLEN_OFF - (LEN_OFF + 4); // 0x58
168/// `[0x184 .. 0x300]` — everything after `code_uncompress_len`.
169const PART2_LEN: usize = IMAGE_HEADER_LEN - (UNCOMPLEN_OFF + 4); // 0x17C
170
171// The three parts plus the two 4-byte linker words must reconstruct 0x300.
172const _: () = assert!(PART0_LEN + 4 + PART1_LEN + 4 + PART2_LEN == IMAGE_HEADER_LEN);
173
174/// Copy a constant `N`-byte slice of `HEADER` starting at `start` (const-fn;
175/// `<[u8]>::try_into`/slicing isn't const-stable for arrays here).
176const fn slice<const N: usize>(start: usize) -> [u8; N] {
177    let mut out = [0u8; N];
178    let mut i = 0;
179    while i < N {
180        out[i] = HEADER[start + i];
181        i += 1;
182    }
183    out
184}

186/// `.boot_header` part 0: header bytes `[0x000 .. 0x124]`. Immediately followed
187/// in the linked output by the linker word `code_area_len` (see `boot-header.x`).
188#[used]
189#[unsafe(no_mangle)]
190#[unsafe(link_section = ".boot_header.part0")]
191pub static BOOT_HEADER_PART0: [u8; PART0_LEN] = slice::<PART0_LEN>(0);
192
193/// `.boot_header` part 1: header bytes `[0x128 .. 0x180]` (the zero
194/// `code_area_hash` lives here, patched post-link). Sits between the two linker
195/// length words.
196#[used]
197#[unsafe(no_mangle)]
198#[unsafe(link_section = ".boot_header.part1")]
199pub static BOOT_HEADER_PART1: [u8; PART1_LEN] = slice::<PART1_LEN>(LEN_OFF + 4);
200
201/// `.boot_header` part 2: header bytes `[0x184 .. 0x300]` — after the linker
202/// word `code_uncompress_len`.
203#[used]
204#[unsafe(no_mangle)]
205#[unsafe(link_section = ".boot_header.part2")]
206pub static BOOT_HEADER_PART2: [u8; PART2_LEN] = slice::<PART2_LEN>(UNCOMPLEN_OFF + 4);
