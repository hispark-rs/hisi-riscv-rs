6/// Interrupt fired by the TIMER channel the embassy-time driver uses for alarms.
7///
8/// WS63 routes TIMER channel 0 to `TIMER_INT0` (IRQ 26, a standard `mie`-bit
9/// local interrupt). The `embassy` time-driver and the app's trap handler reach
10/// this through [`crate::soc::chip::ALARM_INTERRUPT`] so the alarm wiring is
11/// chip-neutral (BS2X uses a different IRQ — see `soc/bs21.rs`).
12pub const ALARM_INTERRUPT: Interrupt = Interrupt::TIMER_INT0;

14/// System clock frequency (240 MHz).
15///
16/// # Clock initialization
17///
18/// **Boot ROM configures the CPU PLL to 240 MHz before loading the application.**
19///
20/// The fbb_ws63 flashboot bootloader sequence:
21/// 1. `boot_clock_adapt()` — detects TCXO (24/40 MHz) via HW_CTL bit[0],
22///    configures UART baud base and WDT tick counter to match TCXO freq
23/// 2. `switch_flash_clock_to_pll()` — sets CLDO_CRG_CLK_SEL bit[18] to switch
24///    the flash/SFC controller clock from TCXO to PLL
25/// 3. Jumps to application entry point
26///
27/// **What the application must do:** switch peripheral clocks (UART, SPI, I2C)
28/// from TCXO to PLL source. Use `clock_init::init_clocks()` for this.
29///
30/// If the PLL is NOT locked (unlikely, indicates hardware issue), the CPU
31/// runs from the TCXO at 24 or 40 MHz. All timing calculations will be wrong.
32/// Call `clock_init::probe_clocks()` to verify.
33pub const SYSTEM_CLOCK_HZ: u32 = 240_000_000;
34
35/// TCXO crystal frequency (24 MHz on standard WS63 modules).
36///
37/// This is the **counting clock of the Timer and Watchdog peripherals and the
38/// TCXO time base** — NOT the 240 MHz CPU/PLL clock ([`SYSTEM_CLOCK_HZ`]).
39///
40/// The vendor SDK programs both the timer and the WDT to the TCXO crystal in
41/// `clock_init` — `timer_porting_clock_value_set(REQ_24M)` and
42/// `watchdog_port_set_clock(REQ_24M)` (fbb_ws63 `clock_init.c`), where
43/// `REQ_24M = 24_000_000`. The 32 MHz `CONFIG_TIMER_CLOCK_VALUE` Kconfig default
44/// and the 24 MHz "FPGA" WDT note are placeholders the silicon path overrides to
45/// this value. ch2 of ws63-guide lists Timer = "晶体分频" (crystal-derived).
46///
47/// 40 MHz-crystal boards run the timer/WDT at 40 MHz instead; override there.
48pub const TCXO_HZ: u32 = 24_000_000;
49
50/// Timer / Watchdog counting clock (= [`TCXO_HZ`], 24 MHz).
51///
52/// Use this — **not** [`SYSTEM_CLOCK_HZ`] — to convert microseconds/milliseconds
53/// to Timer or WDT counter ticks. The timer is clocked by the crystal, so at
54/// 24 MHz one microsecond is 24 ticks (the vendor `timer_porting_us_2_cycle`
55/// computes `us * (clock / 1_000_000)` = `us * 24`).
56pub const TIMER_CLOCK_HZ: u32 = TCXO_HZ;
57
58/// UART baud-base clock (160 MHz, PLL-derived).
59///
60/// After boot, `clock_init` switches the UART clock from TCXO to PLL and sets the
61/// baud base to `UART_PLL_CLOCK = 160_000_000` (fbb_ws63 `clock_init.c`); ch2 of
62/// ws63-guide also lists UART = 160 MHz. The baud divisor is `clock / (16 * baud)`,
63/// so this — not the 240 MHz CPU clock — is the divisor base.
64///
65/// **On-silicon caveat (2026-06-14, see hisi-riscv-rs#10/#15):** examples that do
66/// NOT run the (XIP-unsafe) full `clock_init` inherit flashboot's UART clock, which
67/// is NOT this PLL base — so their baud is wrong on real hardware. The boot UART
68/// clock is the raw TCXO crystal (24 or 40 MHz), selected per board; use
69/// [`crate::uart::UartClock::Boot`] for those examples.
70pub const UART_CLOCK_HZ: u32 = 160_000_000;

72/// Boot (pre-`clock_init`) UART baud base for a 24 MHz-TCXO board.
73pub const UART_BOOT_CLOCK_24M_HZ: u32 = 24_000_000;
74/// Boot (pre-`clock_init`) UART baud base for a 40 MHz-TCXO board.
75pub const UART_BOOT_CLOCK_40M_HZ: u32 = 40_000_000;

77/// Live flashboot UART baud base (the raw TCXO crystal, 24 or 40 MHz) for use
78/// BEFORE `clock_init` runs.
79///
80/// flashboot's `boot_clock_adapt()` sets the console UART baud base to the raw
81/// TCXO — NOT the 160 MHz PLL — keyed on the `HW_CTL` TCXO-detect strap (fbb_ws63
82/// `soc_porting.c` `get_tcxo_freq`/`boot_clock_adapt`). This mirrors that read.
83/// **Confirmed on silicon (2026-06-14):** a 40 MHz board reads `HW_CTL = 0x2`
84/// (bit 0 clear), and at 115200 8N1 that yields the measured flashboot divider
85/// `div = 21, div_fra = 44` — the divider the HAL must reproduce for examples
86/// that skip `clock_init` (hisi-riscv-rs#10/#15).
87///
88/// [`crate::uart::UartClock::Boot`] uses this value so the divider is computed
89/// against the real base.
90pub fn uart_boot_clock_hz() -> u32 {
91    // HW_CTL TCXO-frequency-detect strap; bit 0: 1 = 24 MHz TCXO, 0 = 40 MHz
92    // (CLK24M_TCXO = 1, CLK40M_TCXO = 0 in the vendor `soc_porting.h`).
93    let sys_ctl0 = unsafe { &*crate::soc::pac::SysCtl0::ptr() };
94    if sys_ctl0.hw_ctl().read().refclk_freq_status().bit_is_set() {
95        UART_BOOT_CLOCK_24M_HZ
96    } else {
97        UART_BOOT_CLOCK_40M_HZ
98    }
99}

101/// SPI controller input clock / SSI_CLK (160 MHz, PLL-derived).
102///
103/// SCK = SSI_CLK / SCKDV. WS63 derives the SPI clock from the 480 MHz FNPLL tap via
104/// a two-stage divider: a CLDO_CRG divider sets SSI_CLK (= this value, 480/3), then
105/// the in-controller SCKDV divides to SCK. `spi.rs::configure_spi_source_clock`
106/// programs the CRG divider (`DIV_CTL3`) + switches the source to PLL on init —
107/// mirroring the vendor `spi_porting_clock_init` — so this SSI_CLK is established,
108/// not just assumed. ch2 of ws63-guide lists SPI = 160 MHz; the QEMU model matches.
109pub const SPI_CLOCK_HZ: u32 = 160_000_000;
110
111/// I2C peripheral clock (= [`TCXO_HZ`], 24 MHz).
112///
113/// Unlike UART/SPI, the I2C clock is **not** switched to the PLL: `clock_init` sets
114/// it to the TCXO crystal via `i2c_port_set_clock_value(REQ_24M)` (fbb_ws63
115/// `clock_init.c`), and the SCL divisor math (`SCL = clock / (2*(scl_h+scl_l)*2)`)
116/// is computed against this value. (ch2's nominal "I2C = 80 MHz" is the bus-capability
117/// figure, not the divisor base the SDK uses.)
118pub const I2C_CLOCK_HZ: u32 = TCXO_HZ;
