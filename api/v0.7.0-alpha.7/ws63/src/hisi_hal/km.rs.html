1//! KM (Key Management) driver for WS63.
2//!
3//! The KM peripheral manages cryptographic keys, providing:
4//! - KLAD (Key Loading and Distribution)
5//! - Keyslot locking mechanisms
6//! - RKP (Root Key Provisioning)
7
8use crate::peripherals::Km;
9
10/// Number of available keyslots.
11pub const KEYSLOT_COUNT: u8 = 8;
12
13/// KM driver.
14pub struct KmDriver<'d> {
15    _km: Km<'d>,
16}
17
18impl<'d> KmDriver<'d> {
19    /// Create a new KM driver.
20    pub fn new(km: Km<'d>) -> Self {
21        Self { _km: km }
22    }
23
24    fn regs(&self) -> &'static crate::soc::pac::km::RegisterBlock {
25        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid
26        unsafe { &*Km::ptr() }
27    }
28
29    /// Enable the KM peripheral.
30    pub fn enable(&mut self) {
31        unsafe {
32            self.regs().kl_com_ctrl().write(|w| w.bits(0x01));
33        }
34    }
35
36    /// Check if a keyslot is locked (read-only after lock).
37    ///
38    /// Implements the vendor SDK `inner_kslot_chn_is_locked` sequence:
39    /// 1. Write `KC_RD_SLOT_NUM` to select the keyslot and type (mcipher)
40    /// 2. Read `KC_RD_LOCK_STATUS` to get the lock owner
41    /// 3. Keyslot is locked if `rd_lock_status != 0` (non-zero = some CPU owns the lock)
42    pub fn is_keyslot_locked(&self, slot: u8) -> bool {
43        assert!(slot < KEYSLOT_COUNT, "keyslot {} out of range (max {})", slot, KEYSLOT_COUNT);
44        // Select keyslot to query: mcipher type (bit 15 = 0), slot number in bits [9:0]
45        unsafe {
46            self.regs().kc_rd_slot_num().write(|w| {
47                w.slot_num_cfg().bits(slot as u16);
48                w.slot_cfg_type().clear_bit()
49            });
50        }
51        // Read lock status: non-zero rd_lock_status field means locked
52        self.regs().kc_rd_lock_status().read().rd_lock_status().bits() != 0
53    }
54
55    /// Lock a keyslot (prevents further writes).
56    ///
57    /// Writes to `KC_REECPU_LOCK_CMD` — the REE CPU keyslot lock command register
58    /// (main application processor = REE). Uses mcipher keyslot type.
59    ///
60    /// After locking, the keyslot cannot be modified until the next system reset.
61    pub fn lock_keyslot(&mut self, slot: u8) {
62        assert!(slot < KEYSLOT_COUNT, "keyslot {} out of range (max {})", slot, KEYSLOT_COUNT);
63        // Vendor SDK: write key_slot_num (bits [9:0]) and lock_cmd=1 (bit 20),
64        // flush_hmac_kslot_ind=0 (mcipher), tscipher_ind=0
65        unsafe {
66            self.regs().kc_reecpu_lock_cmd().write(|w| {
67                w.key_slot_num().bits(slot as u16);
68                w.lock_cmd().set_bit()
69            });
70        }
71    }
72
73    /// Get the number of available keyslots.
74    pub fn keyslot_count(&self) -> u8 {
75        KEYSLOT_COUNT
76    }
77}

81#[cfg(all(test, not(target_arch = "riscv32")))]
82mod tests {
83    #[test]
84    fn test_keyslot_lock_bit_position() {
85        // lock_keyslot writes key_slot_num in bits [9:0], lock_cmd=1 at bit 20
86        let slot: u16 = 3;
87        let lock_cmd_bit: u32 = 1 << 20;
88        let val: u32 = (slot as u32) | lock_cmd_bit;
89        assert_eq!(val & 0x3FF, 3); // key_slot_num = 3 in bits [9:0]
90        assert_eq!((val >> 20) & 1, 1); // lock_cmd = 1 at bit 20
91    }
92
93    #[test]
94    fn test_keyslot_lock_status_is_enum_not_bitmask() {
95        // rd_lock_status is a 3-bit enum (bits [2:0]):
96        // 0=unlocked, 1=REE, 2=TEE, 4=PCPU, 6=AIDSP
97        // is_keyslot_locked should check rd_lock_status != 0, NOT (status & (1 << slot))
98        let unlocked: u8 = 0;
99        let ree_locked: u8 = 1;
100        let tee_locked: u8 = 2;
101        assert_eq!(unlocked != 0, false); // unlocked → not locked
102        assert_eq!(ree_locked != 0, true); // REE locked → locked
103        assert_eq!(tee_locked != 0, true); // TEE locked → locked
104    }
105
106    #[test]
107    fn test_keyslot_count() {
108        assert_eq!(super::KEYSLOT_COUNT, 8);
109    }
110
111    #[test]
112    fn test_all_keyslots_lockable() {
113        // All 8 keyslots (0-7) must produce valid lock values
114        for slot in 0..8u16 {
115            let val: u32 = (slot as u32) | (1 << 20);
116            assert_eq!(val & 0x3FF, slot as u32); // key_slot_num in bits [9:0]
117            assert_eq!((val >> 20) & 1, 1); // lock_cmd at bit 20
118            assert!(slot < super::KEYSLOT_COUNT as u16); // within valid range
119        }
120    }
121}

123// ── Property-based fuzz tests ──────────────────────────────────
124
125#[cfg(all(test, not(target_arch = "riscv32")))]
126mod proptests {
127    use super::KEYSLOT_COUNT;
128    use proptest::prelude::*;
129
130    /// Re-derives the `KC_REECPU_LOCK_CMD` word that `lock_keyslot` writes:
131    /// `key_slot_num` in bits [9:0], `lock_cmd` = 1 at bit 20, all other
132    /// fields (flush_hmac_kslot_ind, tscipher_ind, …) left zero.
133    fn lock_cmd_word(slot: u8) -> u32 {
134        (slot as u32) | (1u32 << 20)
135    }
136
137    /// Re-derives the `KC_RD_SLOT_NUM` word that `is_keyslot_locked` writes to
138    /// select a keyslot: `slot_num_cfg` (a u16) in bits [9:0], `slot_cfg_type`
139    /// = 0 (mcipher) at bit 15.
140    fn rd_slot_select_word(slot: u8) -> u16 {
141        slot as u16 // slot_num_cfg, slot_cfg_type cleared (bit 15 = 0)
142    }
143
144    proptest! {
145        /// Fuzz: for every valid keyslot the lock command never panics and lands
146        /// `key_slot_num` exactly in bits [9:0] (round-trip via mask).
147        #[test]
148        fn lock_cmd_slot_round_trips(slot in 0u8..KEYSLOT_COUNT) {
149            let w = lock_cmd_word(slot);
150            prop_assert_eq!(w & 0x3FF, slot as u32);
151        }
152
153        /// Fuzz: the lock command always asserts `lock_cmd` at bit 20 and sets
154        /// NO stray bits outside the two documented fields ([9:0] | bit 20).
155        #[test]
156        fn lock_cmd_has_no_stray_bits(slot in 0u8..KEYSLOT_COUNT) {
157            let w = lock_cmd_word(slot);
158            prop_assert_eq!((w >> 20) & 1, 1); // lock_cmd asserted at bit 20
159            // The slot field never collides with lock_cmd (bit 20) or type (bit 15).
160            let documented = 0x3FFu32 | (1u32 << 20);
161            prop_assert_eq!(w & !documented, 0, "stray bits in lock word {:#x}", w);
162        }
163
164        /// Fuzz: the slot index always fits the [9:0] field, never reaching the
165        /// `slot_cfg_type` selector at bit 15 — so a valid slot can never be
166        /// mis-decoded as the tscipher keyslot type.
167        #[test]
168        fn slot_index_never_reaches_type_bit(slot in 0u8..KEYSLOT_COUNT) {
169            let w = lock_cmd_word(slot);
170            prop_assert_eq!((w >> 15) & 1, 0); // bit 15 (type) stays clear
171            prop_assert!(slot as u32 <= 0x3FF);
172        }
173
174        /// Fuzz: the read-slot-select word puts the slot in bits [9:0] with the
175        /// type selector (bit 15) cleared, for every valid slot.
176        #[test]
177        fn rd_slot_select_encodes_mcipher(slot in 0u8..KEYSLOT_COUNT) {
178            let w = rd_slot_select_word(slot);
179            prop_assert_eq!(w & 0x3FF, slot as u16);
180            prop_assert_eq!((w >> 15) & 1, 0); // slot_cfg_type = 0 → mcipher
181        }
182
183        /// Fuzz: the lock-status decode is "non-zero owner ⇒ locked", NOT a
184        /// per-slot bitmask. For ANY 3-bit owner value (0=unlocked, else a CPU
185        /// owns it) the `status != 0` rule equals "owner present", which a
186        /// buggy `status & (1 << slot)` test would get wrong for owners like
187        /// TEE(2)/AIDSP(6) on slots whose bit isn't set.
188        #[test]
189        fn lock_status_is_owner_nonzero_not_bitmask(status in any::<u8>(), slot in 0u8..KEYSLOT_COUNT) {
190            let locked = status != 0;
191            prop_assert_eq!(locked, status > 0);
192            // The buggy bitmask interpretation must NOT be relied upon: it would
193            // call a real lock "unlocked" whenever the owner enum's bit for this
194            // slot happens to be clear. Demonstrate they genuinely disagree.
195            let buggy_bitmask = (status & (1u8 << (slot & 7))) != 0;
196            if status != 0 && !buggy_bitmask {
197                prop_assert!(locked && !buggy_bitmask,
198                    "owner {} on slot {} is locked but bitmask says unlocked", status, slot);
199            }
200        }
201
202        /// Fuzz: every keyslot the driver accepts is strictly below the count,
203        /// matching the `assert!(slot < KEYSLOT_COUNT)` guard in lock/query.
204        #[test]
205        fn valid_slots_are_below_count(slot in 0u8..KEYSLOT_COUNT) {
206            prop_assert!(slot < KEYSLOT_COUNT);
207        }
208    }
209}
