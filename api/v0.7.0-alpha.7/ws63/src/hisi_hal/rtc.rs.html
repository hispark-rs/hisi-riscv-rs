1//! Real-Time Clock (RTC) driver for WS63.
2//!
3//! The WS63 RTC is a 48-bit free-running counter that can operate in
4//! free-running or periodic mode. It can generate interrupts when the
5//! counter matches a programmed load value.
6//!
7//! # Clock source
8//!
9//! The RTC runs from a 32.768 kHz clock.
10//!
11//! # Preconditions (board / analog)
12//!
13//! The 32.768 kHz domain comes from an **external crystal that many boards do not
14//! populate**. If the crystal is absent its clock domain never comes up, the
15//! counter never advances, and touching the RTC registers can stall the bus / drop
16//! the debug link. There is no software guard for a missing crystal — it is a board
17//! provisioning fact — so the on-silicon RTC test is gated behind the opt-in
18//! `hil-rtc` feature (OFF by default). Treat a populated 32.768 kHz crystal as a
19//! hard precondition before constructing an `RtcDriver`; with it absent, prefer the
20//! `timer` (TCXO) driver instead.
21
22use crate::peripherals::Rtc;
23
24/// RTC operating mode.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26pub enum RtcMode {
27    /// Free-running mode: counter runs continuously.
28    FreeRunning,
29    /// Periodic mode: counter resets when it reaches the load value.
30    Periodic,
31}
32
33/// Real-Time Clock driver.
34pub struct RtcDriver<'d> {
35    _rtc: Rtc<'d>,
36}
37
38/// RTC clock frequency (32.768 kHz).
39pub const RTC_CLOCK_HZ: u32 = 32_768;
40
41// ── Pure control-word encoding ─────────────────────────────────────
42//
43// The RTC control register packs three flags into the low bits:
44//   bit 0 = enable, bit 1 = mode (1 = periodic), bit 2 = int_mask (1 = masked).
45// These are kept as named constants so the register pokes and the host tests
46// agree on the exact bit layout.
47
48/// Control register bit: enable the RTC counter.
49const CTRL_ENABLE: u32 = 0x01;
50/// Control register bit: periodic mode (cleared = free-running).
51const CTRL_PERIODIC: u32 = 1 << 1;
52/// Control register bit: interrupt mask (set = masked / disabled).
53const CTRL_INT_MASK: u32 = 1 << 2;
54
55/// Compute the initial control-register value for [`RtcDriver::configure`].
56///
57/// Pure helper (no MMIO): enables the counter, selects periodic vs.
58/// free-running mode, and starts with the interrupt masked.
59const fn control_bits(mode: RtcMode) -> u32 {
60    let mut ctrl = CTRL_ENABLE;
61    if matches!(mode, RtcMode::Periodic) {
62        ctrl |= CTRL_PERIODIC;
63    }
64    ctrl |= CTRL_INT_MASK;
65    ctrl
66}
67
68impl<'d> RtcDriver<'d> {
69    /// Create a new RTC driver from the RTC peripheral.
70    pub fn new(rtc: Rtc<'d>) -> Self {
71        Self { _rtc: rtc }
72    }
73
74    fn regs(&self) -> &'static crate::soc::pac::rtc::RegisterBlock {
75        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid
76        unsafe { &*Rtc::ptr() }
77    }
78
79    /// Configure the RTC.
80    ///
81    /// # Arguments
82    ///
83    /// * `mode` - Operating mode (free-running or periodic).
84    /// * `load_value` - In periodic mode, the counter resets when reaching this value.
85    ///
86    /// typed-config exemption: `load_value` is written verbatim into the full 32-bit
87    /// `rtc_load_count` register, so every `u32` is a valid, runnable value — nothing
88    /// to truncate or clamp. (The board-crystal precondition is documented at the
89    /// module level.)
90    pub fn configure(&mut self, mode: RtcMode, load_value: u32) {
91        unsafe {
92            self.regs().rtc_load_count().write(|w| w.bits(load_value));
93        }
94
95        let ctrl = control_bits(mode);
96
97        unsafe {
98            self.regs().rtc_control().write(|w| w.bits(ctrl));
99        }
100    }
101
102    /// Enable the RTC counter.
103    pub fn enable(&mut self) {
104        let ctrl = self.regs().rtc_control().read().bits();
105        unsafe {
106            self.regs().rtc_control().write(|w| w.bits(ctrl | 0x01));
107        }
108    }
109
110    /// Disable the RTC counter.
111    pub fn disable(&mut self) {
112        let ctrl = self.regs().rtc_control().read().bits();
113        unsafe {
114            self.regs().rtc_control().write(|w| w.bits(ctrl & !0x01));
115        }
116    }
117
118    /// Set the load (alarm) value.
119    pub fn set_load(&mut self, load_value: u32) {
120        unsafe {
121            self.regs().rtc_load_count().write(|w| w.bits(load_value));
122        }
123    }
124
125    /// Read the current counter value (lower 32 bits of the 48-bit counter).
126    pub fn current_value(&self) -> u32 {
127        self.regs().rtc_current_value().read().bits()
128    }
129
130    /// Enable RTC interrupt (unmask).
131    pub fn enable_interrupt(&mut self) {
132        let ctrl = self.regs().rtc_control().read().bits();
133        unsafe {
134            self.regs().rtc_control().write(|w| w.bits(ctrl & !(1 << 2)));
135        }
136    }
137
138    /// Disable RTC interrupt (mask).
139    pub fn disable_interrupt(&mut self) {
140        let ctrl = self.regs().rtc_control().read().bits();
141        unsafe {
142            self.regs().rtc_control().write(|w| w.bits(ctrl | (1 << 2)));
143        }
144    }
145
146    /// Check if an RTC interrupt is pending.
147    pub fn interrupt_pending(&self) -> bool {
148        self.regs().rtc_int_status().read().bits() & 0x01 != 0
149    }
150
151    /// Clear the RTC interrupt.
152    pub fn clear_interrupt(&self) {
153        let _ = self.regs().rtc_eoi().read().bits();
154    }
155}

157// ── Tests ──────────────────────────────────────────────────────
158
159#[cfg(all(test, not(target_arch = "riscv32")))]
160mod tests {
161    use super::*;
162
163    #[test]
164    fn free_running_control_word() {
165        // Free-running: enable + int-mask set, periodic bit clear → 0b101.
166        let ctrl = control_bits(RtcMode::FreeRunning);
167        assert_eq!(ctrl, CTRL_ENABLE | CTRL_INT_MASK);
168        assert_eq!(ctrl, 0b101);
169    }
170
171    #[test]
172    fn periodic_control_word() {
173        // Periodic: all three low bits set → 0b111.
174        let ctrl = control_bits(RtcMode::Periodic);
175        assert_eq!(ctrl, CTRL_ENABLE | CTRL_PERIODIC | CTRL_INT_MASK);
176        assert_eq!(ctrl, 0b111);
177    }
178
179    #[test]
180    fn periodic_bit_is_the_only_difference() {
181        // The mode flag toggles exactly bit 1 and nothing else.
182        let free = control_bits(RtcMode::FreeRunning);
183        let periodic = control_bits(RtcMode::Periodic);
184        assert_eq!(free ^ periodic, CTRL_PERIODIC);
185    }
186
187    #[test]
188    fn enable_always_set_and_interrupt_starts_masked() {
189        // Both modes power up enabled with the interrupt masked.
190        for mode in [RtcMode::FreeRunning, RtcMode::Periodic] {
191            let ctrl = control_bits(mode);
192            assert_ne!(ctrl & CTRL_ENABLE, 0, "enable bit must be set");
193            assert_ne!(ctrl & CTRL_INT_MASK, 0, "interrupt must start masked");
194        }
195    }
196
197    #[test]
198    fn control_bits_use_distinct_nonoverlapping_flags() {
199        // The three flags occupy distinct single bits in the low nibble.
200        assert_eq!(CTRL_ENABLE & CTRL_PERIODIC, 0);
201        assert_eq!(CTRL_ENABLE & CTRL_INT_MASK, 0);
202        assert_eq!(CTRL_PERIODIC & CTRL_INT_MASK, 0);
203        // No flag leaks above the low nibble that configure() writes.
204        assert_eq!((CTRL_ENABLE | CTRL_PERIODIC | CTRL_INT_MASK) & !0b111, 0);
205    }
206
207    #[test]
208    fn enable_disable_mask_round_trip() {
209        // Re-derive the RMW arithmetic of enable/disable/interrupt methods:
210        // setting then clearing a flag returns the original word, and the
211        // operations are idempotent.
212        let base = control_bits(RtcMode::Periodic);
213        // disable() clears bit 0, enable() sets it back.
214        assert_eq!((base & !CTRL_ENABLE) | CTRL_ENABLE, base);
215        // enable_interrupt() clears the mask, disable_interrupt() sets it.
216        assert_eq!((base & !CTRL_INT_MASK) | CTRL_INT_MASK, base);
217        // Idempotency: setting an already-set bit is a no-op.
218        assert_eq!(base | CTRL_ENABLE, base);
219    }
220
221    #[test]
222    fn rtc_clock_is_32_768_khz() {
223        // The RTC runs from the 32.768 kHz crystal: 2^15 ticks per second.
224        assert_eq!(RTC_CLOCK_HZ, 32_768);
225        assert_eq!(RTC_CLOCK_HZ, 1 << 15);
226    }
227}

229// ── Property-based fuzz tests ──────────────────────────────────
230
231#[cfg(all(test, not(target_arch = "riscv32")))]
232mod proptests {
233    use super::{CTRL_ENABLE, CTRL_INT_MASK, CTRL_PERIODIC, RtcMode, control_bits};
234    use proptest::prelude::*;
235
236    // Strategy producing both RtcMode variants.
237    fn any_mode() -> impl Strategy<Value = RtcMode> {
238        prop_oneof![Just(RtcMode::FreeRunning), Just(RtcMode::Periodic)]
239    }
240
241    proptest! {
242        /// Fuzz: the control word never sets a bit outside the documented low
243        /// nibble (enable|periodic|int_mask), for either mode. A stray high bit
244        /// would corrupt the register write in `configure()`.
245        #[test]
246        fn control_word_has_no_stray_bits(mode in any_mode()) {
247            let ctrl = control_bits(mode);
248            prop_assert_eq!(ctrl & !(CTRL_ENABLE | CTRL_PERIODIC | CTRL_INT_MASK), 0);
249        }
250
251        /// Fuzz: enable and int_mask are unconditionally set in every mode
252        /// (power-up enabled, interrupt masked) — independent of the mode flag.
253        #[test]
254        fn enable_and_mask_always_set(mode in any_mode()) {
255            let ctrl = control_bits(mode);
256            prop_assert_ne!(ctrl & CTRL_ENABLE, 0);
257            prop_assert_ne!(ctrl & CTRL_INT_MASK, 0);
258        }
259
260        /// Fuzz: the periodic bit (and only it) reflects the mode — set iff
261        /// Periodic, clear iff FreeRunning, with all other bits identical.
262        #[test]
263        fn periodic_bit_tracks_mode(mode in any_mode()) {
264            let ctrl = control_bits(mode);
265            let want_periodic = matches!(mode, RtcMode::Periodic);
266            prop_assert_eq!(ctrl & CTRL_PERIODIC != 0, want_periodic);
267            // Masking out the mode flag yields the same base word for both modes.
268            prop_assert_eq!(ctrl & !CTRL_PERIODIC, CTRL_ENABLE | CTRL_INT_MASK);
269        }
270
271        /// Fuzz: encoding is total/deterministic — never panics and is stable
272        /// across repeated calls for any mode input.
273        #[test]
274        fn control_bits_is_deterministic(mode in any_mode()) {
275            prop_assert_eq!(control_bits(mode), control_bits(mode));
276        }
277
278        /// Fuzz: RMW round-trip — clearing then re-setting any single control
279        /// flag (the arithmetic of enable/disable/{enable,disable}_interrupt)
280        /// returns the original word, mirroring the driver's `ctrl & !bit | bit`.
281        #[test]
282        fn flag_clear_then_set_round_trips(mode in any_mode(), bits in any::<u32>()) {
283            // Re-derive against an arbitrary register word, not just the encoder
284            // output, to exercise the RMW logic over the full 32-bit space.
285            let base = control_bits(mode) | bits;
286            for flag in [CTRL_ENABLE, CTRL_PERIODIC, CTRL_INT_MASK] {
287                // If the flag was set, clear+set restores it; if it was clear,
288                // set+clear restores it. Either way the word is recovered.
289                let had = base & flag != 0;
290                if had {
291                    prop_assert_eq!((base & !flag) | flag, base);
292                } else {
293                    prop_assert_eq!((base | flag) & !flag, base);
294                }
295            }
296        }
297
298        /// Fuzz: setting an already-set flag is idempotent; clearing an
299        /// already-clear flag is idempotent (the enable/disable methods rely on
300        /// this — calling enable() twice must not change the word).
301        #[test]
302        fn flag_ops_are_idempotent(bits in any::<u32>()) {
303            for flag in [CTRL_ENABLE, CTRL_PERIODIC, CTRL_INT_MASK] {
304                let set = bits | flag;
305                prop_assert_eq!(set | flag, set);
306                let cleared = bits & !flag;
307                prop_assert_eq!(cleared & !flag, cleared);
308            }
309        }
310    }
