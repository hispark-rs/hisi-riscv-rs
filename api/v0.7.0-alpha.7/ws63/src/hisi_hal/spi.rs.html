1//! SPI master driver for WS63 (SPI0/1, SSI v151).
2//! DesignWare SSI: SCK = SSI_CLK / SCKDV. Two-stage clock: a CLDO_CRG divider sets
3//! SSI_CLK = 160MHz off the 480MHz PLL (`configure_spi_source_clock`, mirrors the
4//! vendor `spi_porting_clock_init`), then SCKDV divides to SCK. SSI_CLK =
5//! [`crate::soc::chip::SPI_CLOCK_HZ`] (NOT the 240MHz CPU clock; SCKDV is even, >= 2).
6
7use crate::peripherals::{Spi0, Spi1};
8use core::marker::PhantomData;

10/// SPI clock polarity/phase mode (CTRA.scpol bit 3 / scph bit 4).
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum SpiMode {
13    /// CPOL=0, CPHA=0: idle-low clock, sample on leading edge.
14    Mode0,
15    /// CPOL=0, CPHA=1: idle-low clock, sample on trailing edge.
16    Mode1,
17    /// CPOL=1, CPHA=0: idle-high clock, sample on leading edge.
18    Mode2,
19    /// CPOL=1, CPHA=1: idle-high clock, sample on trailing edge.
20    Mode3,
21}

23/// A validated SPI bus clock. The DesignWare SSI divides `SPI_CLOCK_HZ` (160 MHz)
24/// by an even SCKDV in `[2, 0xFFFE]`, so the achievable SCK is
25/// `[SPI_CLOCK_HZ/0xFFFE ≈ 2.4 kHz, SPI_CLOCK_HZ/2 = 80 MHz]`. `try_from_hz` rejects
26/// anything outside that band instead of silently clamping the divider (the old
27/// `frequency: u32` path clamped to 2× / ½× the requested SCK without telling you).
28#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
29#[cfg_attr(feature = "defmt", derive(defmt::Format))]
30pub struct SpiHz(u32);
31
32impl SpiHz {
33    /// 1 MHz — the default SPI bus clock.
34    pub const ONE_MHZ: SpiHz = SpiHz(1_000_000);
35
36    /// Construct from a target SCK frequency. `None` if the resulting SCKDV
37    /// divider would fall outside `[2, 0xFFFE]` (frequency too high or too low).
38    pub const fn try_from_hz(hz: u32) -> Option<Self> {
39        if hz == 0 {
40            return None;
41        }
42        let div = crate::soc::chip::SPI_CLOCK_HZ / hz;
43        if div < 2 || div > 0xFFFE {
44            return None;
45        }
46        Some(SpiHz(hz))
47    }
48
49    /// The requested SCK frequency in Hz.
50    pub const fn hz(self) -> u32 {
51        self.0
52    }
53
54    /// The even SCKDV divider for this frequency (always in `[2, 0xFFFE]`).
55    const fn to_sckdv(self) -> u32 {
56        sckdv(crate::soc::chip::SPI_CLOCK_HZ, self.0)
57    }
58}

60/// SPI data frame size in bits, validated to the SSI DFS range `4..=16`.
61///
62/// (The 4-bit DFS field caps the documented range at 16; whether this silicon also
63/// supports the 32-bit DFS extension is an on-board open question — `4..=16` is the
64/// conservative, always-runnable choice. A value outside it is unrepresentable
65/// rather than silently masked into the register.)
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67#[cfg_attr(feature = "defmt", derive(defmt::Format))]
68pub struct DataBits(u8);
69
70impl DataBits {
71    /// 8-bit frames (the common default).
72    pub const EIGHT: DataBits = DataBits(8);
73    /// 16-bit frames.
74    pub const SIXTEEN: DataBits = DataBits(16);
75
76    /// Construct from a frame size. `None` outside `4..=16`.
77    pub const fn new(bits: u8) -> Option<Self> {
78        if bits >= 4 && bits <= 16 { Some(DataBits(bits)) } else { None }
79    }
80
81    /// The frame size in bits (always `4..=16`).
82    pub const fn bits(self) -> u8 {
83        self.0
84    }
85}

132/// SCKDV clock divider for the DesignWare SSI: `SCK = SSI_CLK / SCKDV`.
133///
134/// Matches the WS63 C SDK (`hal_spi_v151.c`: `clk_div = bus_clk / freq`,
135/// clamped to `SPI_MINUMUM_CLK_DIV` = 2). SCKDV bit 0 is read-only 0, so the
136/// value must be even. There is NO `/2` and NO `-1` (an earlier version of this
137/// driver had both, producing ~2x the requested SCK).
138const fn sckdv(pclk: u32, freq: u32) -> u32 {
139    let freq = if freq == 0 { 1 } else { freq };
140    let div = match pclk / freq {
141        d if d < 2 => 2,
142        d if d > 0xFFFF => 0xFFFF,
143        d => d,
144    };
145    div & !1 // SCKDV LSB is read-only 0 (must be even)
146}

148/// Bounded busy-wait. Returns [`SpiError::Timeout`] instead of hanging the CPU
149/// forever if a status bit never asserts (no slave, stuck CS, wrong mode) — the
150/// C SDK guards every equivalent wait with `hal_spi_check_timeout_by_count`.
151const SPI_WAIT_LOOPS: u32 = 1_000_000;
152
153#[inline]
154fn wait_until(mut ready: impl FnMut() -> bool) -> Result<(), SpiError> {
155    let mut n = SPI_WAIT_LOOPS;
156    while !ready() {
157        n -= 1;
158        if n == 0 {
159            return Err(SpiError::Timeout);
160        }
161    }
162    Ok(())
163}

181// ── SPI clock source (CLDO_CRG two-stage divider) ───────────────────────────
182// The SPI controller input clock (SSI_CLK) is derived from the 480 MHz FNPLL tap
183// by a CLDO_CRG divider, then divided again by the in-controller SCKDV to form
184// SCK (SCK = SSI_CLK / SCKDV). The HAL targets SSI_CLK = SPI_CLOCK_HZ (160 MHz);
185// `configure_spi_source_clock` programs the CRG divider to establish it and
186// switches the SPI clock source TCXO→PLL. Mirrors fbb_ws63 `spi_porting_clock_init`
187// (480/bus_clk_MHz into DIV_CTL3[9:5]); see ws63-guide ch8 "时钟树".
188#[cfg(feature = "chip-ws63")]
189const SPI_PLL_ROOT_MHZ: u32 = 480; // FNPLL SPI/QSPI tap (2880 / 6)
190
191/// Establish the two-stage SPI clock: program the CLDO_CRG divider so the SPI
192/// controller input clock (SSI_CLK) = [`crate::soc::chip::SPI_CLOCK_HZ`] off the
193/// 480 MHz PLL, then switch the SPI clock source from TCXO to PLL
194/// (gate-close → switch → gate-open). Bus-agnostic (one divider/select/gate for
195/// the whole SPI domain) and idempotent. Requires the PLL to already be locked
196/// (the app's `clock_init` does this before any driver init).
197#[cfg(feature = "chip-ws63")]
198fn configure_spi_source_clock() {
199    // CRG divider output = 480 MHz / div → div = 480 / SSI_CLK_MHz (e.g. 3 for 160 MHz).
200    // typed-config exemption: both inputs are compile-time constants (this fn takes
201    // no args), so the `.max`/`.clamp` guard the *constant* clock-tree math, not any
202    // user value — the user-facing SPI rate is the validated `SpiHz` newtype.
203    let ssi_mhz = (crate::soc::chip::SPI_CLOCK_HZ / 1_000_000).max(1);
204    let div = (SPI_PLL_ROOT_MHZ / ssi_mhz).clamp(1, 0x1F);
205    let cldo = unsafe { &*crate::peripherals::CldoCrg::ptr() };
206
207    // load-disable (bit10) → set [9:5]=div, [4:0]=1 → load-enable, per the SDK.
208    cldo.div_ctl3().modify(|_, w| w.spi_load_div_en().clear_bit());
209    cldo.div_ctl3().modify(|_, w| unsafe { w.spi_div2_cfg().bits((div & 0x1F) as u8).spi_div1_cfg().bits(1) });
210    cldo.div_ctl3().modify(|_, w| w.spi_load_div_en().set_bit());
211
212    // Gate-close → switch SPI source to PLL (CLK_SEL bit 6) → gate-open (CKEN1 bit 25).
213    cldo.cken_ctl1().modify(|_, w| w.spi_cken().clear_bit());
214    cldo.clk_sel().modify(|_, w| w.spi_clk_sel().set_bit());
215    cldo.cken_ctl1().modify(|_, w| w.spi_cken().set_bit());
216}

218fn configure_spi(idx: u8, config: &Config) {
219    // WS63 two-stage SPI clock (CLDO_CRG @0x4400_11xx → PLL). BS2X has a different
220    // clock tree (64 MHz app core, different CRG layout), so this WS63-specific CRG
221    // sequence must NOT run on BS2X — there the SPI runs off its default input clock
222    // and only the in-controller SCKDV divider (spi_brs, below) is programmed against
223    // soc::chip::SPI_CLOCK_HZ. TODO(bs2x): port the BS2X SPI source-clock setup from
224    // fbb_bs2x spi_porting.c when bringing SPI up on silicon (QEMU ignores the divisor).
225    #[cfg(feature = "chip-ws63")]
226    configure_spi_source_clock();
227    let r = spi_regs(idx);
228    r.spi_er().write(|w| unsafe { w.bits(0) });
229    r.spi_brs().write(|w| unsafe { w.bits(config.frequency.to_sckdv()) });
230
231    let mut ctra = 0u32;
232    match config.mode {
233        SpiMode::Mode0 => {}
234        SpiMode::Mode1 => ctra |= 1 << 3,
235        SpiMode::Mode2 => ctra |= 1 << 4,
236        SpiMode::Mode3 => ctra |= (1 << 3) | (1 << 4),
237    }
238    ctra |= ((config.data_bits.bits() - 1) as u32) << 13;
239    // CTRA.trsm (bits 18:19): 0b00 = transmit-and-receive (full duplex).
240    // (0b11 is EEPROM-read, NOT TX+RX — leaving trsm = 0 is correct.)
241    r.spi_ctra().write(|w| unsafe { w.bits(ctra) });
242    r.spi_slenr().write(|w| unsafe { w.bits(0x1) });
243    r.spi_er().write(|w| unsafe { w.bits(0x1) });
244}

246impl<'d, T> Spi<'d, T> {
247    /// Full-duplex transfer: write `write` while reading into `read`, byte-paced
248    /// through the TX/RX FIFOs (zero-padded TX / discarded RX for the shorter slice).
249    pub fn transfer(&mut self, write: &[u8], read: &mut [u8]) -> Result<(), SpiError> {
250        let r = spi_regs(self.idx);
251        let len = write.len().max(read.len());
252        for i in 0..len {
253            let tx = if i < write.len() { write[i] as u32 } else { 0 };
254            wait_until(|| r.spi_wsr().read().txfnf().bit_is_set())?;
255            unsafe { spi_data_reg(r).write(|w| w.bits(tx)) };
256            wait_until(|| r.spi_wsr().read().rxfne().bit_is_set())?;
257            let rx = spi_data_reg(r).read().bits();
258            if i < read.len() {
259                read[i] = rx as u8;
260            }
261        }
262        Ok(())
263    }
264
265    /// Write-only transfer: push `data` into the TX FIFO, discarding any RX bytes.
266    pub fn write(&mut self, data: &[u8]) -> Result<(), SpiError> {
267        let r = spi_regs(self.idx);
268        for &byte in data {
269            wait_until(|| r.spi_wsr().read().txfnf().bit_is_set())?;
270            unsafe { spi_data_reg(r).write(|w| w.bits(byte as u32)) };
271        }
272        Ok(())
273    }

275    /// Borrow the underlying PAC register block for this SPI instance.
276    ///
277    /// # Safety
278    /// This bypasses the typed SPI driver. The caller must uphold all PAC
279    /// aliasing, ordering, and peripheral-state invariants.
280    #[instability::unstable]
281    pub unsafe fn register_block(&self) -> &'static crate::soc::pac::spi0::RegisterBlock {
282        spi_regs(self.idx)
283    }

285    /// Wait for all pending TX data to be transmitted and the SPI bus to become idle.
286    pub fn wait_idle(&self) -> Result<(), SpiError> {
287        let r = spi_regs(self.idx);
288        // Wait for TX FIFO to drain, then for the bus to leave the busy state.
289        wait_until(|| r.spi_wsr().read().txfe().bit_is_set())?;
290        wait_until(|| !r.spi_wsr().read().busy().bit_is_set())?;
291        Ok(())
292    }

304/// A DMA-capable SPI master. Built from [`Spi::with_dma`]; owns the [`DmaDriver`].
305/// The blocking `Spi` is consumed (blocking + DMA APIs are mutually exclusive).
306///
307/// The DMA transfer methods ([`write_dma`](Self::write_dma),
308/// [`transfer_dma`](Self::transfer_dma)) are **blocking**: they program the
309/// peripheral + channel, bounded-wait for completion, and return the buffer. The
310/// buffer is owned for the whole call (a use-after-free of an in-flight DMA region
311/// is unrepresentable in safe code). Async `.await` variants are behind the `async`
312/// feature (P4).
313#[cfg(feature = "chip-ws63")]
314#[instability::unstable]
315pub struct SpiDma<'d, T> {
316    idx: u8,
317    dma: crate::dma::DmaDriver<'d, crate::dma::Dma0>,
318    _p: PhantomData<&'d T>,
319}

348    /// Write `buf` to MOSI via DMA (mem→peripheral, TX-only). The SSI is
349    /// full-duplex, so the looped-back RX bytes are drained from the FIFO and
350    /// discarded (use [`transfer_dma`](Self::transfer_dma) to capture them).
351    /// `buf` is owned for the whole call and returned on success. `ch` is a claimed
352    /// [`DmaChannel`](crate::dma::DmaChannel) token (from
353    /// [`DmaDriver::split_channels`](crate::dma::DmaDriver::split_channels)).
354    #[instability::unstable]
355    pub fn write_dma<B: embedded_dma::ReadBuffer<Word = u8>>(
356        &mut self,
357        ch: crate::dma::DmaChannel,
358        buf: B,
359    ) -> Result<B, SpiError> {
360        use crate::dma::{DmaChannelConfig, DmaFrame, DmaTransferSize};
361        let r = spi_regs(self.idx);
362        let (ptr, beats) = unsafe { buf.read_buffer() };
363        if beats > 0xFFF {
364            return Err(SpiError::BufferTooLong);
365        }
366        let size = DmaTransferSize::from_beats(beats).ok_or(SpiError::BufferTooLong)?;
367        let bytes = beats;
368        let dr = self.dr_addr();
369        let _ = ();
370
371        // Vendor order (hal_spi_v151.c:634, spi.c:691-712): watermark (DMA-enable
372        // OFF) → clean source → start channel → set tden.
373        unsafe {
374            r.spi_dtdl().write(|w| w.bits(4));
375            r.spi_drdl().write(|w| w.bits(0));
376        }
377        unsafe { crate::cache::clean_range(ptr as usize, bytes) };
378        let cfg = DmaChannelConfig::default().mem_to_peripheral(self.tx_peri()).with_width(DmaFrame::Byte);
379        let chn = ch.logical();
380        self.dma.configure_channel_raw(chn, ptr as u32, dr, size, &cfg);
381        r.spi_dcr().modify(|_, w| w.tden().set_bit());
382
383        // Bounded wait for the channel to auto-clear its enable bit (single-block done).
384        let mut n = SPI_WAIT_LOOPS;
385        while self.dma.channel_enabled_raw(chn) {
386            n -= 1;
387            if n == 0 {
388                // Cancel-then-quiesce: stop the peripheral asserting requests first,
389                // then halt → drain active → disable, before the buffer is returned.
390                r.spi_dcr().modify(|_, w| w.tden().clear_bit());
391                self.dma.halt_channel_raw(chn);
392                let mut m = SPI_WAIT_LOOPS;
393                while self.dma.channel_active_raw(chn) {
394                    m -= 1;
395                    if m == 0 {
396                        break;
397                    }
398                    core::hint::spin_loop();
399                }
400                self.dma.disable_channel_raw(chn);
401                return Err(SpiError::Timeout);
402            }
403            core::hint::spin_loop();
404        }
405        // Drain the looped-back RX FIFO (TX clocks the bus; RX fills) to avoid overrun.
406        while r.spi_wsr().read().rxfne().bit_is_set() {
407            let _ = spi_data_reg(r).read().bits();
408        }
409        r.spi_dcr().modify(|_, w| w.tden().clear_bit());
410        Ok(buf)
411    }

413    /// Async variant of [`write_dma`](Self::write_dma): parks on the DMA completion
414    /// IRQ (IRQ 59) via [`DmaDriver::wait_transfer_done`](crate::dma::DmaDriver::wait_transfer_done)
415    /// instead of bounded-spinning, so the core can `wfi` while the transfer runs.
416    /// Requires the `async` feature + global interrupts enabled
417    /// (`interrupt::enable_global`) at the call site. Silicon-verified (IRQ 59 fires
418    /// for peripheral-paced completion — see `spi_dma_irq59` HIL test).
419    #[cfg(all(feature = "async", feature = "unstable"))]
420    #[instability::unstable]
421    pub async fn write_dma_async<B: embedded_dma::ReadBuffer<Word = u8>>(
422        &mut self,
423        ch: crate::dma::DmaChannel,
424        buf: B,
425    ) -> Result<B, SpiError> {
426        use crate::dma::{DmaChannelConfig, DmaFrame, DmaTransferSize};
427        let r = spi_regs(self.idx);
428        let (ptr, beats) = unsafe { buf.read_buffer() };
429        if beats > 0xFFF {
430            return Err(SpiError::BufferTooLong);
431        }
432        let size = DmaTransferSize::from_beats(beats).ok_or(SpiError::BufferTooLong)?;
433        let bytes = beats;
434        let dr = self.dr_addr();
435
436        unsafe {
437            r.spi_dtdl().write(|w| w.bits(4));
438            r.spi_drdl().write(|w| w.bits(0));
439        }
440        unsafe { crate::cache::clean_range(ptr as usize, bytes) };
441        // transfer_int = true so the channel's done bit sets int_tc → IRQ 59.
442        let cfg = DmaChannelConfig::default()
443            .mem_to_peripheral(self.tx_peri())
444            .with_width(DmaFrame::Byte)
445            .with_transfer_int(true);
446        let chn = ch.logical();
447        self.dma.configure_channel_raw(chn, ptr as u32, dr, size, &cfg);
448        r.spi_dcr().modify(|_, w| w.tden().set_bit());
449
450        // Park on IRQ 59 (per-channel demux) until this channel completes.
451        self.dma.wait_transfer_done(&ch).await;
452
453        // Drain the looped-back RX FIFO, then quiesce the peripheral.
454        while r.spi_wsr().read().rxfne().bit_is_set() {
455            let _ = spi_data_reg(r).read().bits();
456        }
457        r.spi_dcr().modify(|_, w| w.tden().clear_bit());
458        Ok(buf)
459    }

461    /// Full-duplex DMA: `write` → MOSI via `tx_ch` while MISO → `read` via `rx_ch`,
462    /// concurrently. With a MOSI→MISO jumper `read` ends up equal to `write`. Both
463    /// buffers are owned for the whole call and returned on success.
464    #[instability::unstable]
465    pub fn transfer_dma<RB: embedded_dma::WriteBuffer<Word = u8>, TB: embedded_dma::ReadBuffer<Word = u8>>(
466        &mut self,
467        tx_ch: crate::dma::DmaChannel,
468        rx_ch: crate::dma::DmaChannel,
469        mut read: RB,
470        write: TB,
471    ) -> Result<(RB, TB), SpiError> {
472        use crate::dma::{DmaChannelConfig, DmaFrame, DmaTransferSize};
473        let r = spi_regs(self.idx);
474        let (tx_ptr, tx_beats) = unsafe { write.read_buffer() };
475        let (rx_ptr, rx_beats) = unsafe { read.write_buffer() };
476        let beats = tx_beats.min(rx_beats);
477        if tx_beats > 0xFFF || rx_beats > 0xFFF {
478            return Err(SpiError::BufferTooLong);
479        }
480        let size = DmaTransferSize::from_beats(beats).ok_or(SpiError::BufferTooLong)?;
481        let bytes = beats;
482        let dr = self.dr_addr();
483
484        unsafe {
485            r.spi_dtdl().write(|w| w.bits(4));
486            r.spi_drdl().write(|w| w.bits(0));
487        }
488        // Clean TX source; RX destination is invalidated AFTER completion.
489        unsafe { crate::cache::clean_range(tx_ptr as usize, bytes) };
490
491        let tx_cfg = DmaChannelConfig::default().mem_to_peripheral(self.tx_peri()).with_width(DmaFrame::Byte);
492        let rx_cfg = DmaChannelConfig::default().peripheral_to_mem(self.rx_peri()).with_width(DmaFrame::Byte);
493        let tx_chn = tx_ch.logical();
494        let rx_chn = rx_ch.logical();
495        self.dma.configure_channel_raw(tx_chn, tx_ptr as u32, dr, size, &tx_cfg);
496        self.dma.configure_channel_raw(rx_chn, dr, rx_ptr as u32, size, &rx_cfg);
497        r.spi_dcr().modify(|_, w| w.tden().set_bit().rden().set_bit());
498
499        let mut n = SPI_WAIT_LOOPS;
500        while self.dma.channel_enabled_raw(tx_chn) || self.dma.channel_enabled_raw(rx_chn) {
501            n -= 1;
502            if n == 0 {
503                r.spi_dcr().modify(|_, w| w.tden().clear_bit().rden().clear_bit());
504                return Err(SpiError::Timeout);
505            }
506            core::hint::spin_loop();
507        }
508        // Invalidate the RX destination so the CPU re-reads what DMA wrote.
509        unsafe { crate::cache::invalidate_range(rx_ptr as usize, bytes) };
510        r.spi_dcr().modify(|_, w| w.tden().clear_bit().rden().clear_bit());
511        Ok((read, write))
512    }
513
514    /// Async variant of [`transfer_dma`](Self::transfer_dma): arms both channels
515    /// (transfer_int = true), sets tden+rden, and awaits each channel's
516    /// completion on IRQ 59 in turn (the per-channel demux means awaiting the second
517    /// doesn't false-wake on the first). Requires `async` + global interrupts.
518    #[instability::unstable]
519    #[cfg(all(feature = "async", feature = "unstable"))]
520    pub async fn transfer_dma_async<
521        RB: embedded_dma::WriteBuffer<Word = u8>,
522        TB: embedded_dma::ReadBuffer<Word = u8>,
523    >(
524        &mut self,
525        tx_ch: crate::dma::DmaChannel,
526        rx_ch: crate::dma::DmaChannel,
527        mut read: RB,
528        write: TB,
529    ) -> Result<(RB, TB), SpiError> {
530        use crate::dma::{DmaChannelConfig, DmaFrame, DmaTransferSize};
531        let r = spi_regs(self.idx);
532        let (tx_ptr, tx_beats) = unsafe { write.read_buffer() };
533        let (rx_ptr, rx_beats) = unsafe { read.write_buffer() };
534        let beats = tx_beats.min(rx_beats);
535        if tx_beats > 0xFFF || rx_beats > 0xFFF {
536            return Err(SpiError::BufferTooLong);
537        }
538        let size = DmaTransferSize::from_beats(beats).ok_or(SpiError::BufferTooLong)?;
539        let bytes = beats;
540        let dr = self.dr_addr();
541
542        unsafe {
543            r.spi_dtdl().write(|w| w.bits(4));
544            r.spi_drdl().write(|w| w.bits(0));
545        }
546        unsafe { crate::cache::clean_range(tx_ptr as usize, bytes) };
547
548        let tx_cfg = DmaChannelConfig::default()
549            .mem_to_peripheral(self.tx_peri())
550            .with_width(DmaFrame::Byte)
551            .with_transfer_int(true);
552        let rx_cfg = DmaChannelConfig::default()
553            .peripheral_to_mem(self.rx_peri())
554            .with_width(DmaFrame::Byte)
555            .with_transfer_int(true);
556        let tx_chn = tx_ch.logical();
557        let rx_chn = rx_ch.logical();
558        self.dma.configure_channel_raw(tx_chn, tx_ptr as u32, dr, size, &tx_cfg);
559        self.dma.configure_channel_raw(rx_chn, dr, rx_ptr as u32, size, &rx_cfg);
560        r.spi_dcr().modify(|_, w| w.tden().set_bit().rden().set_bit());
561
562        // Await each channel's completion (per-channel demux — no false-wake cross-talk).
563        self.dma.wait_transfer_done(&tx_ch).await;
564        self.dma.wait_transfer_done(&rx_ch).await;
565
566        unsafe { crate::cache::invalidate_range(rx_ptr as usize, bytes) };
567        r.spi_dcr().modify(|_, w| w.tden().clear_bit().rden().clear_bit());
568        Ok((read, write))
569    }
570
571    /// Reclaim the blocking `Spi` and the `DmaDriver`. Clears any peripheral
572    /// DMA-enable the DMA paths may have set.
573    #[instability::unstable]
574    pub fn release(self) -> (Spi<'d, T>, crate::dma::DmaDriver<'d, crate::dma::Dma0>) {
575        let r = spi_regs(self.idx);
576        r.spi_dcr().modify(|_, w| w.tden().clear_bit().rden().clear_bit());
577        (Spi { idx: self.idx, _peripheral: PhantomData }, self.dma)
578    }
579}
580
581fn transfer_in_place_on(idx: u8, buf: &mut [u8]) -> Result<(), SpiError> {
582    let r = spi_regs(idx);
583    for byte in buf.iter_mut() {
584        let tx = *byte as u32;
585        wait_until(|| r.spi_wsr().read().txfnf().bit_is_set())?;
586        unsafe { spi_data_reg(r).write(|w| w.bits(tx)) };
587        wait_until(|| r.spi_wsr().read().rxfne().bit_is_set())?;
588        *byte = spi_data_reg(r).read().bits() as u8;
589    }
590    Ok(())
591}

593/// Errors returned by the SPI driver.
594#[derive(Debug)]
595#[cfg_attr(feature = "defmt", derive(defmt::Format))]
596#[non_exhaustive]
597pub enum SpiError {
598    /// FIFO overrun (maps to [`embedded_hal::spi::ErrorKind::Overrun`]).
599    Overflow,
600    /// A status bit never asserted within the bounded wait (no slave, stuck CS, wrong mode),
601    /// or a DMA channel never completed within the bounded poll.
602    Timeout,
603    /// A DMA buffer exceeded the 12-bit `trans_size` field (4095 beats per single block).
604    BufferTooLong,
605}
606
607impl embedded_hal::spi::Error for SpiError {
608    fn kind(&self) -> embedded_hal::spi::ErrorKind {
609        match self {
610            SpiError::Overflow => embedded_hal::spi::ErrorKind::Overrun,
611            _ => embedded_hal::spi::ErrorKind::Other,
612        }
613    }
614}
615impl embedded_hal::spi::ErrorType for Spi<'_, Spi0<'_>> {
616    type Error = SpiError;
617}
618impl embedded_hal::spi::SpiBus for Spi<'_, Spi0<'_>> {
619    fn transfer(&mut self, read: &mut [u8], write: &[u8]) -> Result<(), Self::Error> {
620        self.transfer(write, read)
621    }
622    fn read(&mut self, buf: &mut [u8]) -> Result<(), Self::Error> {
623        self.transfer(&[], buf)
624    }
625    fn write(&mut self, buf: &[u8]) -> Result<(), Self::Error> {
626        Spi::write(self, buf)
627    }
628    fn transfer_in_place(&mut self, buf: &mut [u8]) -> Result<(), Self::Error> {
629        transfer_in_place_on(self.idx, buf)
630    }
631    fn flush(&mut self) -> Result<(), Self::Error> {
632        self.wait_idle()
633    }
634}

689    }

691    #[test]
692    fn spi_hz_rejects_out_of_range() {
693        use super::SpiHz;

696        // Too high: > SPI_CLOCK_HZ/2 (=80 MHz) → div < 2.
697        assert!(SpiHz::try_from_hz(SPI_CLOCK_HZ / 2 + 1).is_none());
698        assert!(SpiHz::try_from_hz(SPI_CLOCK_HZ).is_none());

703        assert_eq!(SpiHz::ONE_MHZ.to_sckdv(), 160);
704        // Exactly the edges resolve to the min/max even divider.
705        assert_eq!(SpiHz::try_from_hz(SPI_CLOCK_HZ / 2).unwrap().to_sckdv(), 2);
706    }

748// ── Async SPI (embedded-hal-async) ──────────────────────────────────────────
749// WS63 SPI transfers are FIFO-paced (and the ws63-qemu model loops back
750// synchronously), so the async SpiBus methods complete promptly by reusing the
751// blocking transfer logic — valid `async fn`s usable from embassy/async tasks. A
752// genuinely IRQ-parking variant would add an on_interrupt + IrqSignal once the
753// SPI completion IRQ (43/52) is modelled.
754#[cfg(feature = "async")]
755mod asynch_impl {
756    use super::{Spi, Spi0, Spi1};
757
758    macro_rules! async_spi {
759        ($inst:ty) => {
760            impl embedded_hal_async::spi::SpiBus<u8> for Spi<'_, $inst> {
761                async fn read(&mut self, buf: &mut [u8]) -> Result<(), Self::Error> {
762                    embedded_hal::spi::SpiBus::read(self, buf)
763                }
764                async fn write(&mut self, buf: &[u8]) -> Result<(), Self::Error> {
765                    embedded_hal::spi::SpiBus::write(self, buf)
766                }
767                async fn transfer(&mut self, read: &mut [u8], write: &[u8]) -> Result<(), Self::Error> {
768                    embedded_hal::spi::SpiBus::transfer(self, read, write)
769                }
770                async fn transfer_in_place(&mut self, buf: &mut [u8]) -> Result<(), Self::Error> {
771                    embedded_hal::spi::SpiBus::transfer_in_place(self, buf)
772                }
773                async fn flush(&mut self) -> Result<(), Self::Error> {
774                    embedded_hal::spi::SpiBus::flush(self)
775                }
776            }
777        };
778    }
779    async_spi!(Spi0<'_>);
780    async_spi!(Spi1<'_>);
