1//! BS2X GADC — 13-bit general-purpose ADC (IP version v153).
2//!
3//! BS2X-only (`chip-bs21`). The WS63 has a different ADC (LSADC v154, the `lsadc`
4//! module); this is a ground-up driver for the BS2X GADC, whose register layout
5//! shares nothing with the LSADC.
6//!
7//! A single conversion touches **three** hardware blocks (from the fbb_bs2x
8//! `hal_adc_v153` driver, the sole ground truth):
9//! - the GADC digital block @ `0x5703_6000` (in `bs2x-pac` as `Gadc`): reset/clock,
10//!   channel select (`cfg_amux_1`), the done flag (`rpt_gadc_data_3` bit0) and the
11//!   result (`rpt_gadc_data_2`);
12//! - the ANA/LDO sub-block @ `0x5703_63D0` (GADC base + 0x3D0, modeled in the
13//!   same `Gadc` PAC register block): the AFE/ADC/VREF LDO power-up;
14//! - the PMU AFE sub-block @ `0x5700_8700` (in `bs2x-pac` as `AdcPmuAfe`):
15//!   MTCMOS power / isolation / reset / clock and the `afe_gadc_cfg` enable
16//!   handshake; plus `AonAfe.afe_iso[10]` @ `0x5702_C230`.
17//!
18//! The exact analog timing/trim magic values are the SDK defaults; they are
19//! reproduced where they matter for control flow and marked `TODO(bs2x)` where
20//! only silicon cares (the QEMU GADC model returns a fixed sample regardless of
21//! the analog config, so this driver's control path — power-up → channel select
22//! → trigger → poll done → read — is what is exercised on `-M bs21/bs22/bs20`).
23
24use crate::peripherals::Gadc as GadcPeriph;
25use core::marker::PhantomData;

27/// Rough microsecond busy-wait (the analog settling delays). Sized for the 64 MHz
28/// BS2X app core; only matters on silicon (QEMU ignores it).
29#[inline]
30fn delay_us(us: u32) {
31    let cycles = us.saturating_mul(64);
32    for _ in 0..cycles {
33        core::hint::spin_loop();
34    }
35}

37/// GADC input channels (AIN0..AIN7, single-ended; n-side tied to VSSAFE1).
38#[derive(Clone, Copy, Debug, PartialEq, Eq)]
39#[repr(u8)]
40pub enum AdcChannel {
41    /// Analog input 0 (p-side one-hot `BIT(0)` in `cfg_amux_1`).
42    Ain0 = 0,
43    /// Analog input 1 (p-side one-hot `BIT(1)` in `cfg_amux_1`).
44    Ain1 = 1,
45    /// Analog input 2 (p-side one-hot `BIT(2)` in `cfg_amux_1`).
46    Ain2 = 2,
47    /// Analog input 3 (p-side one-hot `BIT(3)` in `cfg_amux_1`).
48    Ain3 = 3,
49    /// Analog input 4 (p-side one-hot `BIT(4)` in `cfg_amux_1`).
50    Ain4 = 4,
51    /// Analog input 5 (p-side one-hot `BIT(5)` in `cfg_amux_1`).
52    Ain5 = 5,
53    /// Analog input 6 (p-side one-hot `BIT(6)` in `cfg_amux_1`).
54    Ain6 = 6,
55    /// Analog input 7 (p-side one-hot `BIT(7)` in `cfg_amux_1`).
56    Ain7 = 7,
57}
58
59const VSSAFE1_BIT: u32 = 9; // n-side reference channel (cfg_amux_1 amuxn)
60
61/// Bounded spin limit for the conversion-done poll (`rpt_gadc_data_3` bit 0). A
62/// single GADC conversion latches within microseconds at the SDK clock divider;
63/// this cap is several orders above that so a wedged AFE/LDO (e.g. analog never
64/// powered) yields [`GadcError::ConversionTimeout`] instead of hanging forever.
65const GADC_DONE_POLL_LIMIT: u32 = 1_000_000;
66
67/// GADC conversion error.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69#[cfg_attr(feature = "defmt", derive(defmt::Format))]
70#[non_exhaustive]
71pub enum GadcError {
72    /// The conversion-done flag did not assert within `GADC_DONE_POLL_LIMIT`
73    /// spins — the analog front-end is unpowered or the trigger handshake stalled.
74    ConversionTimeout,
75}

77/// The BS2X GADC driver (single instance).
78pub struct Gadc<'d> {
79    _gadc: PhantomData<GadcPeriph<'d>>,
80    first_sample: bool,
81}
82
83impl<'d> Gadc<'d> {
84    fn regs(&self) -> &'static crate::soc::pac::gadc::RegisterBlock {
85        // SAFETY: static physical MMIO address from bs2x-pac.
86        unsafe { &*GadcPeriph::ptr() }
87    }
88
89    fn pmu_regs(&self) -> &'static crate::soc::pac::adc_pmu_afe::RegisterBlock {
90        // SAFETY: static physical MMIO address from bs2x-pac.
91        unsafe { &*crate::soc::pac::AdcPmuAfe::ptr() }
92    }
93
94    fn aon_afe_regs(&self) -> &'static crate::soc::pac::aon_afe::RegisterBlock {
95        // SAFETY: static physical MMIO address from bs2x-pac.
96        unsafe { &*crate::soc::pac::AonAfe::ptr() }
97    }
98
99    fn write_gadc_cfg(
100        pmu: &'static crate::soc::pac::adc_pmu_afe::RegisterBlock,
101        en: bool,
102        mux_en: bool,
103        iso_en: bool,
104    ) {
105        pmu.afe_gadc_cfg().write(|w| {
106            let w = if en { w.s2d_gadc_en().set_bit() } else { w.s2d_gadc_en().clear_bit() };
107            let w = if mux_en {
108                w.s2d_gadc_mux_en().set_bit()
109            } else {
110                w.s2d_gadc_mux_en().clear_bit()
111            };
112            if iso_en { w.s2d_gadc_iso_en().set_bit() } else { w.s2d_gadc_iso_en().clear_bit() }
113        });
114    }
115
116    /// Power up the GADC and bring it live. Mirrors `hal_adc_v153_init` ->
117    /// `_power_on` -> `_enable` -> `hal_gafe_enable` (fbb_bs2x).
118    ///
119    /// # Preconditions (analog)
120    ///
121    /// This drives the GADC digital block, the ANA/LDO sub-block, and the PMU AFE
122    /// (MTCMOS power / isolation / reset / clock) through the vendor power-up
123    /// sequence with the SDK's settling delays. It assumes the **board's analog
124    /// supply and 32 MHz reference are up** (the AON `XO32M` path); the efuse LDO
125    /// trim is skipped (`TODO(bs2x)`). Without a live AFE, a conversion never
126    /// completes — [`Gadc::read`] then returns [`GadcError::ConversionTimeout`]
127    /// rather than hanging. Not validated on silicon (QEMU returns a fixed sample).
128    pub fn new(_gadc: GadcPeriph<'d>) -> Self {
129        let this = Self { _gadc: PhantomData, first_sample: true };
130        let r = this.regs();
131        let pmu = this.pmu_regs();
132        let aon = this.aon_afe_regs();
133        // Phase A — block bring-up.
134        aon.afe_iso().modify(|_, w| w.afe_iso_en().set_bit()); // AFE iso enable (AON)
135        pmu.afe_iso_cfg().write(|w| unsafe { w.bits(0) }); // release XO32M iso
136        delay_us(30);
137        // MTCMOS power-on (pwr_en|iso_en).
138        pmu.afe_dig_pwr_en().write(|w| w.afe_pwr_en().set_bit().afe_iso_en().set_bit());
139        delay_us(50);
140        pmu.afe_dig_pwr_en().modify(|_, w| w.afe_iso_en().clear_bit()); // release iso
141        pmu.afe_adc_rst_n().write(|w| w.afe_adc_rst_n().set_bit()); // ana rstn release
142        pmu.afe_clk_en().write(|w| w.afe_clk_en().set_bit()); // dig clk release
143        pmu.afe_soft_rst().write(|w| w.afe_soft_rst().set_bit()); // dig apb rstn release
144        r.cfg_clken().reset();
145        r.cfg_rstn().reset();
146        r.cfg_clken().write(|w| {
147            w.cfg_clken_tst()
148                .set_bit()
149                .cfg_gadc_clken_bc()
150                .set_bit()
151                .cfg_gadc_clken_fc()
152                .set_bit()
153                .cfg_gadc_clken_byp()
154                .set_bit()
155                .cfg_gadc_clken_prechg()
156                .set_bit()
157                .cfg_gadc_clken_ctrl()
158                .set_bit()
159        }); // tst/bc/fc/byp/prechg/ctrl clocks on
160
161        // Phase B — LDO power-on (ANA sub-block). TODO(bs2x): efuse LDO trim skipped.
162        r.cfg_ana_6().write(|w| w.cfg_afe_afeldo_en().set_bit()); // afeldo
163        delay_us(150);
164        r.cfg_ana_7().modify(|_, w| w.cfg_afe_afeldo_en_dly().set_bit());
165        r.cfg_ana_4().write(|w| w.cfg_afe_adcldo_en().set_bit()); // adcldo
166        delay_us(150);
167        r.cfg_ana_5().modify(|_, w| w.cfg_afe_adcldo_en_dly().set_bit());
168        r.cfg_ana_0().modify(|_, w| w.cfg_vrefldo_en().set_bit()); // vrefldo_en
169        delay_us(150);
170
171        // Phase C — common + GADC analog enable.
172        r.cfg_clk_div_0().write(|w| unsafe { w.cfg_adc_ana_div_th().bits(0x27) }); // TODO(bs2x): COMMON_DEFAULT clk div
173        r.cfg_ana_1().modify(|_, w| w.cfg_bufp_en().set_bit().cfg_bufn_en().set_bit());
174        r.cfg_freg_5().reset();
175        r.cfg_freg_9().write(|w| unsafe { w.bits(2) });
176        r.cfg_tst_1().write(|w| unsafe { w.diag_node().bits(2) }); // ACCUMULATED_AVERAGE_OUTPUT diag node
177        r.cfg_rstn().write(|w| {
178            w.cfg_rstn_tst()
179                .set_bit()
180                .cfg_gadc_rstn_bc()
181                .set_bit()
182                .cfg_gadc_rstn_fc()
183                .set_bit()
184                .cfg_gadc_rstn_data()
185                .set_bit()
186                .cfg_gadc_rstn_ana()
187                .set_bit()
188        }); // release all GADC resets
189        r.cfg_iso().write(|w| unsafe { w.bits(0) });
190
191        // hal_gafe_enable: afe_gadc_cfg power-up handshake.
192        Self::write_gadc_cfg(pmu, false, true, true);
193        delay_us(30);
194        Self::write_gadc_cfg(pmu, false, true, false);
195        delay_us(30);
196        Self::write_gadc_cfg(pmu, true, true, false);
197        this
198    }

200    /// Select `channel`, run one conversion, and return the raw 18-bit signed
201    /// accumulated sample (`rpt_gadc_data_2`, sign-extended at bit 17). Mirrors
202    /// `hal_adc_v153_auto_sample`.
203    ///
204    /// # Errors
205    ///
206    /// [`GadcError::ConversionTimeout`] if the conversion-done flag never asserts
207    /// within the bounded poll (an unpowered AFE/LDO) — instead of spinning forever.
208    pub fn read(&mut self, channel: AdcChannel) -> Result<i32, GadcError> {
209        let r = self.regs();
210        // Channel select: p-side one-hot = BIT(channel), n-side = BIT(VSSAFE1),
211        // both divide-disable. cfg_amux_1: amuxn[10:0], amuxp[22:12].
212        let amuxp = 1u16 << (channel as u16);
213        let amuxn = 1u16 << VSSAFE1_BIT;
214        r.cfg_amux_1().write(|w| unsafe {
215            w.amuxn_sensor_ch_sel()
216                .bits(amuxn)
217                .amuxn_devide_disable()
218                .set_bit()
219                .amuxp_sensor_ch_sel()
220                .bits(amuxp)
221                .amuxp_devide_disable()
222                .set_bit()
223        });
224        r.cfg_amux_2().reset();
225
226        // The very first sample after power-up is discarded.
227        if self.first_sample {
228            self.convert_once()?;
229            self.first_sample = false;
230        }
231        self.convert_once()
232    }
233
234    /// Trigger one conversion, poll done (bounded), read + sign-extend the result.
235    fn convert_once(&self) -> Result<i32, GadcError> {
236        let r = self.regs();
237        let pmu = self.pmu_regs();
238        // Trigger (hal_gadc_iso_on): un-isolate + enable -> free-running scan.
239        delay_us(5);
240        pmu.afe_gadc_cfg().modify(|_, w| w.s2d_gadc_mux_en().set_bit());
241        pmu.afe_gadc_cfg().modify(|_, w| w.s2d_gadc_iso_en().clear_bit());
242        delay_us(5);
243        pmu.afe_gadc_cfg().modify(|_, w| w.s2d_gadc_en().set_bit());
244
245        // Poll done: rpt_gadc_data_3 bit0 (GADC block). Bounded so an unpowered
246        // AFE returns ConversionTimeout instead of hanging.
247        let mut done = false;
248        for _ in 0..GADC_DONE_POLL_LIMIT {
249            if r.rpt_gadc_data_3().read().single_sample_done().bit_is_set() {
250                done = true;
251                break;
252            }
253            core::hint::spin_loop();
254        }
255
256        // Stop / re-isolate (hal_gadc_iso_off) — always, even on timeout, so the
257        // AFE is left isolated rather than free-running.
258        pmu.afe_gadc_cfg().modify(|_, w| w.s2d_gadc_en().clear_bit());
259        delay_us(5);
260        pmu.afe_gadc_cfg().modify(|_, w| {
261            w.s2d_gadc_mux_en().clear_bit();
262            w.s2d_gadc_iso_en().set_bit()
263        });
264
265        if !done {
266            return Err(GadcError::ConversionTimeout);
267        }
268
269        // Read result + sign-extend (18-bit signed, sign bit 17).
270        let raw = r.rpt_gadc_data_2().read().sample_data().bits();
271        Ok(sign_extend18(raw))
272    }
273}
274
275/// Sign-extend an 18-bit GADC sample to `i32` (sign bit = 17).
276#[inline]
277fn sign_extend18(raw: u32) -> i32 {
278    let v = raw & 0x3FFFF;
279    if v & (1 << 17) != 0 { (v as i32) - 0x4_0000 } else { v as i32 }
280}

284#[cfg(all(test, not(target_arch = "riscv32")))]
285mod tests {

288    /// Re-derivation of the `cfg_amux_1` packing inlined in `Gadc::read`: p-side
289    /// one-hot `BIT(channel)`, n-side one-hot `BIT(VSSAFE1)`, both divide-disable.
290    /// Layout: amuxn[10:0], div-n bit11, amuxp[22:12], div-p bit23.
291    fn amux_1(channel: AdcChannel) -> u32 {
292        let amuxp = 1u32 << (channel as u32);
293        let amuxn = 1u32 << VSSAFE1_BIT;
294        (amuxn & 0x7FF) | (1 << 11) | ((amuxp & 0x7FF) << 12) | (1 << 23)
295    }
296
297    #[test]
298    fn channel_discriminants_are_index() {
299        // AdcChannel is repr(u8) numbered 0..=7; `channel as u32` is the AIN index.
300        assert_eq!(AdcChannel::Ain0 as u32, 0);
301        assert_eq!(AdcChannel::Ain3 as u32, 3);
302        assert_eq!(AdcChannel::Ain7 as u32, 7);
303    }
304
305    #[test]
306    fn sign_extend_positive_passthrough() {
307        // Values with bit17 clear are non-negative and unchanged.
308        assert_eq!(sign_extend18(0), 0);
309        assert_eq!(sign_extend18(1), 1);
310        // 0x1FFFF = largest positive 18-bit sample (bit17 clear) = 131071.
311        assert_eq!(sign_extend18(0x1FFFF), 131_071);
312    }
313
314    #[test]
315    fn sign_extend_negative_values() {
316        // bit17 set → negative. 0x20000 is the most-negative sample (-2^17).
317        assert_eq!(sign_extend18(0x2_0000), -131_072);
318        // 0x3FFFF (all 18 bits set) = -1.
319        assert_eq!(sign_extend18(0x3_FFFF), -1);
320        // 0x3FFFE = -2.
321        assert_eq!(sign_extend18(0x3_FFFE), -2);
322    }
323
324    #[test]
325    fn sign_extend_ignores_high_bits() {
326        // Only bits [17:0] are significant; junk above bit17 is masked off.
327        assert_eq!(sign_extend18(0xFFFC_0000), 0);
328        // High junk plus the bit17 sign + low bits round-trips to the masked value.
329        assert_eq!(sign_extend18(0xFFFF_FFFF), -1);
330        assert_eq!(sign_extend18(0xDEAD_0000 | 0x1FFFF), 131_071);
331    }
332
333    #[test]
334    fn sign_extend_range_is_18bit_signed() {
335        // The output is always within the signed 18-bit range [-2^17, 2^17).
336        for raw in [0u32, 0x1, 0x1_FFFF, 0x2_0000, 0x3_FFFF, 0xFFFF_FFFF] {
337            let v = sign_extend18(raw);
338            assert!((-131_072..=131_071).contains(&v), "raw={raw:#x} -> {v}");
339        }
340    }
341
342    #[test]
343    fn amux_n_side_fixed_to_vssafe1() {
344        // The n-side is always BIT(VSSAFE1)=BIT(9) regardless of channel, and both
345        // divide-disable bits (11 and 23) are always set.
346        for ch in [AdcChannel::Ain0, AdcChannel::Ain4, AdcChannel::Ain7] {
347            let v = amux_1(ch);
348            assert_eq!(v & 0x7FF, 1 << VSSAFE1_BIT, "amuxn for {ch:?}");
349            assert_ne!(v & (1 << 11), 0, "div-n bit for {ch:?}");
350            assert_ne!(v & (1 << 23), 0, "div-p bit for {ch:?}");
351        }
352    }
353
354    #[test]
355    fn amux_p_side_is_channel_onehot() {
356        // The p-side field [22:12] is a one-hot of the selected channel index.
357        assert_eq!((amux_1(AdcChannel::Ain0) >> 12) & 0x7FF, 1 << 0);
358        assert_eq!((amux_1(AdcChannel::Ain5) >> 12) & 0x7FF, 1 << 5);
359        assert_eq!((amux_1(AdcChannel::Ain7) >> 12) & 0x7FF, 1 << 7);
360    }
361
362    #[test]
363    fn amux_known_value_ain0() {
364        // Exact word for AIN0: amuxn=BIT(9), div-n=BIT(11), amuxp=BIT(12), div-p=BIT(23).
365        let expected = (1 << 9) | (1 << 11) | (1 << 12) | (1 << 23);
366        assert_eq!(amux_1(AdcChannel::Ain0), expected);
367    }
368
369    #[test]
370    fn delay_us_saturates() {
371        // delay_us computes a cycle count `us * 64` with a saturating multiply, so
372        // a huge µs request clamps to u32::MAX rather than wrapping.
373        assert_eq!(0u32.saturating_mul(64), 0);
374        assert_eq!(1u32.saturating_mul(64), 64);
375        assert_eq!(u32::MAX.saturating_mul(64), u32::MAX);
376    }

381#[cfg(all(test, not(target_arch = "riscv32")))]
382mod proptests {
383    use super::*;
384    use proptest::prelude::*;
385
386    proptest! {
387        /// Fuzz: sign_extend18 output is always within the signed 18-bit range and
388        /// depends only on the low 18 bits of the input (high bits are masked).
389        #[test]
390        fn sign_extend_in_range_and_masked(raw in any::<u32>()) {
391            let v = sign_extend18(raw);
392            prop_assert!((-131_072..=131_071).contains(&v));
393            // Masking the input to 18 bits gives the identical result.
394            prop_assert_eq!(v, sign_extend18(raw & 0x3FFFF));
395        }
396
397        /// Fuzz: re-encoding the sign-extended value back to 18 bits round-trips
398        /// to the masked raw (two's-complement is bijective over 18 bits).
399        #[test]
400        fn sign_extend_roundtrips(raw in any::<u32>()) {
401            let v = sign_extend18(raw);
402            let reencoded = (v as u32) & 0x3FFFF;
403            prop_assert_eq!(reencoded, raw & 0x3FFFF);
404        }
405
406        /// Fuzz: the delay_us cycle count saturates and never panics for any input.
407        #[test]
408        fn delay_cycles_never_panic(us in any::<u32>()) {
409            let cycles = us.saturating_mul(64);
410            prop_assert!(cycles <= u32::MAX);
411            // Saturation: result is either the exact product (as u64) or u32::MAX.
412            let exact = us as u64 * 64;
413            if exact <= u32::MAX as u64 {
414                prop_assert_eq!(cycles as u64, exact);
415            } else {
416                prop_assert_eq!(cycles, u32::MAX);
417            }
418        }
419    }
