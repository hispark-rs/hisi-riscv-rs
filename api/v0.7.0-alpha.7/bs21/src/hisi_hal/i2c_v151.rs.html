1//! BS2X I2C master driver — DesignWare SSI core (IP version v151).
2//!
3//! BS2X's I2C is a Synopsys/DesignWare `ic_*` controller, a COMPLETELY different
4//! IP from WS63's custom v150 I2C (`i2c.rs`). This module is the BS2X (`chip-bs21`)
5//! implementation of the `i2c` module; lib.rs picks it per chip. The register map
6//! and the master transaction sequences are from the fbb_bs2x `hal_i2c_v151`
7//! driver (sole ground truth); the v151 register block was rewritten into
8//! `bs2x-pac` (BS2X.svd) for this.
9//!
10//! API mirrors `i2c.rs`: `I2c::new_i2c0/1`, `write`, `read`, `write_read`, plus a
11//! `probe` (ACK detection) for bus scans.
12
13use crate::peripherals::{I2c0, I2c1};
14use core::marker::PhantomData;

16/// I2C bus speed.
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub enum Speed {
19    /// Standard mode (100 kHz).
20    Standard,
21    /// Fast mode (400 kHz).
22    Fast,
23}
24
25/// Error returned by the BS2X I2C master operations.
26#[derive(Debug, PartialEq, Eq)]
27#[cfg_attr(feature = "defmt", derive(defmt::Format))]
28#[non_exhaustive]
29pub enum I2cError {
30    /// A status bit never asserted within the bounded wait.
31    Timeout,
32    /// The addressed device did not ACK (DesignWare TX_ABRT / addr_7b_noack).
33    Nack,
34}
35
36// embedded-hal 1.0 portability: wire the BS2X I2C error + driver into the standard
37// traits so generic drivers work against it exactly like the WS63 i2c.rs core.
38impl embedded_hal::i2c::Error for I2cError {
39    fn kind(&self) -> embedded_hal::i2c::ErrorKind {
40        match self {
41            I2cError::Nack => {
42                embedded_hal::i2c::ErrorKind::NoAcknowledge(embedded_hal::i2c::NoAcknowledgeSource::Unknown)
43            }
44            I2cError::Timeout => embedded_hal::i2c::ErrorKind::Other,
45        }
46    }
47}
48
49impl<T> embedded_hal::i2c::ErrorType for I2c<'_, T> {
50    type Error = I2cError;
51}

53impl<T> embedded_hal::i2c::I2c for I2c<'_, T> {
54    fn transaction(&mut self, addr: u8, operations: &mut [embedded_hal::i2c::Operation<'_>]) -> Result<(), I2cError> {
55        for op in operations {
56            match op {
57                embedded_hal::i2c::Operation::Read(buf) => self.read(addr, buf)?,
58                embedded_hal::i2c::Operation::Write(data) => self.write(addr, data)?,
59            }
60        }
61        Ok(())
62    }
63}

65/// Bound on status-bit polling so a missing device/stuck bus can't hang the CPU.
66const POLL_LIMIT: u32 = 1_000_000;
67
68fn wait_until(mut ready: impl FnMut() -> bool) -> Result<(), I2cError> {
69    for _ in 0..POLL_LIMIT {
70        if ready() {
71            return Ok(());
72        }
73        core::hint::spin_loop();
74    }
75    Err(I2cError::Timeout)
76}

78/// BS2X I2C master driver over the DesignWare `ic_*` (v151) controller.
79pub struct I2c<'d, T> {
80    idx: u8,
81    _peripheral: PhantomData<&'d T>,
82}
83
84fn i2c_regs(idx: u8) -> &'static crate::soc::pac::i2c0::RegisterBlock {
85    // SAFETY: I2c0/I2c1 share one register-block layout; the pointer is a static
86    // physical MMIO base (0x5208_3000 + idx*0x1000) from bs2x-pac.
87    unsafe { &*(if idx == 0 { I2c0::ptr() } else { I2c1::ptr() }) }
88}

105/// IC_CON.speed field + (SCL high, SCL low) counts for a bus speed.
106///
107/// Pure mapping extracted from `configure` so it can be unit-tested without
108/// touching MMIO. IC_CON.speed encoding: 1=SS, 2=FS, 3=HS. The SCL counts are
109/// nominal values for a ~32 MHz I2C clock; the exact divisor only matters on
110/// silicon (the QEMU model ignores it). TODO(bs2x): derive from
111/// soc::chip::I2C_CLOCK_HZ per hal_i2c_v151's scl-count formula.
112const fn scl_params(speed: Speed) -> (u32, u32, u32) {
113    match speed {
114        Speed::Standard => (1u32, 160u32, 190u32),
115        Speed::Fast => (2u32, 40u32, 50u32),
116    }
117}
118
119/// Master init (fbb_bs2x `hal_i2c_v151_master_init`): disable, program IC_CON +
120/// the SCL counts, re-enable.
121fn configure(idx: u8, speed: Speed) {
122    let r = i2c_regs(idx);
123    let (speed_field, hcnt, lcnt) = scl_params(speed);
124    unsafe {
125        r.ic_enable().write(|w| w.enable().clear_bit());
126        r.ic_con().write(|w| {
127            w.master_mode().set_bit();
128            w.speed().bits(speed_field as u8);
129            w.ic_restart_en().set_bit();
130            w.rx_fifo_full_hld_ctrl().set_bit();
131            w.ic_slave_disable().set_bit()
132        });
133        match speed {
134            Speed::Standard => {
135                r.ic_ss_scl_hcnt().write(|w| w.ic_ss_scl_hcnt().bits(hcnt as u16));
136                r.ic_ss_scl_lcnt().write(|w| w.ic_ss_scl_lcnt().bits(lcnt as u16));
137            }
138            Speed::Fast => {
139                r.ic_fs_scl_hcnt().write(|w| w.ic_fs_scl_hcnt().bits(hcnt as u16));
140                r.ic_fs_scl_lcnt().write(|w| w.ic_fs_scl_lcnt().bits(lcnt as u16));
141            }
142        }
143        r.ic_rx_tl().write(|w| w.rx_tl().bits(0));
144        r.ic_tx_tl().write(|w| w.tx_tl().bits(0));
145        r.ic_enable().write(|w| w.enable().set_bit());
146    }
147}

149impl<T> I2c<'_, T> {
150    /// Point the controller at a 7-bit target address (disable → set IC_TAR →
151    /// enable), per `hal_i2c_cfg_taget_address`.
152    fn set_target(&self, addr: u8) {
153        let r = i2c_regs(self.idx);
154        unsafe {
155            r.ic_enable().write(|w| w.enable().clear_bit());
156            r.ic_con().modify(|_, w| w.ic_10bitaddr_master().clear_bit());
157            r.ic_tar().write(|w| {
158                w.special().clear_bit();
159                w.gc_or_start().clear_bit();
160                w.ic_tar().bits(addr as u16 & 0x3FF)
161            });
162            r.ic_enable().write(|w| w.enable().set_bit());
163            // Clear any stale interrupt/abort state (read-to-clear).
164            let _ = r.ic_clr_intr().read().bits();
165        }
166    }
167
168    /// True if the DesignWare master aborted on an address NACK (device absent).
169    fn aborted(&self) -> bool {
170        let r = i2c_regs(self.idx);
171        r.ic_raw_intr_stat().read().tx_abrt().bit_is_set()
172    }
173
174    fn clear_abort(&self) {
175        let r = i2c_regs(self.idx);
176        let _ = r.ic_clr_intr().read().bits();
177    }
178
179    /// Probe `addr`: issue a 1-byte read and report whether the device ACKed.
180    /// This is the I2C bus-scan primitive (ACK = present, NACK = absent).
181    pub fn probe(&mut self, addr: u8) -> bool {
182        let r = i2c_regs(self.idx);
183        self.set_target(addr);
184        // Push a read command with STOP (cmd=1, stop=1 → 0x300).
185        r.ic_data_cmd().write(|w| {
186            w.cmd().set_bit();
187            w.stop().set_bit()
188        });
189        // Wait for the transfer to resolve: stop-detect (ACK) or tx-abort (NACK).
190        let _ = wait_until(|| {
191            let s = r.ic_raw_intr_stat().read();
192            s.stop_det().bit_is_set() || s.tx_abrt().bit_is_set()
193        });
194        let present = !self.aborted();
195        // If it ACKed there is a byte in the RX FIFO — drain it.
196        if present && r.ic_status().read().rfne().bit_is_set() {
197            let _ = r.ic_data_cmd().read().dat().bits();
198        }
199        self.clear_abort();
200        present
201    }
202
203    /// Write `data` to `addr` (STOP after the last byte).
204    pub fn write(&mut self, addr: u8, data: &[u8]) -> Result<(), I2cError> {
205        let r = i2c_regs(self.idx);
206        self.set_target(addr);
207        for (i, &b) in data.iter().enumerate() {
208            let last = i + 1 == data.len();
209            wait_until(|| r.ic_status().read().tfnf().bit_is_set())?;
210            unsafe {
211                r.ic_data_cmd().write(|w| {
212                    w.dat().bits(b);
213                    if last {
214                        w.stop().set_bit();
215                    }
216                    w
217                });
218            }
219            if self.aborted() {
220                self.clear_abort();
221                return Err(I2cError::Nack);
222            }
223        }
224        wait_until(|| r.ic_raw_intr_stat().read().stop_det().bit_is_set())?;
225        Ok(())
226    }
227
228    /// Read `buf.len()` bytes from `addr` (STOP after the last byte).
229    pub fn read(&mut self, addr: u8, buf: &mut [u8]) -> Result<(), I2cError> {
230        let r = i2c_regs(self.idx);
231        self.set_target(addr);
232        let n = buf.len();
233        for (i, slot) in buf.iter_mut().enumerate() {
234            let last = i + 1 == n;
235            wait_until(|| r.ic_status().read().tfnf().bit_is_set())?;
236            r.ic_data_cmd().write(|w| {
237                w.cmd().set_bit();
238                if last {
239                    w.stop().set_bit();
240                }
241                w
242            });
243            wait_until(|| r.ic_status().read().rfne().bit_is_set())?;
244            if self.aborted() {
245                self.clear_abort();
246                return Err(I2cError::Nack);
247            }
248            *slot = r.ic_data_cmd().read().dat().bits();
249        }
250        Ok(())
251    }
252
253    /// Write then read (repeated START) — `wr` out, then `rd` in.
254    pub fn write_read(&mut self, addr: u8, wr: &[u8], rd: &mut [u8]) -> Result<(), I2cError> {
255        if !wr.is_empty() {
256            self.write(addr, wr)?;
257        }
258        if !rd.is_empty() {
259            self.read(addr, rd)?;
260        }
261        Ok(())
262    }
263}

267#[cfg(all(test, not(target_arch = "riscv32")))]
268mod tests {
269    use super::*;
270
271    #[test]
272    fn standard_scl_params_exact() {
273        // Standard mode → IC_CON.speed=1 (SS) with the SS hcnt/lcnt pair.
274        assert_eq!(scl_params(Speed::Standard), (1, 160, 190));
275    }
276
277    #[test]
278    fn fast_scl_params_exact() {
279        // Fast mode → IC_CON.speed=2 (FS) with the FS hcnt/lcnt pair.
280        assert_eq!(scl_params(Speed::Fast), (2, 40, 50));
281    }
282
283    #[test]
284    fn scl_speed_field_fits_byte() {
285        // configure() writes speed_field with `.bits(speed_field as u8)`, so the
286        // IC_CON.speed field must fit a u8 for every Speed.
287        for s in [Speed::Standard, Speed::Fast] {
288            let (field, _, _) = scl_params(s);
289            assert!(field <= u8::MAX as u32);
290            assert!(field != 0, "0 is the reserved/illegal IC_CON.speed value");
291        }
292    }
293
294    #[test]
295    fn scl_counts_fit_u16() {
296        // The SCL counts are written with `.bits(_ as u16)`; both must fit u16.
297        for s in [Speed::Standard, Speed::Fast] {
298            let (_, hcnt, lcnt) = scl_params(s);
299            assert!(hcnt <= u16::MAX as u32);
300            assert!(lcnt <= u16::MAX as u32);
301        }
302    }
303
304    #[test]
305    fn fast_is_shorter_than_standard() {
306        // Fast mode (400 kHz) halves the SCL period vs Standard (100 kHz), so its
307        // high/low counts must both be strictly smaller for the same clock.
308        let (_, sh, sl) = scl_params(Speed::Standard);
309        let (_, fh, fl) = scl_params(Speed::Fast);
310        assert!(fh < sh);
311        assert!(fl < sl);
312    }
313
314    #[test]
315    fn scl_low_exceeds_high() {
316        // I2C SCL low time is longer than high time in both modes (the asymmetry
317        // the DesignWare SS/FS count pairs encode).
318        for s in [Speed::Standard, Speed::Fast] {
319            let (_, hcnt, lcnt) = scl_params(s);
320            assert!(lcnt > hcnt);
321        }
322    }
323
324    #[test]
325    fn target_addr_mask_keeps_7bit() {
326        // set_target masks with `& 0x3FF`; any valid 7-bit address passes through
327        // unchanged (the top bits are only relevant for 10-bit addressing).
328        for addr in 0u8..=0x7F {
329            assert_eq!(addr as u16 & 0x3FF, addr as u16);
330        }
331    }
332
333    #[test]
334    fn target_addr_mask_drops_nothing_for_u8() {
335        // A u8 address can never exceed 0x3FF, so the mask is a no-op for the whole
336        // u8 range — confirms the cast widens before masking (no truncation).
337        for addr in 0u8..=0xFF {
338            assert_eq!(addr as u16 & 0x3FF, addr as u16);
339        }
340    }
341
342    #[test]
343    fn wait_until_returns_ok_when_ready() {
344        // A condition that is already true resolves immediately with Ok.
345        assert_eq!(wait_until(|| true), Ok(()));
346    }
347
348    #[test]
349    fn wait_until_times_out_when_never_ready() {
350        // A condition that never asserts exhausts POLL_LIMIT and reports Timeout.
351        assert_eq!(wait_until(|| false), Err(I2cError::Timeout));
352    }
353
354    #[test]
355    fn wait_until_polls_at_most_poll_limit() {
356        // The poll loop is bounded: it calls the predicate no more than POLL_LIMIT
357        // times before giving up (so a stuck bus can't hang the CPU).
358        let mut calls: u32 = 0;
359        let r = wait_until(|| {
360            calls += 1;
361            false
362        });
363        assert_eq!(r, Err(I2cError::Timeout));
364        assert_eq!(calls, POLL_LIMIT);
365    }
366
367    #[test]
368    fn wait_until_stops_at_first_ready() {
369        // Once the predicate is true the loop returns; later polls don't happen.
370        let mut calls: u32 = 0;
371        let r = wait_until(|| {
372            calls += 1;
373            calls >= 3
374        });
375        assert_eq!(r, Ok(()));
376        assert_eq!(calls, 3);
377    }
378
379    #[test]
380    fn i2c_error_variants_distinct() {
381        // Timeout and Nack are distinct error values (the two failure modes the
382        // public API surfaces via PartialEq).
383        assert_ne!(I2cError::Timeout, I2cError::Nack);
384        assert_eq!(I2cError::Nack, I2cError::Nack);
385    }
386
387    #[test]
388    fn speed_is_copy_and_eq() {
389        // Speed derives Copy + Eq; mapping the same value twice is identical.
390        let s = Speed::Fast;
391        let t = s; // Copy, not move
392        assert_eq!(s, t);
393        assert_eq!(scl_params(s), scl_params(t));
394    }

400mod proptests {
401    use super::{Speed, scl_params};
402    use proptest::prelude::*;

404    /// Pick a `Speed` from a bool so proptest can drive the enum-keyed mapping.
405    fn speed_of(fast: bool) -> Speed {
406        if fast { Speed::Fast } else { Speed::Standard }
407    }
408
409    proptest! {
410        /// Fuzz: `scl_params` never panics for either speed (pure mapping).
411        #[test]
412        fn scl_params_never_panics(fast in any::<bool>()) {
413            let _ = scl_params(speed_of(fast));
414        }
415
416        /// Fuzz: the IC_CON.speed field survives the `as u8` cast in `configure`
417        /// without truncation — i.e. it fits a u8 (round-trips through u8) and is
418        /// never the reserved/illegal value 0. This guards the narrowing cast
419        /// `w.speed().bits(speed_field as u8)`.
420        #[test]
421        fn speed_field_fits_u8_and_nonzero(fast in any::<bool>()) {
422            let (field, _, _) = scl_params(speed_of(fast));
423            // round-trip the actual cast `configure` performs.
424            prop_assert_eq!((field as u8) as u32, field);
425            prop_assert!(field != 0);
426        }
427
428        /// Fuzz: both SCL counts survive the `as u16` cast in `configure` without
429        /// truncation (they round-trip through u16). This guards the narrowing
430        /// casts `w.cnt().bits(hcnt as u16)` / `(lcnt as u16)`.
431        #[test]
432        fn scl_counts_fit_u16(fast in any::<bool>()) {
433            let (_, hcnt, lcnt) = scl_params(speed_of(fast));
434            prop_assert_eq!((hcnt as u16) as u32, hcnt);
435            prop_assert_eq!((lcnt as u16) as u32, lcnt);
436        }
437
438        /// Fuzz: in every mode the SCL low time exceeds the high time (the
439        /// asymmetry the DesignWare SS/FS count pairs must encode).
440        #[test]
441        fn scl_low_exceeds_high(fast in any::<bool>()) {
442            let (_, hcnt, lcnt) = scl_params(speed_of(fast));
443            prop_assert!(lcnt > hcnt);
444        }
445
446        /// Fuzz: the target-address mask `addr as u16 & 0x3FF` (from `set_target`)
447        /// is a no-op for ANY u8 address — the `as u16` widens BEFORE the mask, so
448        /// nothing is ever truncated and the result equals the input. This is the
449        /// "mask applied to a too-narrow / pre-cast value" bug shape; a u8 can
450        /// never exceed the 10-bit field, so the masked value must round-trip.
451        #[test]
452        fn target_addr_mask_roundtrips_for_u8(addr in any::<u8>()) {
453            let masked = addr as u16 & 0x3FF;
454            prop_assert_eq!(masked, addr as u16);
455        }
456
457        /// Fuzz: the masked target address never sets a bit outside the 10-bit
458        /// IC_TAR field (bits above bit 9 are always clear).
459        #[test]
460        fn target_addr_mask_clears_high_bits(addr in any::<u8>()) {
461            let masked = addr as u16 & 0x3FF;
462            prop_assert_eq!(masked & !0x3FF, 0);
463        }
464    }
