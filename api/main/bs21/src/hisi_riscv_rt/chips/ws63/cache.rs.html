1//! WS63 cache setup required by the vendor application runtime.
2//!
3//! Flashboot leaves the cache geometry selected but the caches disabled. The
4//! WS63 ROM contains 48-bit `l.li` instructions; executing that ROM before the
5//! application runtime enables and invalidates the instruction cache can split
6//! the final instruction parcel into a separate load. Keep this sequence in
7//! lockstep with the vendor `cpu_cache_init()` implementation.
8
9use core::arch::asm;
10
11const CACHE_INVALIDATE_ALL: usize = 1 << 2;
12const ICACHE_32_KIB_ENABLE: usize = 6 | 1;
13const ICACHE_PREFETCH_ONE_LINE: usize = 1;
14const DCACHE_4_KIB_ENABLE: usize = 1;
15
16/// Invalidate and enable the WS63 instruction and data caches.
17///
18/// # Safety
19///
20/// Must run exactly once during early WS63 startup, before application data is
21/// relocated or vendor ROM code is called.
22#[unsafe(no_mangle)]
23#[unsafe(link_section = ".text.runtime.init")]
24pub(super) unsafe extern "C" fn __hisi_ws63_cache_init() {
25    unsafe {
26        asm!(
27            "csrw 0x7c2, {invalidate_all}",
28            "fence",
29            "csrw 0x7c3, {invalidate_all}",
30            "fence",
31            "csrw 0x7c0, {icache_ctl}",
32            "csrw 0x7c6, {icache_prefetch}",
33            "csrw 0x7c1, {dcache_ctl}",
34            invalidate_all = in(reg) CACHE_INVALIDATE_ALL,
35            icache_ctl = in(reg) ICACHE_32_KIB_ENABLE,
36            icache_prefetch = in(reg) ICACHE_PREFETCH_ONE_LINE,
37            dcache_ctl = in(reg) DCACHE_4_KIB_ENABLE,
38            options(nostack),
39        );
40    }
41}
