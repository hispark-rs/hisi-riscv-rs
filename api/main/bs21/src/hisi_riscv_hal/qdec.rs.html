1//! BS2X QDEC — quadrature decoder (IP v150).
2//!
3//! BS2X-only (`chip-bs21`); no WS63 analogue. An nRF-style quadrature decoder: it
4//! samples a 2-bit A/B gray code and accumulates SIGNED motion into ACC. The
5//! "count" is the signed `qdec_acc_data` value (negative = reverse). Register map
6//! from fbb_bs2x `hal_qdec_v150`; bs2x-pac's `qdec` block is a faithful match (the
7//! ACC field is unsigned in the PAC, so this driver sign-extends).
8
9use crate::peripherals::Qdec as QdecPeriph;
10use core::marker::PhantomData;
11
12/// BS2X quadrature decoder driver: samples the A/B gray code and accumulates signed motion into ACC.
13pub struct Qdec<'d> {
14    _q: PhantomData<QdecPeriph<'d>>,
15}
16
17impl<'d> Qdec<'d> {
18    fn regs(&self) -> &'static crate::soc::pac::qdec::RegisterBlock {
19        // SAFETY: static physical MMIO base (0x5200_0200) from bs2x-pac.
20        unsafe { &*QdecPeriph::ptr() }
21    }
22
23    /// Enable the decoder and begin sampling/accumulation (`hal_qdec_v150_enable`
24    /// + `QDEC_TASK_START`).
25    pub fn new(_q: QdecPeriph<'d>) -> Self {
26        let this = Self { _q: PhantomData };
27        let r = this.regs();
28        r.qdec_enable().write(|w| w.en().set_bit());
29        r.qdec_task_start().write(|w| w.start().set_bit());
30        this
31    }
32
33    /// Read the live signed accumulated quadrature count (`qdec_acc_data`,
34    /// sign-extended from 16 bits). Negative = reverse rotation.
35    pub fn read_count(&self) -> i16 {
36        self.regs().qdec_acc_data().read().acc_rd_val().bits() as i16
37    }
38
39    /// Read the double/invalid-transition error count (`qdec_accdbl_data`).
40    pub fn read_error_count(&self) -> u8 {
41        self.regs().qdec_accdbl_data().read().dbl_rd_val().bits()
42    }
43
44    /// Stop sampling (counter state retained).
45    pub fn stop(&self) {
46        self.regs().qdec_task_stop().write(|w| w.stop().set_bit());
47    }
48}
