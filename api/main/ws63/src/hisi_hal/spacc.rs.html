1//! SPACC (Security Accelerator) driver for WS63.
2//!
3//! The SPACC peripheral provides hardware acceleration for:
4//! - AES (128/192/256)
5//! - SM4 (Chinese national standard)
6//! - LEA (Lightweight Encryption Algorithm)
7//! - TDES (Triple DES)
8//! - HASH (SHA-256, SM3)
9//! - HMAC
10
11use crate::peripherals::Spacc;
12
13/// SPACC driver.
14pub struct SpaccDriver<'d> {
15    _spacc: Spacc<'d>,
16}
17
18/// Cipher algorithm selection.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum CipherAlg {
21    /// AES with a 128-bit key.
22    Aes128,
23    /// AES with a 192-bit key.
24    Aes192,
25    /// AES with a 256-bit key.
26    Aes256,
27    /// SM4 block cipher (Chinese national standard).
28    Sm4,
29    /// LEA (Lightweight Encryption Algorithm).
30    Lea,
31    /// Triple DES (TDES).
32    Tdes,
33}
34
35/// Cipher operation mode.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub enum CipherMode {
38    /// Electronic Codebook (ECB) mode.
39    Ecb,
40    /// Cipher Block Chaining (CBC) mode.
41    Cbc,
42    /// Counter (CTR) mode.
43    Ctr,
44    /// Counter with CBC-MAC (CCM) authenticated mode.
45    Ccm,
46    /// Galois/Counter Mode (GCM) authenticated mode.
47    Gcm,
48}
49
50/// Cipher direction.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum CipherDir {
53    /// Encrypt the input data.
54    Encrypt,
55    /// Decrypt the input data.
56    Decrypt,
57}
58
59/// Hash algorithm selection.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub enum HashAlg {
62    /// SHA-256 hash algorithm.
63    Sha256,
64    /// SM3 hash algorithm (Chinese national standard).
65    Sm3,
66}
