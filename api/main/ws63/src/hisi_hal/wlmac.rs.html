1//! Read-only WS63 Wi-Fi MAC diagnostics.
2//!
3//! The RF stack owns the WLMAC lifecycle. This module therefore does not expose
4//! a safe constructor or add WLMAC to [`crate::Peripherals`]. It only provides a
5//! bounded snapshot for diagnostics after the caller has established that the
6//! radio block is powered and initialized.
7
8use core::marker::PhantomData;
9
10use crate::soc::pac;
11
12/// A snapshot of the receive counters used by the WS63 mask-ROM helper.
13#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
14pub struct WlmacRxCounters {
15    /// Received aggregate MPDU count.
16    pub rx_ampdu: u16,
17    /// MPDUs with a valid FCS received inside an aggregate.
18    pub rx_success_mpdu_in_ampdu: u32,
19    /// MPDUs with an invalid FCS received inside an aggregate.
20    pub rx_failed_mpdu_in_ampdu: u32,
21    /// Received non-aggregate MPDUs with a valid FCS.
22    pub rx_success_mpdu: u32,
23    /// Received non-aggregate MPDUs with an invalid FCS.
24    pub rx_failed_mpdu: u32,
25    /// MPDUs rejected by the MAC receive filter.
26    pub rx_filtered_mpdu: u16,
27}
28
29/// Read-only access to initialized WS63 WLMAC diagnostic counters.
30///
31/// This capability is neither `Send` nor `Sync`. Constructing another instance
32/// remains possible through the unsafe constructor because the RF integration,
33/// rather than the HAL peripheral singleton, owns this register block.
34pub struct WlmacDiagnostics {
35    _not_send_sync: PhantomData<*mut ()>,
36}
37
38impl WlmacDiagnostics {
39    /// Assumes that the RF integration has initialized and retained WLMAC.
40    ///
41    /// # Safety
42    ///
43    /// The caller must ensure the WLMAC register block is powered, clocked, and
44    /// not concurrently reset or powered down for the lifetime of this value.
45    /// The caller must also ensure that reading these counters is permitted by
46    /// the active RF firmware profile.
47    #[inline]
48    pub const unsafe fn assume_radio_ready() -> Self {
49        Self {
50            _not_send_sync: PhantomData,
51        }
52    }
53
54    /// Reads the six WLMAC receive counters used by the mask-ROM statistics API.
55    ///
56    /// Interrupts are masked only for the bounded register snapshot, matching
57    /// the mask-ROM helper's software-coherence behavior. Hardware can continue
58    /// incrementing counters while the snapshot is read, so the fields are not
59    /// an atomic hardware-time sample.
60    #[inline]
61    pub fn snapshot_rx(&self) -> WlmacRxCounters {
62        critical_section::with(|_| {
63            // SAFETY: construction requires the caller to keep WLMAC accessible.
64            let regs = unsafe { &*pac::WlmacStats::ptr() };
65
66            WlmacRxCounters {
67                rx_ampdu: regs.rx_ampdu_count().read().count().bits(),
68                rx_success_mpdu_in_ampdu: regs
69                    .rx_success_mpdu_in_ampdu_count()
70                    .read()
71                    .count()
72                    .bits(),
73                rx_failed_mpdu_in_ampdu: regs
74                    .rx_failed_mpdu_in_ampdu_count()
75                    .read()
76                    .count()
77                    .bits(),
78                rx_success_mpdu: regs.rx_success_mpdu_count().read().count().bits(),
79                rx_failed_mpdu: regs.rx_failed_mpdu_count().read().count().bits(),
80                rx_filtered_mpdu: regs.rx_filtered_mpdu_count().read().count().bits(),
81            }
82        })
83    }
84}
