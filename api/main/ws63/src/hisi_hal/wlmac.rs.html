1//! Read-only WS63 Wi-Fi MAC diagnostics.
2//!
3//! The RF stack owns the WLMAC lifecycle. This module therefore does not expose
4//! a safe constructor or add WLMAC to [`crate::Peripherals`]. It only provides a
5//! bounded snapshot for diagnostics after the caller has established that the
6//! radio block is powered and initialized.
7
8use core::marker::PhantomData;
9
10use crate::soc::pac;
11
12/// A snapshot of the receive counters used by the WS63 mask-ROM helper.
13#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
14pub struct WlmacRxCounters {
15    /// Received aggregate MPDU count.
16    pub rx_ampdu: u16,
17    /// MPDUs with a valid FCS received inside an aggregate.
18    pub rx_success_mpdu_in_ampdu: u32,
19    /// MPDUs with an invalid FCS received inside an aggregate.
20    pub rx_failed_mpdu_in_ampdu: u32,
21    /// Received non-aggregate MPDUs with a valid FCS.
22    pub rx_success_mpdu: u32,
23    /// Received non-aggregate MPDUs with an invalid FCS.
24    pub rx_failed_mpdu: u32,
25    /// MPDUs rejected by the MAC receive filter.
26    pub rx_filtered_mpdu: u16,
27}

29/// A snapshot of receive-security failures used by the WS63 mask-ROM helper.
30#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
31pub struct WlmacRxSecurityCounters {
32    /// CCMP replay failures.
33    pub ccmp_replay_failures: u16,
34    /// TKIP replay failures.
35    pub tkip_replay_failures: u16,
36    /// CCMP MIC failures.
37    pub ccmp_mic_failures: u16,
38    /// TKIP MIC failures.
39    pub tkip_mic_failures: u16,
40    /// Receive key-search failures.
41    pub key_search_failures: u16,
42}

44/// A read-only snapshot of the active VAP0 receive-filter identity.
45#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
46pub struct WlmacFilterState {
47    /// Packed receive-filter control state currently programmed into WLMAC.
48    pub rx_filter_control: u32,
49    /// Station address programmed for hardware VAP0.
50    pub station_address: [u8; 6],
51    /// BSSID programmed for hardware VAP0.
52    pub bssid: [u8; 6],
53}
54
55/// Read-only access to initialized WS63 WLMAC diagnostic counters.
56///
57/// This capability is neither `Send` nor `Sync`. Constructing another instance
58/// remains possible through the unsafe constructor because the RF integration,
59/// rather than the HAL peripheral singleton, owns this register block.
60pub struct WlmacDiagnostics {
61    _not_send_sync: PhantomData<*mut ()>,
62}
63
64impl WlmacDiagnostics {
65    /// Assumes that the RF integration has initialized and retained WLMAC.
66    ///
67    /// # Safety
68    ///
69    /// The caller must ensure the WLMAC register block is powered, clocked, and
70    /// not concurrently reset or powered down for the lifetime of this value.
71    /// The caller must also ensure that reading these counters is permitted by
72    /// the active RF firmware profile.
73    #[inline]
74    pub const unsafe fn assume_radio_ready() -> Self {
75        Self {
76            _not_send_sync: PhantomData,
77        }
78    }
79
80    /// Reads the six WLMAC receive counters used by the mask-ROM statistics API.
81    ///
82    /// Interrupts are masked only for the bounded register snapshot, matching
83    /// the mask-ROM helper's software-coherence behavior. Hardware can continue
84    /// incrementing counters while the snapshot is read, so the fields are not
85    /// an atomic hardware-time sample.
86    #[inline]
87    pub fn snapshot_rx(&self) -> WlmacRxCounters {
88        critical_section::with(|_| {
89            // SAFETY: construction requires the caller to keep WLMAC accessible.
90            let regs = unsafe { &*pac::WlmacStats::ptr() };
91
92            WlmacRxCounters {
93                rx_ampdu: regs.rx_ampdu_count().read().count().bits(),
94                rx_success_mpdu_in_ampdu: regs
95                    .rx_success_mpdu_in_ampdu_count()
96                    .read()
97                    .count()
98                    .bits(),
99                rx_failed_mpdu_in_ampdu: regs
100                    .rx_failed_mpdu_in_ampdu_count()
101                    .read()
102                    .count()
103                    .bits(),
104                rx_success_mpdu: regs.rx_success_mpdu_count().read().count().bits(),
105                rx_failed_mpdu: regs.rx_failed_mpdu_count().read().count().bits(),
106                rx_filtered_mpdu: regs.rx_filtered_mpdu_count().read().count().bits(),
107            }
108        })
109    }
110
111    /// Reads the WLMAC receive replay, MIC, and key-search failure counters.
112    ///
113    /// These registers are the three security-counter inputs consumed by the
114    /// mask-ROM `hh503_get_mac_statistics_data` helper. As with
115    /// [`Self::snapshot_rx`], hardware can increment them during the bounded
116    /// snapshot.
117    #[inline]
118    pub fn snapshot_rx_security(&self) -> WlmacRxSecurityCounters {
119        critical_section::with(|_| {
120            // SAFETY: construction requires the caller to keep WLMAC accessible.
121            let regs = unsafe { &*pac::WlmacStats::ptr() };
122            let replay = regs.rx_replay_failure_count().read();
123            let mic = regs.rx_mic_failure_count().read();
124
125            WlmacRxSecurityCounters {
126                ccmp_replay_failures: replay.ccmp().bits(),
127                tkip_replay_failures: replay.tkip().bits(),
128                ccmp_mic_failures: mic.ccmp().bits(),
129                tkip_mic_failures: mic.tkip().bits(),
130                key_search_failures: regs
131                    .rx_key_search_failure_count()
132                    .read()
133                    .count()
134                    .bits(),
135            }
136        })
137    }
138
139    /// Reads the active receive-filter command and VAP0 address identity.
140    ///
141    /// Interrupts are masked only while taking the four-register snapshot. The
142    /// vendor WLMAC state machine owns these registers, so callers must treat
143    /// the result as diagnostic evidence rather than a configuration surface.
144    #[inline]
145    pub fn snapshot_filter_state(&self) -> WlmacFilterState {
146        critical_section::with(|_| {
147            // SAFETY: construction requires the caller to keep WLMAC accessible.
148            let regs = unsafe { &*pac::WlmacStats::ptr() };
149            let heads = regs.vap0_address_heads().read();
150
151            WlmacFilterState {
152                rx_filter_control: regs.rx_filter_control().read().control().bits(),
153                station_address: decode_address(
154                    heads.station().bits(),
155                    regs.vap0_station_address_tail().read().bytes().bits(),
156                ),
157                bssid: decode_address(
158                    heads.bssid().bits(),
159                    regs.vap0_bssid_tail().read().bytes().bits(),
160                ),
161            }
162        })
163    }
164}
165
166#[inline]
167fn decode_address(head: u16, tail: u32) -> [u8; 6] {
168    let head = head.to_be_bytes();
169    let tail = tail.to_be_bytes();
170    [head[0], head[1], tail[0], tail[1], tail[2], tail[3]]
171}
