1//! I2S (Inter-IC Sound) / PCM audio interface driver for WS63.
2//!
3//! The WS63 I2S peripheral (vendor "SIO v151") supports both I2S and PCM modes,
4//! master and slave operation, 2/4/8/16 channels, 16/18/20/24/32-bit samples, and
5//! separate TX/RX FIFOs with threshold interrupts.
6//!
7//! # Typed config — "if it compiles, it runs on silicon"
8//!
9//! The role is encoded in the **type** ([`Master`] / [`Slave`]) via two fused
10//! constructors — [`I2sDriver::new_master`] / [`I2sDriver::new_slave`] — so there
11//! is no separate `configure()` step that could be skipped or mis-ordered. A
12//! **master generates BCLK and FS**, so its clock dividers are *derived* from the
13//! `(data_width, channels)` pair exactly as the vendor `sio_porting` does, never
14//! supplied raw:
15//!
16//! * `FS_DIV_NUM      = data_width · channels`
17//! * `FS_DIV_RATIO    = data_width · channels / 2`  (50 % duty)
18//! * `BCLK_DIV_NUM    = round(I2S_MCLK / (32 · data_width · channels))`
19//!
20//! Because the dividers are derived from non-zero enums, a **zero-divider master is
21//! unrepresentable** and every divider is provably in field range (see the
22//! exhaustive `derived_dividers_in_range` host test) — there is no silent `& mask`
23//! truncation of a user-supplied value. A **slave** receives BCLK/FS externally, so
24//! [`SlaveConfig`] carries no dividers at all. `new_master` also self-enables the
25//! I2S clock tree (CMU + CKEN bus/clk gates), the class-C "construct → clocked"
26//! guarantee.
27//!
28//! Register bit layouts follow the authoritative vendor `hal_sio_v151_regs_def.h`
29//! (the ws63-pac SVD encodes a fabricated textbook layout for `mode`/`i2s_crg`/
30//! `data_width_set`; the driver writes the correct bits via raw `.bits()`), and the
31//! [`DataWidth`]/[`ChannelCount`] enums are tabled to the silicon codes (16/18/20/
32//! 24/32-bit; 2/4/8/16-channel — there is no 8/10/12/14-bit or 6-channel mode).
33//!
34//! On the current HIL board no audio codec is wired, so only register read-back and
35//! the IP `version` are validated on silicon; the master audio waveform itself is
36//! not codec-verified.
37

51/// Clock edge selection (`mode` register bit 6). Meaningful for a slave; a master
52/// drives the vendor-default falling edge.
53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54#[cfg_attr(feature = "defmt", derive(defmt::Format))]
55#[instability::unstable]
56pub enum ClockEdge {
57    /// Rising edge.
58    Rising,
59    /// Falling edge.
60    Falling,
61}

63/// Number of audio channels (`mode.chn_num`, bits [5:4]).
64///
65/// The field codes 0..3 map to **2/4/8/16** channels — there is no 6-channel mode
66/// (the previous `Six` table entry was wrong).
67#[derive(Debug, Clone, Copy, PartialEq, Eq)]
68#[cfg_attr(feature = "defmt", derive(defmt::Format))]
69pub enum ChannelCount {
70    /// 2 channels (stereo).
71    Two = 0,
72    /// 4 channels.
73    Four = 1,
74    /// 8 channels.
75    Eight = 2,
76    /// 16 channels.
77    Sixteen = 3,
78}
79
80impl ChannelCount {
81    /// The actual channel count (2/4/8/16), used to derive the master dividers.
82    pub const fn count(self) -> u32 {
83        match self {
84            ChannelCount::Two => 2,
85            ChannelCount::Four => 4,
86            ChannelCount::Eight => 8,
87            ChannelCount::Sixteen => 16,
88        }
89    }
90}

92/// TX/RX sample width (`data_width_set.tx_mode`/`rx_mode`, 3-bit fields).
93///
94/// The field codes are the silicon enum: `1=16, 2=18, 3=20, 4=24, 5=32` bits (code
95/// 0 is reserved). The previous `Bits8..Bits24 = 0..7` table did not exist on the
96/// hardware.
97#[derive(Debug, Clone, Copy, PartialEq, Eq)]
98#[cfg_attr(feature = "defmt", derive(defmt::Format))]
99pub enum DataWidth {
100    /// 16-bit samples.
101    Bits16 = 1,
102    /// 18-bit samples.
103    Bits18 = 2,
104    /// 20-bit samples.
105    Bits20 = 3,
106    /// 24-bit samples.
107    Bits24 = 4,
108    /// 32-bit samples.
109    Bits32 = 5,
110}
111
112impl DataWidth {
113    /// The sample width in bits (16/18/20/24/32), used to derive the master dividers.
114    pub const fn bits(self) -> u32 {
115        match self {
116            DataWidth::Bits16 => 16,
117            DataWidth::Bits18 => 18,
118            DataWidth::Bits20 => 20,
119            DataWidth::Bits24 => 24,
120            DataWidth::Bits32 => 32,
121        }
122    }
123}

125// ── Master clock-divider derivation (vendor `sio_porting`) ──────────────────────
126//
127// I2S_MCLK_RATE = 12288 (kHz units), FREQ_OF_NEED = 32, I2S_DUTY_CYCLE = 2.
128// All three dividers are derived from (data_width · channels), so a master never
129// takes a raw divider and can never present a zero/out-of-range one.
130
131/// `I2S_MCLK_RATE` in the same kHz units the vendor uses for the BCLK divider.
132const I2S_MCLK: u32 = 12288;
133/// `FREQ_OF_NEED` — the BCLK derivation's per-sample reference factor.
134const FREQ_OF_NEED: u32 = 32;
135
136/// The three derived master dividers `(bclk_div_num, fs_div_num, fs_div_ratio_num)`
137/// for a `(data_width, channels)` pair, matching the vendor derivation exactly:
138///   fs_div_num   = dw·ch                    (≤ 512, fits the 10-bit field)
139///   fs_div_ratio = dw·ch / 2  (50 % duty)   (≤ 256, fits the 11-bit field)
140///   bclk_div_num = round(MCLK / (32·dw·ch)) (∈ [1, 12], fits the 7-bit field)
141/// `round` is round-half-up: `(2·N + D) / (2·D)` integer division.
142const fn derive_dividers(dw: DataWidth, ch: ChannelCount) -> (u32, u32, u32) {
143    let prod = dw.bits() * ch.count(); // dw·ch, always even and ≥ 32
144    let fs_div_num = prod;
145    let fs_div_ratio = prod / 2;
146    let d = FREQ_OF_NEED * prod; // denominator
147    // round-half-up(MCLK / d) = (2·MCLK + d) / (2·d); never 0 for these inputs.
148    let bclk = (2 * I2S_MCLK + d) / (2 * d);
149    let bclk = if bclk == 0 { 1 } else { bclk };
150    (bclk, fs_div_num, fs_div_ratio)
151}

236/// I2S audio interface driver, parameterised by clock role ([`Master`] / [`Slave`]).
237pub struct I2sDriver<'d, R> {
238    _i2s: I2s<'d>,
239    _role: PhantomData<R>,
240}
241
242/// Build the `mode` register value per the vendor `hal_sio_v151_regs_def.h`:
243///   bit0 mode (0=i2s,1=pcm) · bits[5:4] chn_num · bit6 clk_edge · bit7 ms_mode_sel.
244const fn mode_bits(mode: I2sMode, channels: ChannelCount, edge: ClockEdge, master: bool) -> u32 {
245    let mut m = 0u32;
246    if matches!(mode, I2sMode::Pcm) {
247        m |= 1 << 0;
248    }
249    m |= ((channels as u32) & 0x3) << 4;
250    if matches!(edge, ClockEdge::Falling) {
251        m |= 1 << 6;
252    }
253    if master {
254        m |= 1 << 7;
255    }
256    m
257}
258
259/// Build the `data_width_set` value: `tx_mode` bits[2:0], `rx_mode` bits[5:3], both
260/// set to the same width (matching the vendor `hal_sio_v151_data_width_set`).
261const fn data_width_bits(dw: DataWidth) -> u32 {
262    let code = (dw as u32) & 0x7;
263    code | (code << 3)
264}
265
266impl<'d> I2sDriver<'d, Master> {
267    /// Create and fully configure an I2S **master**. Self-enables the I2S clock
268    /// tree, programs the framing/width/FIFO registers, and derives + programs the
269    /// BCLK/FS dividers from `(data_width, channels)`.
270    pub fn new_master(i2s: I2s<'d>, config: &MasterConfig) -> Self {
271        enable_i2s_clock();
272        let driver = Self { _i2s: i2s, _role: PhantomData };
273        let r = driver.regs();
274
275        // mode: master, vendor-default falling edge.
276        let mode = mode_bits(config.mode, config.channels, ClockEdge::Falling, true);
277        // Derived dividers (provably in field range for every enum combination).
278        let (bclk, fs_num, fs_ratio) = derive_dividers(config.data_width, config.channels);
279        unsafe {
280            r.mode().write(|w| w.bits(mode));
281            r.i2s_fs_div_num().write(|w| w.bits(fs_num));
282            r.i2s_fs_div_ratio_num().write(|w| w.bits(fs_ratio));
283            r.i2s_bclk_div_num().write(|w| w.bits(bclk));
284            // i2s_crg: bclk_div_en (bit0) + crg_clken (bit1); phase bits 0.
285            r.i2s_crg().write(|w| w.bits(0b11));
286        }
287        driver.apply_common(config.data_width, config.tx_fifo_threshold, config.rx_fifo_threshold, config.loopback);
288        driver
289    }
290}

292impl<'d> I2sDriver<'d, Slave> {
293    /// Create and fully configure an I2S **slave** (BCLK/FS supplied externally;
294    /// no dividers). Self-enables the bus clock so the FIFOs/registers are reachable.
295    #[instability::unstable]
296    pub fn new_slave(i2s: I2s<'d>, config: &SlaveConfig) -> Self {
297        enable_i2s_clock();
298        let driver = Self { _i2s: i2s, _role: PhantomData };
299        let r = driver.regs();
300
301        let mode = mode_bits(config.mode, config.channels, config.clock_edge, false);
302        unsafe {
303            r.mode().write(|w| w.bits(mode));
304            // Slave does not generate clocks: leave crg dividers off (phase bits 0).
305            r.i2s_crg().write(|w| w.bits(0));
306        }
307        driver.apply_common(config.data_width, config.tx_fifo_threshold, config.rx_fifo_threshold, config.loopback);
308        driver
309    }
310}

312impl<'d, R: sealed::Role> I2sDriver<'d, R> {
313    fn regs(&self) -> &'static crate::soc::pac::i2s::RegisterBlock {
314        // SAFETY: PAC peripheral pointer is a static physical MMIO address, always valid
315        unsafe { &*I2s::ptr() }
316    }
317
318    /// Shared tail of `new_master`/`new_slave`: data width, FIFO thresholds, sign
319    /// extension, loopback, and FIFO reset — the role-independent register writes.
320    fn apply_common(&self, data_width: DataWidth, tx_thresh: u8, rx_thresh: u8, loopback: bool) {
321        let r = self.regs();
322        unsafe {
323            r.data_width_set().write(|w| w.bits(data_width_bits(data_width)));
324            let thresh = (tx_thresh as u32) | ((rx_thresh as u32) << 8);
325            r.fifo_threshold().write(|w| w.bits(thresh));
326            // version.loop (bit 8): internal TX→RX loopback.
327            r.version().write(|w| w.bits(if loopback { 1 << 8 } else { 0 }));
328            // Signed sample extension for correct audio handling.
329            r.signed_ext().write(|w| w.bits(0x01));
330            // Reset both FIFOs (ct_set bits 2/3).
331            r.ct_set().write(|w| w.bits(0x0C));
332        }
333    }
334
335    /// Enable TX.
336    #[instability::unstable]
337    pub fn enable_tx(&mut self) {
338        unsafe {
339            self.regs().ct_set().write(|w| w.bits(0x01)); // tx_en
340        }
341    }
342
343    /// Enable RX.
344    #[instability::unstable]
345    pub fn enable_rx(&mut self) {
346        unsafe {
347            self.regs().ct_set().write(|w| w.bits(0x02)); // rx_en
348        }
349    }
350
351    /// Disable TX.
352    #[instability::unstable]
353    pub fn disable_tx(&mut self) {
354        unsafe {
355            self.regs().ct_clr().write(|w| w.bits(0x01));
356        }
357    }
358
359    /// Disable RX.
360    #[instability::unstable]
361    pub fn disable_rx(&mut self) {
362        unsafe {
363            self.regs().ct_clr().write(|w| w.bits(0x02));
364        }
365    }
366
367    /// Reset TX FIFO.
368    #[instability::unstable]
369    pub fn reset_tx(&mut self) {
370        unsafe {
371            self.regs().ct_set().write(|w| w.bits(0x04));
372        }
373    }
374
375    /// Reset RX FIFO.
376    #[instability::unstable]
377    pub fn reset_rx(&mut self) {
378        unsafe {
379            self.regs().ct_set().write(|w| w.bits(0x08));
380        }
381    }
382
383    /// Enable TX interrupt.
384    #[instability::unstable]
385    pub fn enable_tx_interrupt(&mut self) {
386        unsafe {
387            self.regs().ct_set().write(|w| w.bits(0x10));
388        }
389    }
390
391    /// Enable RX interrupt.
392    #[instability::unstable]
393    pub fn enable_rx_interrupt(&mut self) {
394        unsafe {
395            self.regs().ct_set().write(|w| w.bits(0x20));
396        }
397    }
398
399    /// Write a sample to the left TX channel.
400    #[instability::unstable]
401    pub fn write_left(&mut self, data: u32) {
402        unsafe {
403            self.regs().left_tx().write(|w| w.bits(data));
404        }
405    }
406
407    /// Write a sample to the right TX channel.
408    #[instability::unstable]
409    pub fn write_right(&mut self, data: u32) {
410        unsafe {
411            self.regs().right_tx().write(|w| w.bits(data));
412        }
413    }
414
415    /// Read a sample from the left RX channel.
416    #[instability::unstable]
417    pub fn read_left(&self) -> u32 {
418        self.regs().left_rx().read().bits()
419    }
420
421    /// Read a sample from the right RX channel.
422    #[instability::unstable]
423    pub fn read_right(&self) -> u32 {
424        self.regs().right_rx().read().bits()
425    }
426
427    /// Get TX FIFO depth (left channel).
428    #[instability::unstable]
429    pub fn tx_fifo_left_depth(&self) -> u8 {
430        (self.regs().tx_sta().read().bits() & 0xFF) as u8
431    }
432
433    /// Get TX FIFO depth (right channel).
434    #[instability::unstable]
435    pub fn tx_fifo_right_depth(&self) -> u8 {
436        ((self.regs().tx_sta().read().bits() >> 8) & 0xFF) as u8
437    }
438
439    /// Get RX FIFO depth (left channel).
440    #[instability::unstable]
441    pub fn rx_fifo_left_depth(&self) -> u8 {
442        (self.regs().rx_sta().read().bits() & 0xFF) as u8
443    }
444
445    /// Get RX FIFO depth (right channel).
446    #[instability::unstable]
447    pub fn rx_fifo_right_depth(&self) -> u8 {
448        ((self.regs().rx_sta().read().bits() >> 8) & 0xFF) as u8
449    }
450
451    /// Check interrupt status.
452    ///
453    /// Returns `(rx_int, tx_int, rx_overflow, tx_underflow)`.
454    #[instability::unstable]
455    pub fn interrupt_status(&self) -> (bool, bool, bool, bool) {
456        let sts = self.regs().intstatus().read().bits();
457        ((sts & 0x01) != 0, (sts & 0x02) != 0, (sts & 0x04) != 0, (sts & 0x08) != 0)
458    }
459
460    /// Clear all interrupts.
461    #[instability::unstable]
462    pub fn clear_interrupts(&mut self) {
463        unsafe {
464            self.regs().intclr().write(|w| w.bits(0x0F));
465        }
466    }
467
468    /// Set the interrupt mask.
469    ///
470    /// * `rx_int_mask` — Mask RX interrupt
471    /// * `tx_int_mask` — Mask TX interrupt
472    /// * `rx_overflow_mask` — Mask RX overflow interrupt
473    /// * `tx_underflow_mask` — Mask TX underflow interrupt
474    #[instability::unstable]
475    pub fn set_interrupt_mask(
476        &mut self,
477        rx_int_mask: bool,
478        tx_int_mask: bool,
479        rx_overflow_mask: bool,
480        tx_underflow_mask: bool,
481    ) {
482        let mut val: u32 = 0;
483        if rx_int_mask {
484            val |= 0x01;
485        }
486        if tx_int_mask {
487            val |= 0x02;
488        }
489        if rx_overflow_mask {
490            val |= 0x04;
491        }
492        if tx_underflow_mask {
493            val |= 0x08;
494        }
495        unsafe {
496            self.regs().intmask().write(|w| w.bits(val));
497        }
498    }
499
500    /// Get the I2S IP version (low byte of the `version` register).
501    pub fn version(&self) -> u8 {
502        (self.regs().version().read().bits() & 0xFF) as u8
503    }
504}

506/// Bring up the I2S clock tree the way the vendor `sio_porting_clock_enable` does on
507/// WS63: release the CMU divider reset-sync, then enable the CKEN_CTL0 clk (bit 12)
508/// and bus (bit 11) gates. WITHOUT this the SIO registers do not latch — the class-C
509/// "construct → clocked" guarantee.
510///
511/// The addresses are WS63-specific (CMU/CLDO_CRG), so the body is gated to
512/// `chip-ws63`; on BS2X this is a no-op (their SIO clock tree differs).
513#[cfg(feature = "chip-ws63")]
514fn enable_i2s_clock() {
515    let cmu = unsafe { &*crate::soc::pac::Cmu::ptr() };
516    let cldo = unsafe { &*crate::peripherals::CldoCrg::ptr() };
517    cmu.cmu_new_cfg0().modify(|_, w| w.cmu_div_ad_rstn_sync().set_bit());
518    cldo.cken_ctl0().modify(|_, w| w.i2s_cken().set_bit().i2s_bus_cken().set_bit());
519}
520
521#[cfg(not(feature = "chip-ws63"))]
522fn enable_i2s_clock() {}

577        assert_eq!(data_width_bits(DataWidth::Bits16), 0x09);
578        // 32-bit (code 5): 0b101 | (0b101 << 3) = 0x2D.
579        assert_eq!(data_width_bits(DataWidth::Bits32), 0x2D);
580    }

583    fn derived_dividers_known_value() {
584        // 16-bit stereo: dw·ch = 32. fs_num=32, fs_ratio=16,
585        // bclk = round(12288 / (32·32)) = round(12.0) = 12.
586        let (bclk, fs_num, fs_ratio) = derive_dividers(DataWidth::Bits16, ChannelCount::Two);

588        assert_eq!(fs_ratio, 16);

592    /// Exhaustive proof: for EVERY (data_width, channels) pair the derived dividers
593    /// fit their hardware field widths — bclk 7-bit, fs_num 10-bit, fs_ratio 11-bit —
594    /// and are all non-zero. This is what makes a zero/overflowing master divider
595    /// unrepresentable instead of silently `& mask`-truncated.
596    #[test]
597    fn derived_dividers_in_range() {
598        let widths = [DataWidth::Bits16, DataWidth::Bits18, DataWidth::Bits20, DataWidth::Bits24, DataWidth::Bits32];
599        let chans = [ChannelCount::Two, ChannelCount::Four, ChannelCount::Eight, ChannelCount::Sixteen];
600        for &dw in &widths {
601            for &ch in &chans {
602                let (bclk, fs_num, fs_ratio) = derive_dividers(dw, ch);
603                assert!(bclk >= 1 && bclk <= 0x7F, "bclk {bclk} out of 7-bit range for {dw:?}/{ch:?}");
604                assert!(fs_num >= 1 && fs_num <= 0x3FF, "fs_num {fs_num} out of 10-bit range");
605                assert!(fs_ratio >= 1 && fs_ratio <= 0x7FF, "fs_ratio {fs_ratio} out of 11-bit range");
606            }
607        }

612    fn encode_int_mask(rx: bool, tx: bool, rx_ovf: bool, tx_unf: bool) -> u32 {
613        let mut val: u32 = 0;
614        if rx {
615            val |= 0x01;
616        }
617        if tx {
618            val |= 0x02;
619        }
620        if rx_ovf {
621            val |= 0x04;
622        }
623        if tx_unf {
624            val |= 0x08;
625        }
626        val
627    }
628    fn decode_int_status(sts: u32) -> (bool, bool, bool, bool) {
629        ((sts & 0x01) != 0, (sts & 0x02) != 0, (sts & 0x04) != 0, (sts & 0x08) != 0)
630    }
