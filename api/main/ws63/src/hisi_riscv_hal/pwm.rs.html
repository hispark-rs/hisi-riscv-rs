1//! PWM driver for WS63 (8 channels; usable period is 16-bit on silicon — see
2//! [`PwmPeriod`]).
3//!
4//! ## Two-layer API: typed config + embedded-hal operations
5//!
6//! The configuration surface is **typed so that an unrunnable config is
7//! unrepresentable**: [`Duty`] cannot hold a value above 100 %, and a
8//! [`PwmPeriod`] is always a non-zero 32-bit counter value (and `try_from_hz`
9//! rejects frequencies that would round to a 0 count). `configure(period, duty)`
10//! therefore can never be handed garbage — there is no "compiles but writes a
11//! dead waveform" path. `configure` also brings up the PWM clock tree itself (the
12//! precondition the old `freq: u32` API silently assumed; without it the 32-bit
13//! `pwm_freq`/`pwm_duty` registers do not fully latch).
14//!
15//! The *operational* surface stays standard `embedded-hal`
16//! ([`embedded_hal::pwm::SetDutyCycle`]): its trait signatures are fixed
17//! (`u16` + `Result`), so it cannot take the typed values — that is the
18//! embedded-hal idiom and the two layers coexist (typed config for direct HAL
19//! users, the trait for generic drivers).
20use crate::peripherals::Pwm;
21use crate::soc::chip::SYSTEM_CLOCK_HZ;
22use core::marker::PhantomData;
23
24/// PWM counter clock rate. The vendor `pwm_port_clock_enable` on the default WS63
25/// build (`CONFIG_HIGH_FREQUENCY`) selects the high-frequency source (CLK_SEL
26/// bit 7) and divides it by 6 (`PWM_DIV_6`), so the counter ticks at
27/// `SYSTEM_CLOCK_HZ / 6` — **not** the raw CPU clock. Period/duty counts are in
28/// these ticks.
29///
30/// NOTE: the high-frequency source is taken as `SYSTEM_CLOCK_HZ` (240 MHz) per the
31/// vendor default; the resulting 40 MHz tick rate is not yet scope-confirmed.
32/// [`PwmPeriod::from_count`] is exact regardless; [`PwmPeriod::try_from_hz`]
33/// depends on this constant.
34pub const PWM_CLOCK_HZ: u32 = SYSTEM_CLOCK_HZ / 6;
35
36/// A duty-cycle percentage, validated to `0..=100` at construction.
37///
38/// Because a `Duty` cannot hold a value above 100, [`PwmChannel::configure`] can
39/// never be handed an out-of-range duty — the invalid state is unrepresentable,
40/// not caught at runtime.
41#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
42pub struct Duty(u8);
43
44impl Duty {
45    /// 0 % — output held inactive.
46    pub const ZERO: Duty = Duty(0);
47    /// 50 %.
48    pub const HALF: Duty = Duty(50);
49    /// 100 % — output held active.
50    pub const FULL: Duty = Duty(100);
51
52    /// Construct from a percentage. Returns `None` for `> 100`, so an invalid duty
53    /// can never reach the hardware.
54    pub const fn from_percent(percent: u8) -> Option<Self> {
55        if percent <= 100 { Some(Duty(percent)) } else { None }
56    }
57
58    /// The percentage (always `0..=100`).
59    pub const fn percent(self) -> u8 {
60        self.0
61    }
62}

64/// A validated PWM period: a non-zero **16-bit** count of [`PWM_CLOCK_HZ`] ticks.
65///
66/// The vendor `hal_pwm_v151_regs_def.h` documents a 32-bit period
67/// (`pwm_freq_l_j[15:0]` + `pwm_freq_h_j[15:0]`), but on WS63 silicon the high half
68/// **does not latch** — measured: writing `0x0001` to `pwm_freq_h0` reads back 0
69/// even with the full clock tree up (CKEN on, divider loaded, CLK_SEL set). So the
70/// *usable* period is 16-bit, and this newtype encodes exactly that: a period the
71/// hardware cannot actually run is **unrepresentable**, not silently truncated —
72/// the precise "compiles but won't run" failure this typed API removes.
73/// [`try_from_hz`](PwmPeriod::try_from_hz) likewise rejects a frequency whose count
74/// would exceed 16 bits.
75#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
76pub struct PwmPeriod(u16);
77
78impl PwmPeriod {
79    /// Wrap a raw counter period (in [`PWM_CLOCK_HZ`] ticks). `None` for 0.
80    pub const fn from_count(count: u16) -> Option<Self> {
81        if count == 0 { None } else { Some(PwmPeriod(count)) }
82    }
83
84    /// Derive the period from a target output frequency. Returns `None` when the
85    /// frequency is 0, above [`PWM_CLOCK_HZ`] (count rounds to 0), or so low that
86    /// the count would exceed the 16-bit hardware range.
87    pub const fn try_from_hz(hz: u32) -> Option<Self> {
88        if hz == 0 {
89            return None;
90        }
91        let count = PWM_CLOCK_HZ / hz;
92        if count == 0 || count > u16::MAX as u32 {
93            return None;
94        }
95        Some(PwmPeriod(count as u16))
96    }
97
98    /// The raw counter period (in [`PWM_CLOCK_HZ`] ticks), always `>= 1`.
99    pub const fn count(self) -> u16 {
100        self.0
101    }
102
103    /// The on-time counter value for `duty` at this period (`count * percent / 100`,
104    /// widened to `u32` so the product never overflows).
105    pub const fn duty_count(self, duty: Duty) -> u16 {
106        ((self.0 as u32 * duty.0 as u32) / 100) as u16
107    }
108}

110/// Bring up the PWM channel-0 clock tree the way the vendor `pwm_port_clock_enable`
111/// does on WS63 (high-frequency source, ÷6): select the source, enable the bus +
112/// channel clock gates, and load the divider. WITHOUT this the `pwm_freq`/`pwm_duty`
113/// registers do not fully latch (the high 16-bit halves read back 0) — the
114/// precondition the old `configure(freq, …)` silently assumed.
115///
116/// The register addresses are WS63-specific (CLDO_CRG), so the body is gated to
117/// `chip-ws63`; on BS2X this is a no-op (their PWM clock tree differs — a
118/// follow-up). Loads the PWM0 divider only (DIV_CTL3); channels 1–7 share the bus
119/// gate but their dividers live in DIV_CTL3/4/5.
120fn enable_pwm_clock() {
121    #[cfg(feature = "chip-ws63")]
122    {
123        const PWM_CKEN_FIELD: u16 = 0x1FF; // all 9 clock-enable bits
124        const PWM_DIV_6: u8 = 6; // CONFIG_HIGH_FREQUENCY default divider
125
126        let cldo = unsafe { &*crate::peripherals::CldoCrg::ptr() };
127        cldo.clk_sel().modify(|_, w| w.pwm_clk_sel().set_bit());
128        cldo.cken_ctl0().modify(|_, w| unsafe { w.pwm_cken().bits(PWM_CKEN_FIELD) });
129        // Reload the PWM0 divider: clear LOAD_DIV_EN, set the 4-bit divider,
130        // then set LOAD_DIV_EN (the rising edge latches it).
131        cldo.div_ctl3().modify(|_, w| w.pwm0_load_div_en().clear_bit());
132        cldo.div_ctl3().modify(|_, w| unsafe { w.pwm0_div1_cfg().bits(PWM_DIV_6) });
133        cldo.div_ctl3().modify(|_, w| w.pwm0_load_div_en().set_bit());
134    }
135}

211    /// The configured 32-bit period count for this channel (reassembled from the
212    /// `pwm_freq_l`/`pwm_freq_h` register halves). 0 if never configured.
213    fn period_count(&self) -> u32 {
214        let r = self.regs();
215        let (lo, hi) = match self.channel_index() {
216            0 => (r.pwm_freq_l0().read().bits(), r.pwm_freq_h0().read().bits()),
217            1 => (r.pwm_freq_l1().read().bits(), r.pwm_freq_h1().read().bits()),
218            2 => (r.pwm_freq_l2().read().bits(), r.pwm_freq_h2().read().bits()),
219            3 => (r.pwm_freq_l3().read().bits(), r.pwm_freq_h3().read().bits()),
220            4 => (r.pwm_freq_l4().read().bits(), r.pwm_freq_h4().read().bits()),
221            5 => (r.pwm_freq_l5().read().bits(), r.pwm_freq_h5().read().bits()),
222            6 => (r.pwm_freq_l6().read().bits(), r.pwm_freq_h6().read().bits()),
223            7 => (r.pwm_freq_l7().read().bits(), r.pwm_freq_h7().read().bits()),
224            _ => unreachable!(),
225        };
226        (lo & 0xFFFF) | ((hi & 0xFFFF) << 16)
227    }

229    /// Configure this channel's period and duty cycle, bringing up the PWM clock
230    /// tree first so the 32-bit `pwm_freq`/`pwm_duty` registers actually latch.
231    /// Both arguments are pre-validated ([`PwmPeriod`] is non-zero, [`Duty`] is
232    /// `0..=100`), so any call programs a real waveform — there is no invalid input
233    /// to reject at runtime.
234    pub fn configure(&mut self, period_cfg: PwmPeriod, duty_cfg: Duty) {
235        enable_pwm_clock();
236        let r = self.regs();
237        // 16-bit usable period/duty (the *_h halves do not latch on silicon); the
238        // match below still writes both halves, so the high write clears to 0.
239        let period = period_cfg.count() as u32;
240        let duty = period_cfg.duty_count(duty_cfg) as u32;
241        match self.channel_index() {
242            0 => {
243                r.pwm_freq_l0().write(|w| unsafe { w.bits(period & 0xFFFF) });
244                r.pwm_freq_h0().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
245                r.pwm_duty_l0().write(|w| unsafe { w.bits(duty & 0xFFFF) });
246                r.pwm_duty_h0().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
247            }
248            1 => {
249                r.pwm_freq_l1().write(|w| unsafe { w.bits(period & 0xFFFF) });
250                r.pwm_freq_h1().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
251                r.pwm_duty_l1().write(|w| unsafe { w.bits(duty & 0xFFFF) });
252                r.pwm_duty_h1().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
253            }
254            2 => {
255                r.pwm_freq_l2().write(|w| unsafe { w.bits(period & 0xFFFF) });
256                r.pwm_freq_h2().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
257                r.pwm_duty_l2().write(|w| unsafe { w.bits(duty & 0xFFFF) });
258                r.pwm_duty_h2().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
259            }
260            3 => {
261                r.pwm_freq_l3().write(|w| unsafe { w.bits(period & 0xFFFF) });
262                r.pwm_freq_h3().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
263                r.pwm_duty_l3().write(|w| unsafe { w.bits(duty & 0xFFFF) });
264                r.pwm_duty_h3().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
265            }
266            4 => {
267                r.pwm_freq_l4().write(|w| unsafe { w.bits(period & 0xFFFF) });
268                r.pwm_freq_h4().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
269                r.pwm_duty_l4().write(|w| unsafe { w.bits(duty & 0xFFFF) });
270                r.pwm_duty_h4().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
271            }
272            5 => {
273                r.pwm_freq_l5().write(|w| unsafe { w.bits(period & 0xFFFF) });
274                r.pwm_freq_h5().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
275                r.pwm_duty_l5().write(|w| unsafe { w.bits(duty & 0xFFFF) });
276                r.pwm_duty_h5().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
277            }
278            6 => {
279                r.pwm_freq_l6().write(|w| unsafe { w.bits(period & 0xFFFF) });
280                r.pwm_freq_h6().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });
281                r.pwm_duty_l6().write(|w| unsafe { w.bits(duty & 0xFFFF) });
282                r.pwm_duty_h6().write(|w| unsafe { w.bits((duty >> 16) & 0xFFFF) });
283            }
284            7 => {
285                r.pwm_freq_l7().write(|w| unsafe { w.bits(period & 0xFFFF) });
286                r.pwm_freq_h7().write(|w| unsafe { w.bits((period >> 16) & 0xFFFF) });

362    /// Consume the channel, leaving it **running** past this scope — the escape
363    /// hatch from the disabling [`Drop`](PwmChannel#impl-Drop-for-PwmChannel) (e.g.
364    /// a PWM backlight that must keep driving after setup returns). Returns a
365    /// [`PwmRunning`] marker so the intent is explicit and the channel is not
366    /// silently leaked.
367    #[must_use = "dropping the PwmRunning marker is fine, but assign it to make the leak intentional"]
368    #[instability::unstable]
369    pub fn into_running(self) -> PwmRunning {
370        core::mem::forget(self); // skip the disabling Drop — keep the output live
371        PwmRunning(())
372    }

382impl Drop for PwmChannel<'_> {
383    /// Scoped safety: a dropped channel stops driving its output (clears only this
384    /// channel's `pwm_enN` enable bit — never a shared clock gate). Use
385    /// [`PwmChannel::into_running`] to keep it live past the handle's scope.
386    fn drop(&mut self) {
387        self.disable();
388    }
389}

410impl embedded_hal::pwm::SetDutyCycle for PwmChannel<'_> {
411    fn max_duty_cycle(&self) -> u16 {
412        // Full scale = the configured period (the duty register is in period
413        // units), saturated into embedded-hal's u16 duty granularity.
414        self.period_count().min(u16::MAX as u32) as u16
415    }
416
417    fn set_duty_cycle(&mut self, duty: u16) -> Result<(), Self::Error> {
418        if duty > self.max_duty_cycle() {
419            return Err(PwmError::DutyOutOfRange);
420        }
421        let r = self.regs();
422        let duty_val = duty as u32;
423        match self.channel_index() {
424            0 => {
425                unsafe { r.pwm_duty_l0().write(|w| w.bits(duty_val & 0xFFFF)) };
426                unsafe { r.pwm_duty_h0().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
427            }
428            1 => {
429                unsafe { r.pwm_duty_l1().write(|w| w.bits(duty_val & 0xFFFF)) };
430                unsafe { r.pwm_duty_h1().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
431            }
432            2 => {
433                unsafe { r.pwm_duty_l2().write(|w| w.bits(duty_val & 0xFFFF)) };
434                unsafe { r.pwm_duty_h2().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
435            }
436            3 => {
437                unsafe { r.pwm_duty_l3().write(|w| w.bits(duty_val & 0xFFFF)) };
438                unsafe { r.pwm_duty_h3().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
439            }
440            4 => {
441                unsafe { r.pwm_duty_l4().write(|w| w.bits(duty_val & 0xFFFF)) };
442                unsafe { r.pwm_duty_h4().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
443            }
444            5 => {
445                unsafe { r.pwm_duty_l5().write(|w| w.bits(duty_val & 0xFFFF)) };
446                unsafe { r.pwm_duty_h5().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };
447            }
448            6 => {
449                unsafe { r.pwm_duty_l6().write(|w| w.bits(duty_val & 0xFFFF)) };
450                unsafe { r.pwm_duty_h6().write(|w| w.bits((duty_val >> 16) & 0xFFFF)) };

502        // Rejected: 0 Hz, above-clock (count rounds to 0), and too-low (count would
503        // exceed the 16-bit hardware range — 1 Hz → PWM_CLOCK_HZ count ≫ u16).
504        assert!(PwmPeriod::try_from_hz(0).is_none());
505        assert!(PwmPeriod::try_from_hz(PWM_CLOCK_HZ + 1).is_none());
506        assert!(PwmPeriod::try_from_hz(1).is_none());
507        // At the clock rate the period is exactly 1 tick.
508        assert_eq!(PwmPeriod::try_from_hz(PWM_CLOCK_HZ).unwrap().count(), 1);
509        // A normal frequency: count = PWM_CLOCK_HZ / hz (fits 16 bits).
