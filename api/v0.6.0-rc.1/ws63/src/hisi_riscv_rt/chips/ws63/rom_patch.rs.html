1//! WS63 mask-ROM instruction patch controller setup.
2
3use core::arch::asm;
4
5const INSTRUCTION_COMPARE_COUNT: usize = 192;
6const COMPARE_HEADER_WORDS: usize = 3;
7
8/// Load the post-link generated comparison table into the hardware controller.
9///
10/// A zero entry count leaves the controller untouched, so ordinary WS63
11/// firmware pays no behavioral cost for reserving the mask-ROM ABI prefix.
12///
13/// # Safety
14///
15/// Must run once after `.patch` has been relocated to ITCM and before any code
16/// that depends on a redirected mask-ROM entry point.
17#[unsafe(no_mangle)]
18#[unsafe(link_section = ".text.runtime.init")]
19pub(super) unsafe extern "C" fn __hisi_ws63_rom_patch_enable() {
20    unsafe extern "C" {
21        static __rom_patch_begin__: u32;
22        static __rom_patch_cmp_begin__: u32;
23    }
24
25    let remap = &raw const __rom_patch_begin__;
26    let compare = &raw const __rom_patch_cmp_begin__;
27    let entry_count = unsafe { compare.add(2).read() } as usize;
28    if entry_count == 0 || entry_count > INSTRUCTION_COMPARE_COUNT {
29        return;
30    }
31
32    // SAFETY: this chip adapter owns the one-time early-boot transition of the
33    // WS63 patch controller. The complete register block is SVD/PAC modeled.
34    let registers = unsafe { &*ws63_pac::RiscvPatch::ptr() };
35
36    registers.flpctrl().write(|w| {
37        w.enable()
38            .set_bit()
39            .write_protect()
40            .clear_bit()
41            .outside_1m()
42            .set_bit()
43            .load_compare0_enable()
44            .clear_bit()
45            .load_compare1_enable()
46            .clear_bit()
47    });
48
49    for (index, register) in registers.flpiacmp_iter().enumerate() {
50        let value = unsafe { compare.add(COMPARE_HEADER_WORDS + index).read() };
51        register.write(|w| unsafe { w.bits(value) });
52    }
53    registers.flprmp().write(|w| unsafe { w.bits(remap as u32) });
54    registers.flpctrl().write(|w| {
55        w.enable()
56            .set_bit()
57            .write_protect()
58            .set_bit()
59            .outside_1m()
60            .set_bit()
61            .load_compare0_enable()
62            .clear_bit()
63            .load_compare1_enable()
64            .clear_bit()
65    });
66
67    // Match the vendor `riscv_patch_init` ordering barrier after write protect.
68    unsafe { asm!("fence iorw, iorw", options(nostack)) };
69}
